25/// Reader returned by this backend.
26pub struct GithubReader {
27    backend: GithubBackend,
28    ctx: OperationContext,
29    path: String,
30}
31
32impl GithubReader {
33    pub(super) fn new(
34        backend: GithubBackend,
35        ctx: OperationContext,
36        path: &str,
37        _: OpRead,
38    ) -> Self {
39        Self {
40            backend,
41            ctx,
42            path: path.to_string(),
43        }
44    }
45}
46
47impl oio::StreamRead for GithubReader {
48    async fn open(&self, range: BytesRange) -> Result<(RpRead, Box<dyn oio::ReadStreamDyn>)> {
49        let backend = &self.backend;
50        let path = self.path.as_str();
51        let resp = backend.core.get(&self.ctx, path, range).await?;
52
53        let status = resp.status();
54
55        let (rp, stream) = match status {
56            StatusCode::OK | StatusCode::PARTIAL_CONTENT => {
57                let (part, mut body) = resp.into_parts();
58                let meta = parse_into_metadata(path, &part.headers)?;
59
60                // GitHub can ignore the Range header on authenticated requests
61                // and return the full content with 200, so slice the body
62                // client-side when the server did not honor the range.
63                if status == StatusCode::PARTIAL_CONTENT || range.is_full() {
64                    (
65                        RpRead::new(meta),
66                        Box::new(body) as Box<dyn oio::ReadStreamDyn>,
67                    )
68                } else {
69                    let bs = body.to_buffer().await?;
70                    let total_size = bs.len() as u64;
71                    let sliced = bs.slice(range.to_content_range(bs.len())?);
72                    let mut metadata = meta.into_builder();
73                    metadata.set_file(total_size);
74                    (
75                        RpRead::new(metadata.build()),
76                        Box::new(sliced) as Box<dyn oio::ReadStreamDyn>,
77                    )
78                }
79            }
80            _ => {
81                let (part, mut body) = resp.into_parts();
82                let buf = body.to_buffer().await?;
83                return Err(parse_error(
84                    ErrorContext::new(ServiceOperation("GetRawContent")),
85                    Response::from_parts(part, buf),
86                ));
87            }
88        };
89
90        Ok((rp, stream))
91    }
92}

94#[cfg(test)]
95mod tests {
96    use super::*;
97    use http::Request;
98    use opendal_core::raw::oio::{Read, ReadStream};
99
100    #[derive(Debug)]
101    struct FullBodyTransport;
102
103    impl HttpTransport for FullBodyTransport {
104        async fn fetch(&self, _req: Request<Buffer>) -> Result<Response<HttpBody>> {
105            Ok(Response::builder()
106                .status(StatusCode::OK)
107                .header("content-length", "10")
108                .header("content-type", "text/plain")
109                .header("etag", "\"fixture\"")
110                .body(HttpBody::new(
111                    futures::stream::iter([Ok(Buffer::from("0123456789"))]),
112                    Some(10),
113                ))
114                .expect("response must build"))
115        }
116    }
117
118    #[tokio::test]
119    async fn test_ignored_range_preserves_metadata() {
120        let service = GithubBuilder::default()
121            .owner("owner")
122            .repo("repo")
123            .token("test-token")
124            .build()
125            .expect("github service must build");
126        let ctx =
127            OperationContext::new().with_http_transport(HttpTransporter::new(FullBodyTransport));
128        let reader = service
129            .read(&ctx, "file.txt", OpRead::default())
130            .expect("reader must be created");
131
132        for (range, expected) in [
133            (BytesRange::new(2, Some(3)), b"234".as_slice()),
134            (BytesRange::suffix(3), b"789".as_slice()),
135        ] {
136            let (reply, mut stream) = reader.open(range).await.expect("range must open");
137            let metadata = reply.metadata().expect("read must return metadata");
138            assert_eq!(metadata.content_length(), 10);
139            assert_eq!(metadata.etag(), Some("\"fixture\""));
140            assert_eq!(metadata.content_type(), Some("text/plain"));
141            assert_eq!(
142                stream.read().await.expect("range must read").to_vec(),
143                expected
144            );
145            assert!(stream.read().await.expect("EOF must read").is_empty());
146        }
147    }
