53impl oio::PageList for AzfileLister {
54    async fn next_page(&self, ctx: &mut oio::PageContext) -> Result<()> {
55        let resp = self
56            .core
57            .azfile_list(&self.ctx, &self.path, &self.limit, &ctx.token)
58            .await?;
59
60        let status = resp.status();
61
62        if status != StatusCode::OK {
63            if status == StatusCode::NOT_FOUND {
64                ctx.done = true;
65                return Ok(());
66            }
67            return Err(parse_error(
68                ErrorContext::new(ServiceOperation("ListDirectoriesAndFiles")),
69                resp,
70            ));
71        }
72
73        // Return self at the first page.
74        if ctx.token.is_empty() && !ctx.done {
75            let e = oio::Entry::new(&self.path, MetadataBuilder::dir().build());
76            ctx.entries.push_back(e);
77        }
78
79        let bs = resp.into_body();
80
81        let results: EnumerationResults =
82            de::from_reader(bs.reader()).map_err(new_xml_deserialize_error)?;
83
84        if results.next_marker.is_empty() {
85            ctx.done = true;
86        } else {
87            ctx.token = results.next_marker;
88        }
89
90        for file in results.entries.file {
91            let content_length = file.properties.content_length.ok_or_else(|| {
92                Error::new(
93                    ErrorKind::Unexpected,
94                    "azfile list response does not contain file content length",
95                )
96            })?;
97            let mut meta = MetadataBuilder::file(content_length);
98            meta.etag(file.properties.etag)
99                .last_modified(Timestamp::parse_rfc2822(&file.properties.last_modified)?);
100            let path = self.path.clone().trim_start_matches('/').to_string() + &file.name;
101            ctx.entries.push_back(oio::Entry::new(&path, meta.build()));
102        }
103
104        for dir in results.entries.directory {
105            let mut meta = MetadataBuilder::dir();
106            meta.etag(dir.properties.etag)
107                .last_modified(Timestamp::parse_rfc2822(&dir.properties.last_modified)?);
108            let path = self.path.clone().trim_start_matches('/').to_string() + &dir.name + "/";
109            ctx.entries.push_back(oio::Entry::new(&path, meta.build()));
110        }
111
112        Ok(())
113    }
114}

116#[derive(Debug, Deserialize, PartialEq)]
117#[serde(rename_all = "PascalCase")]
118struct EnumerationResults {
119    marker: Option<String>,
120    prefix: Option<String>,
121    max_results: Option<u32>,
122    directory_id: Option<String>,
123    entries: Entries,
124    #[serde(default)]
125    next_marker: String,
126}
127
128#[derive(Debug, Deserialize, PartialEq)]
129#[serde(rename_all = "PascalCase")]
130struct Entries {
131    #[serde(default)]
132    file: Vec<File>,
133    #[serde(default)]
134    directory: Vec<Directory>,
135}
136
137#[derive(Debug, Deserialize, PartialEq)]
138#[serde(rename_all = "PascalCase")]
139struct File {
140    #[serde(rename = "FileId")]
141    file_id: String,
142    name: String,
143    properties: Properties,
144}
145
146#[derive(Debug, Deserialize, PartialEq)]
147#[serde(rename_all = "PascalCase")]
148struct Directory {
149    #[serde(rename = "FileId")]
150    file_id: String,
151    name: String,
152    properties: Properties,
153}
154
155#[derive(Debug, Deserialize, PartialEq)]
156#[serde(rename_all = "PascalCase")]
157struct Properties {
158    #[serde(rename = "Content-Length")]
159    content_length: Option<u64>,
160    #[serde(rename = "CreationTime")]
161    creation_time: String,
162    #[serde(rename = "LastAccessTime")]
163    last_access_time: String,
164    #[serde(rename = "LastWriteTime")]
165    last_write_time: String,
166    #[serde(rename = "ChangeTime")]
167    change_time: String,
168    #[serde(rename = "Last-Modified")]
169    last_modified: String,
170    #[serde(rename = "Etag")]
171    etag: String,
172}

180    #[test]
181    fn test_parse_list_result() {
182        let xml = r#"
183<?xml version="1.0" encoding="utf-8"?>
184<EnumerationResults ServiceEndpoint="https://myaccount.file.core.windows.net/" ShareName="myshare" ShareSnapshot="date-time" DirectoryPath="directory-path">
185  <Marker>string-value</Marker>
186  <Prefix>string-value</Prefix>
187  <MaxResults>100</MaxResults>
188  <DirectoryId>directory-id</DirectoryId>
189  <Entries>
190     <File>
191        <Name>Rust By Example.pdf</Name>
192        <FileId>13835093239654252544</FileId>
193        <Properties>
194            <Content-Length>5832374</Content-Length>
195            <CreationTime>2023-09-25T12:43:05.8483527Z</CreationTime>
196            <LastAccessTime>2023-09-25T12:43:05.8483527Z</LastAccessTime>
197            <LastWriteTime>2023-09-25T12:43:08.6337775Z</LastWriteTime>
198            <ChangeTime>2023-09-25T12:43:08.6337775Z</ChangeTime>
199            <Last-Modified>Mon, 25 Sep 2023 12:43:08 GMT</Last-Modified>
200            <Etag>\"0x8DBBDC4F8AC4AEF\"</Etag>
201        </Properties>
202    </File>
203    <Directory>
204        <Name>test_list_rich_dir</Name>
205        <FileId>12105702186650959872</FileId>
206        <Properties>
207            <CreationTime>2023-10-15T12:03:40.7194774Z</CreationTime>
208            <LastAccessTime>2023-10-15T12:03:40.7194774Z</LastAccessTime>
209            <LastWriteTime>2023-10-15T12:03:40.7194774Z</LastWriteTime>
210            <ChangeTime>2023-10-15T12:03:40.7194774Z</ChangeTime>
211            <Last-Modified>Sun, 15 Oct 2023 12:03:40 GMT</Last-Modified>
212            <Etag>\"0x8DBCD76C58C3E96\"</Etag>
213        </Properties>
214    </Directory>
215  </Entries>
216  <NextMarker />
217</EnumerationResults>
218        "#;

220        let results: EnumerationResults = from_str(xml).unwrap();

222        assert_eq!(results.entries.file[0].name, "Rust By Example.pdf");

229        assert_eq!(results.entries.directory[0].name, "test_list_rich_dir");
230
231        assert_eq!(
232            results.entries.directory[0].properties.etag,
233            "\\\"0x8DBCD76C58C3E96\\\""
234        );
