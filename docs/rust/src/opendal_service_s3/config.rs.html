29/// Config for Aws S3 and compatible services (including minio, digitalocean space,
30/// Tencent Cloud Object Storage(COS) and so on) support.
31#[derive(Default, Serialize, Deserialize, Clone, PartialEq, Eq)]
32#[serde(default)]
33#[non_exhaustive]
34pub struct S3Config {
35    /// root of this backend.
36    ///
37    /// All operations will happen under this root.
38    ///
39    /// default to `/` if not set.
40    ///
41    /// <!-- @group General -->
42    /// <!-- @default / -->
43    pub root: Option<String>,
44    /// bucket name of this backend.
45    ///
46    /// required.
47    ///
48    /// <!-- @group General -->
49    /// <!-- @example my-bucket -->
50    #[serde(alias = "aws_bucket", alias = "aws_bucket_name", alias = "bucket_name")]
51    pub bucket: String,
52    /// Deprecated: S3 versioning capability is enabled by default.
53    ///
54    /// <!-- @group Deprecated -->
55    #[deprecated(
56        since = "0.57.0",
57        note = "S3 versioning capability is enabled by default and this option is no longer needed."
58    )]
59    pub enable_versioning: bool,
60    /// endpoint of this backend.
61    ///
62    /// Endpoint must be full uri, e.g.
63    ///
64    /// - AWS S3: `https://s3.amazonaws.com` or `https://s3.{region}.amazonaws.com`
65    /// - Cloudflare R2: `https://<ACCOUNT_ID>.r2.cloudflarestorage.com`
66    /// - Aliyun OSS: `https://{region}.aliyuncs.com`
67    /// - Tencent COS: `https://cos.{region}.myqcloud.com`
68    /// - Minio: `http://127.0.0.1:9000`
69    ///
70    /// If user inputs endpoint without scheme like "s3.amazonaws.com", we
71    /// will prepend "https://" before it.
72    ///
73    /// - If endpoint is set, we will take user's input first.
74    /// - If not, we will try to load it from environment.
75    /// - If still not set, default to `https://s3.amazonaws.com`.
76    ///
77    /// <!-- @group General -->
78    /// <!-- @default https://s3.amazonaws.com -->
79    #[serde(
80        alias = "aws_endpoint",
81        alias = "aws_endpoint_url",
82        alias = "endpoint_url"
83    )]
84    pub endpoint: Option<String>,
85    /// Region represent the signing region of this endpoint. This is required
86    /// if you are using the default AWS S3 endpoint.
87    ///
88    /// If using a custom endpoint,
89    /// - If region is set, we will take user's input first.
90    /// - If not, we will try to load it from environment.
91    ///
92    /// <!-- @group General -->
93    /// <!-- @example us-east-1 -->
94    #[serde(alias = "aws_region")]
95    pub region: Option<String>,
96    /// Fallback signing region when neither `region`, `AWS_REGION`, nor
97    /// `AWS_DEFAULT_REGION` supplies one.
98    ///
99    /// <!-- @group General -->
100    #[serde(alias = "aws_default_region")]
101    pub default_region: Option<String>,
102
103    /// AWS profile.
104    ///
105    /// By default, reqsign which is the default credential provider, supplies profile in order:
106    /// - explicit option
107    /// - `AWS_PROFILE` environment variable, from which reqsign reads profile from:
108    ///    - `~/.aws/credentials` (or the path specified by `AWS_SHARED_CREDENTIALS_FILE`)
109    ///    - `~/.aws/config` (or the path specified by `AWS_CONFIG_FILE`)
110    /// <!-- @group Credentials -->
111    /// <!-- @example development -->
112    #[serde(alias = "aws_profile")]
113    pub profile: Option<String>,
114
115    /// access_key_id of this backend.
116    ///
117    /// - If access_key_id is set, we will take user's input first.
118    /// - If not, we will try to load it from environment.
119    ///
120    /// <!-- @group Credentials -->
121    #[serde(alias = "aws_access_key_id")]
122    pub access_key_id: Option<String>,
123    /// secret_access_key of this backend.
124    ///
125    /// - If secret_access_key is set, we will take user's input first.
126    /// - If not, we will try to load it from environment.
127    ///
128    /// <!-- @group Credentials -->
129    #[serde(alias = "aws_secret_access_key")]
130    pub secret_access_key: Option<String>,
131    /// session_token (aka, security token) of this backend.
132    ///
133    /// This token will expire after sometime, it's recommended to set session_token
134    /// by hand.
135    ///
136    /// <!-- @group Credentials -->
137    #[serde(alias = "aws_session_token", alias = "aws_token", alias = "token")]
138    pub session_token: Option<String>,
139    /// Relative URI for AWS container credential metadata.
140    ///
141    /// An explicit full endpoint takes precedence over this value. This value
142    /// takes precedence over the ambient full and relative container credential
143    /// URIs. OpenDAL joins it to `container_metadata_uri_override` when set,
144    /// otherwise to `http://169.254.170.2`.
145    ///
146    /// Setting any container credential field replaces the ECS slot in the
147    /// default credential chain. It does not move that slot ahead of earlier
148    /// default providers.
149    ///
150    /// <!-- @group Credentials -->
151    #[serde(alias = "aws_container_credentials_relative_uri")]
152    pub container_credentials_relative_uri: Option<String>,
153    /// Full URI for AWS container credential metadata.
154    ///
155    /// This endpoint takes precedence over the relative container credential
156    /// URI and both ambient container credential URI forms. It does not change
157    /// the position or priority of the default ECS credential provider.
158    ///
159    /// <!-- @group Credentials -->
160    #[serde(
161        alias = "container_credentials_full_uri",
162        alias = "aws_container_credentials_full_uri"
163    )]
164    pub container_credentials_endpoint: Option<String>,
165    /// Authorization token sent to the AWS container credential endpoint.
166    ///
167    /// This token takes precedence over the configured token file and the
168    /// ambient direct and file token forms.
169    ///
170    /// <!-- @group Credentials -->
171    #[serde(alias = "aws_container_authorization_token")]
172    pub container_authorization_token: Option<String>,
173    /// File containing the authorization token sent to the AWS container
174    /// credential endpoint.
175    ///
176    /// This file takes precedence over both ambient token forms. The credential
177    /// provider trims surrounding whitespace after reading the file.
178    ///
179    /// <!-- @group Credentials -->
180    #[serde(alias = "aws_container_authorization_token_file")]
181    pub container_authorization_token_file: Option<String>,
182    /// Base URI used to resolve a relative AWS container credential URI.
183    ///
184    /// This override affects relative URIs only. It does not alter a configured
185    /// or ambient full URI. The override alone does not select an endpoint.
186    ///
187    /// <!-- @group Credentials -->
188    #[serde(alias = "aws_container_metadata_uri_override")]
189    pub container_metadata_uri_override: Option<String>,
190    /// role_arn for this backend.
191    ///
192    /// If `role_arn` is set, we will use already known config as source
193    /// credential to assume role with `role_arn`.
194    ///
195    /// <!-- @group Assume role -->
196    pub role_arn: Option<String>,
197    /// external_id for this backend.
198    ///
199    /// <!-- @group Assume role -->
200    pub external_id: Option<String>,
201    /// role_session_name for this backend.
202    ///
203    /// <!-- @group Assume role -->
204    pub role_session_name: Option<String>,
205    /// assume_role_duration_seconds for this backend.
206    ///
207    /// <!-- @group Assume role -->
208    pub assume_role_duration_seconds: Option<u32>,
209    /// assume_role_session_tags for this backend.
210    ///
211    /// <!-- @group Assume role -->
212    pub assume_role_session_tags: Option<HashMap<String, String>>,
213    /// Disable automatic endpoint loading and the environment/profile
214    /// credential providers.
215    ///
216    /// This skips endpoint environment variables and removes the static
217    /// environment and shared-profile credential slots. Region environment
218    /// lookup and other credential providers, including AWS container
219    /// credentials and EC2 instance metadata, remain enabled.
220    ///
221    /// <!-- @group Credentials -->
222    pub disable_config_load: bool,
223    /// Disable load credential from ec2 metadata.
224    ///
225    /// This option is used to disable the default behavior of opendal
226    /// to load credential from ec2 metadata, a.k.a., IMDSv2
227    ///
228    /// <!-- @group Credentials -->
229    pub disable_ec2_metadata: bool,
230    /// Skip signature will skip loading credentials and signing requests.
231    ///
232    /// <!-- @group Credentials -->
233    #[serde(alias = "aws_skip_signature")]
234    pub skip_signature: bool,
235    /// Allow anonymous will allow opendal to send request without signing
236    /// when credential is not loaded.
237    ///
238    /// <!-- @group Deprecated -->
239    #[deprecated(
240        since = "0.57.0",
241        note = "Please use `skip_signature` instead of `allow_anonymous`"
242    )]
243    pub allow_anonymous: bool,
244    /// server_side_encryption for this backend.
245    ///
246    /// Available values: `AES256`, `aws:kms`.
247    ///
248    /// <!-- @group Encryption -->
249    #[serde(alias = "aws_server_side_encryption")]
250    pub server_side_encryption: Option<String>,
251    /// server_side_encryption_aws_kms_key_id for this backend
252    ///
253    /// - If `server_side_encryption` set to `aws:kms`, and `server_side_encryption_aws_kms_key_id`
254    ///   is not set, S3 will use aws managed kms key to encrypt data.
255    /// - If `server_side_encryption` set to `aws:kms`, and `server_side_encryption_aws_kms_key_id`
256    ///   is a valid kms key id, S3 will use the provided kms key to encrypt data.
257    /// - If the `server_side_encryption_aws_kms_key_id` is invalid or not found, an error will be
258    ///   returned.
259    /// - If `server_side_encryption` is not `aws:kms`, setting `server_side_encryption_aws_kms_key_id`
260    ///   is a noop.
261    ///
262    /// <!-- @group Encryption -->
263    #[serde(alias = "aws_sse_kms_key_id")]
264    pub server_side_encryption_aws_kms_key_id: Option<String>,
265    /// Control S3 Bucket Keys for SSE-KMS writes and copies.
266    ///
267    /// Omit this option to inherit the bucket setting. `false` disables Bucket
268    /// Keys for destination objects, and `true` enables them.
269    ///
270    /// <!-- @group Encryption -->
271    #[serde(alias = "aws_sse_bucket_key_enabled")]
272    pub server_side_encryption_bucket_key_enabled: Option<bool>,
273    /// server_side_encryption_customer_algorithm for this backend.
274    ///
275    /// Available values: `AES256`.
276    ///
277    /// <!-- @group Encryption -->
278    pub server_side_encryption_customer_algorithm: Option<String>,
279    /// server_side_encryption_customer_key for this backend.
280    ///
281    /// Value: BASE64-encoded key that matches algorithm specified in
282    /// `server_side_encryption_customer_algorithm`.
283    ///
284    /// <!-- @group Encryption -->
285    #[serde(alias = "aws_sse_customer_key_base64")]
286    pub server_side_encryption_customer_key: Option<String>,
287    /// Set server_side_encryption_customer_key_md5 for this backend.
288    ///
289    /// Value: MD5 digest of key specified in `server_side_encryption_customer_key`.
290    ///
291    /// <!-- @group Encryption -->
292    pub server_side_encryption_customer_key_md5: Option<String>,
293    /// default storage_class for this backend.
294    ///
295    /// Available values:
296    /// - `DEEP_ARCHIVE`
297    /// - `GLACIER`
298    /// - `GLACIER_IR`
299    /// - `INTELLIGENT_TIERING`
300    /// - `ONEZONE_IA`
301    /// - `EXPRESS_ONEZONE`
302    /// - `OUTPOSTS`
303    /// - `REDUCED_REDUNDANCY`
304    /// - `STANDARD`
305    /// - `STANDARD_IA`
306    ///
307    /// S3 compatible services don't support all of them
308    ///
309    /// <!-- @group Behavior -->
310    pub default_storage_class: Option<String>,
311    /// Enable virtual host style so that opendal will send API requests
312    /// in virtual host style instead of path style.
313    ///
314    /// - By default, opendal will send API to `https://s3.us-east-1.amazonaws.com/bucket_name`
315    /// - Enabled, opendal will send API to `https://bucket_name.s3.us-east-1.amazonaws.com`
316    ///
317    /// <!-- @group Behavior -->
318    #[serde(
319        alias = "aws_virtual_hosted_style_request",
320        alias = "virtual_hosted_style_request"
321    )]
322    pub enable_virtual_host_style: bool,
323    /// Deprecated: S3 delete batch capability is enabled by default.
324    ///
325    /// <!-- @group Deprecated -->
326    #[deprecated(
327        since = "0.57.0",
328        note = "S3 delete batch capability is enabled by default. Use CapabilityOverrideLayer to override delete_max_size for specific endpoints."
329    )]
330    pub batch_max_operations: Option<usize>,
331    /// Deprecated: S3 delete batch capability is enabled by default.
332    ///
333    /// <!-- @group Deprecated -->
334    #[deprecated(
335        since = "0.57.0",
336        note = "S3 delete batch capability is enabled by default. Use CapabilityOverrideLayer to override delete_max_size for specific endpoints."
337    )]
338    pub delete_max_size: Option<usize>,
339    /// Deprecated: S3 stat override capabilities are enabled by default.
340    ///
341    /// <!-- @group Deprecated -->
342    #[deprecated(
343        since = "0.57.0",
344        note = "S3 stat override capabilities are enabled by default. Use CapabilityOverrideLayer to override them for specific endpoints."
345    )]
346    pub disable_stat_with_override: bool,
347    /// Checksum Algorithm to use when sending checksums in HTTP headers.
348    /// This is necessary when writing to AWS S3 Buckets with Object Lock enabled for example.
349    ///
350    /// Available options:
351    /// - "crc32c"
352    /// - "md5"
353    ///
354    /// <!-- @group Behavior -->
355    #[serde(alias = "aws_checksum_algorithm")]
356    pub checksum_algorithm: Option<String>,
357    /// Deprecated: S3 write with If-Match capability is enabled by default.
358    ///
359    /// <!-- @group Deprecated -->
360    #[deprecated(
361        since = "0.57.0",
362        note = "S3 write with If-Match capability is enabled by default and this option is no longer needed."
363    )]
364    pub disable_write_with_if_match: bool,
365
366    /// Deprecated: S3 append capability is enabled by default.
367    ///
368    /// <!-- @group Deprecated -->
369    #[deprecated(
370        since = "0.57.0",
371        note = "S3 append capability is enabled by default and this option is no longer needed."
372    )]
373    pub enable_write_with_append: bool,
374
375    /// OpenDAL uses List Objects V2 by default to list objects.
376    /// However, some legacy services do not yet support V2.
377    /// This option allows users to switch back to the older List Objects V1.
378    ///
379    /// <!-- @group Behavior -->
380    pub disable_list_objects_v2: bool,
381
382    /// Indicates whether the client agrees to pay for the requests made to the S3 bucket.
383    ///
384    /// <!-- @group Behavior -->
385    #[serde(alias = "aws_request_payer", alias = "request_payer")]
386    pub enable_request_payer: bool,
387
388    /// Default ACL for new objects.
389    /// Note that some s3 services like minio do not support this option.
390    ///
391    /// <!-- @group Behavior -->
392    pub default_acl: Option<String>,
393}

395impl Debug for S3Config {
396    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
397        f.debug_struct("S3Config")
398            .field("root", &self.root)
399            .field("bucket", &self.bucket)
400            .field("endpoint", &self.endpoint)
401            .field("region", &self.region)
402            .finish_non_exhaustive()
403    }
404}
405
406impl Configurator for S3Config {
407    type Builder = S3Builder;
408
409    fn from_uri(uri: &OperatorUri) -> Result<Self> {
410        let mut map = uri.options().clone();
411
412        if let Some(name) = uri.name() {
413            map.insert("bucket".to_string(), name.to_string());
414        }
415
416        if let Some(root) = uri.root() {
417            map.insert("root".to_string(), root.to_string());
418        }
419
420        Self::from_iter(map)
421    }
422
423    #[allow(deprecated)]
424    fn into_builder(self) -> Self::Builder {
425        S3Builder {
426            config: self,
427            credential_providers: None,
428        }
429    }
430}
