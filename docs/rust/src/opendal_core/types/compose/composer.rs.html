25/// Creates one destination object from ordered complete source objects.
26///
27/// [`Composer::compose`] accepts sources incrementally and may start backend
28/// work before returning. [`Composer::close`] waits for pending work and commits
29/// the destination.
30pub struct Composer {
31    composer: oio::Composer,
32    to: String,
33    scheme: &'static str,
34    capability: Capability,
35}
36
37impl Composer {
38    pub(crate) fn create(
39        ctx: OperationContext,
40        srv: Servicer,
41        to: String,
42        args: OpCompose,
43    ) -> Result<Self> {
44        let scheme = srv.info().scheme();
45        let capability = srv.capability();
46        let composer = srv.compose(&ctx, &to, args)?;
47
48        Ok(Self {
49            composer,
50            to,
51            scheme,
52            capability,
53        })
54    }
55
56    /// Accept one complete source object in sequence.
57    ///
58    /// A successful return means the source was accepted in order. Backend work
59    /// may still be pending. This method applies backpressure when the
60    /// configured concurrency window is full.
61    pub async fn compose(&mut self, input: impl IntoComposeInput) -> Result<()> {
62        self.compose_with(input).await
63    }
64
65    /// Accept one complete source object with additional options.
66    ///
67    /// Visit [`options::ComposeSourceOptions`] for all available options.
68    pub fn compose_with(
69        &mut self,
70        input: impl IntoComposeInput,
71    ) -> FutureComposeSource<'_, impl Future<Output = Result<()>> + '_> {
72        let (path, options) = input.into_compose_input();
73        FutureComposeSource {
74            composer: self,
75            path,
76            options,
77            f: Self::compose_inner,
78        }
79    }
80
81    async fn compose_inner(
82        composer: &mut Composer,
83        path: String,
84        options: options::ComposeSourceOptions,
85    ) -> Result<()> {
86        composer.compose_options(&path, options).await
87    }
88
89    /// Accept one complete source object with explicit options.
90    ///
91    /// Visit [`options::ComposeSourceOptions`] for all available options.
92    pub async fn compose_options(
93        &mut self,
94        path: &str,
95        options: options::ComposeSourceOptions,
96    ) -> Result<()> {
97        let path = normalize_path(path);
98        if !validate_path(&path, EntryMode::FILE) {
99            return Err(
100                Error::new(ErrorKind::IsADirectory, "source path is a directory")
101                    .with_operation(Operation::Compose.into_static())
102                    .with_context("service", self.scheme)
103                    .with_context("path", path),
104            );
105        }
106        if path == self.to {
107            return Err(Error::new(
108                ErrorKind::IsSameFile,
109                "source and destination paths are same",
110            )
111            .with_operation(Operation::Compose.into_static())
112            .with_context("service", self.scheme)
113            .with_context("path", path));
114        }
115
116        let args = OpRead::from_compose_source_options(&self.capability, options)
117            .map_err(|err| err.with_context("service", self.scheme))?;
118        self.composer.compose(&path, args).await.map_err(|err| {
119            err.with_operation(Operation::Compose.into_static())
120                .with_context("service", self.scheme)
121                .with_context("from", path)
122                .with_context("to", &self.to)
123        })?;
124        Ok(())
125    }
126
127    /// Commit all accepted sources and return destination metadata.
128    ///
129    /// Closing a composer with no accepted sources returns
130    /// [`ErrorKind::ConfigInvalid`].
131    pub async fn close(&mut self) -> Result<Metadata> {
132        self.composer.close().await.map_err(|err| {
133            err.with_operation(Operation::Compose.into_static())
134                .with_context("service", self.scheme)
135                .with_context("to", &self.to)
136        })
137    }
138}

140/// [`Composer::compose_with`] returns this future.
141///
142/// Use its methods to configure one source object before awaiting it.
143pub struct FutureComposeSource<'a, F: Future<Output = Result<()>>> {
144    composer: &'a mut Composer,
145    path: String,
146    options: options::ComposeSourceOptions,
147    f: fn(&'a mut Composer, String, options::ComposeSourceOptions) -> F,
148}
149
150impl<F: Future<Output = Result<()>>> FutureComposeSource<'_, F> {
151    /// Compose this version of the source object.
152    pub fn version(mut self, value: &str) -> Self {
153        self.options.version = Some(value.to_string());
154        self
155    }
156
157    /// Require this source ETag.
158    pub fn if_match(mut self, value: &str) -> Self {
159        self.options.if_match = Some(value.to_string());
160        self
161    }
162
163    /// Require the source to retain the identity in `metadata`.
164    pub fn if_not_changed(mut self, metadata: &Metadata) -> Self {
165        self.options.if_not_changed = Some(metadata.clone());
166        self
167    }
168}
169
170impl<F: Future<Output = Result<()>>> IntoFuture for FutureComposeSource<'_, F> {
171    type Output = Result<()>;
172    type IntoFuture = F;
173
174    fn into_future(self) -> Self::IntoFuture {
175        let Self {
176            composer,
177            path,
178            options,
179            f,
180        } = self;
181        (f)(composer, path, options)
182    }
183}
