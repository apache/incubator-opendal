34pub const BUFFER_SIZE: usize = 2 * 1024 * 1024; // 2 MiB
35
36/// a boxed function that spawns task in current monoio runtime
37type TaskSpawner = Box<dyn FnOnce() + Send>;
38
39#[derive(Debug)]
40pub struct MonoiofsCore {
41    pub info: ServiceInfo,
42    pub capability: Capability,
43
44    root: PathBuf,
45    /// sender that sends [`TaskSpawner`] to worker threads
46    tx: Sender<TaskSpawner>,
47    /// join handles of worker threads
48    threads: Mutex<Vec<std::thread::JoinHandle<()>>>,
49    pub buf_pool: oio::PooledBuf,
50}
51
52impl MonoiofsCore {
53    pub fn new(root: PathBuf, worker_threads: usize, io_uring_entries: u32) -> Self {
54        // Since users use monoiofs in a context of tokio, all monoio
55        // operations need to be dispatched to a dedicated thread pool
56        // where a monoio runtime runs on each thread. Here we spawn
57        // these worker threads.
58        let (tx, rx) = flume::unbounded();
59        let threads = (0..worker_threads)
60            .map(move |i| {
61                let rx = rx.clone();
62                std::thread::Builder::new()
63                    .name(format!("monoiofs-worker-{i}"))
64                    .spawn(move || Self::worker_entrypoint(rx, io_uring_entries))
65                    .expect("spawn worker thread should success")
66            })
67            .collect();
68        let threads = Mutex::new(threads);
69
70        Self {
71            info: ServiceInfo::new(MONOIOFS_SCHEME, root.to_string_lossy(), ""),
72            capability: Capability {
73                stat: true,
74
75                read: true,
76
77                write: true,
78                write_can_append: true,
79
80                delete: true,
81                rename: true,
82                create_dir: true,
83                copy: true,
84                shared: true,
85                ..Default::default()
86            },
87            root,
88            tx,
89            threads,
90            buf_pool: oio::PooledBuf::new(16).with_initial_capacity(BUFFER_SIZE),
91        }
92    }
93
94    /// Reject `..` traversal and absolute keys so they cannot escape `root`,
95    /// matching the `fs` backend (#7684). Confinement lives here because
96    /// `normalize_path` deliberately leaves `.`/`..` unresolved (RFC 0112) and
97    /// `PathBuf::join` discards the base when the key is absolute.
98    pub fn prepare_path(&self, path: &str) -> Result<PathBuf> {
99        use std::path::Component;
100        let trimmed = path.trim_end_matches('/');
101        if Path::new(trimmed).components().any(|c| {
102            matches!(
103                c,
104                Component::ParentDir | Component::RootDir | Component::Prefix(_)
105            )
106        }) {
107            return Err(
108                Error::new(ErrorKind::NotFound, "path escapes the configured root")
109                    .with_context("path", path),
110            );
111        }
112        Ok(self.root.join(trimmed))
113    }
114
115    /// join root and path, create parent dirs
116    pub async fn prepare_write_path(&self, path: &str) -> Result<PathBuf> {
117        let path = self.prepare_path(path)?;
118        let parent = path
119            .parent()
120            .ok_or_else(|| {
121                Error::new(
122                    ErrorKind::Unexpected,
123                    "path should have parent but not, it must be malformed",
124                )
125                .with_context("input", path.to_string_lossy())
126            })?
127            .to_path_buf();
128        self.dispatch(move || monoio::fs::create_dir_all(parent))
129            .await
130            .map_err(new_std_io_error)?;
131        Ok(path)
132    }
133
134    /// entrypoint of each worker thread, sets up monoio runtimes and channels
135    fn worker_entrypoint(rx: Receiver<TaskSpawner>, io_uring_entries: u32) {
136        let mut rt = RuntimeBuilder::<FusionDriver>::new()
137            .enable_all()
138            .with_entries(io_uring_entries)
139            .build()
140            .expect("monoio runtime initialize should success");
141        // run an infinite loop that receives TaskSpawner and calls
142        // them in a context of monoio
143        rt.block_on(async {
144            while let Ok(spawner) = rx.recv_async().await {
145                spawner();
146            }
147        })
148    }
149
150    /// Create a TaskSpawner, send it to the thread pool and wait
151    /// for its result. Task panic will propagate.
152    pub async fn dispatch<F, Fut, T>(&self, f: F) -> T
153    where
154        F: FnOnce() -> Fut + 'static + Send,
155        Fut: Future<Output = T>,
156        T: 'static + Send,
157    {
158        // oneshot channel to send result back
159        let (tx, rx) = oneshot::channel();
160        let result = self
161            .tx
162            .send_async(Box::new(move || {
163                // task will be spawned on current thread, task panic
164                // will cause current worker thread panic
165                monoio::spawn(async move {
166                    // discard the result if send failed due to
167                    // MonoiofsCore::dispatch cancelled
168                    let _ = tx.send(f().await);
169                });
170            }))
171            .await;
172        self.unwrap(result);
173        self.unwrap(rx.await)
174    }
175
176    /// Create a TaskSpawner, send it to the thread pool and spawn the task.
177    pub async fn spawn<F, Fut, T>(&self, f: F)
178    where
179        F: FnOnce() -> Fut + 'static + Send,
180        Fut: Future<Output = T> + 'static,
181        T: 'static,
182    {
183        let result = self
184            .tx
185            .send_async(Box::new(move || {
186                // task will be spawned on current thread, task panic
187                // will cause current worker thread panic
188                monoio::spawn(f());
189            }))
190            .await;
191        self.unwrap(result);
192    }
193
194    /// This method always panics. It is called only when at least a
195    /// worker thread has panicked or meet a broken rx, which is
196    /// unrecoverable. It propagates worker thread's panic if there
197    /// is any and panics on normally exited thread.
198    pub fn propagate_worker_panic(&self) -> ! {
199        let mut guard = self.threads.lock().expect("worker thread has panicked");
200        // wait until the panicked thread exits
201        std::thread::sleep(Duration::from_millis(100));
202        let threads = mem::take(&mut *guard);
203        // we don't know which thread panicked, so check them one by one
204        for thread in threads {
205            if thread.is_finished() {
206                // worker thread runs an infinite loop, hence finished
207                // thread must have panicked or meet a broken rx.
208                match thread.join() {
209                    // rx is broken
210                    Ok(()) => panic!("worker thread should not exit, tx may be dropped"),
211                    // thread has panicked
212                    Err(e) => std::panic::resume_unwind(e),
213                }
214            }
215        }
216        unreachable!("this method should panic")
217    }
218
219    /// Unwrap result if result is Ok, otherwise propagates worker thread's
220    /// panic. This method facilitates panic propagation in situation where
221    /// Err returned by broken channel indicates that the worker thread has
222    /// panicked.
223    pub fn unwrap<T, E>(&self, result: Result<T, E>) -> T {
224        match result {
225            Ok(result) => result,
226            Err(_) => self.propagate_worker_panic(),
227        }
228    }
229}

240    fn new_core(worker_threads: usize) -> Arc<MonoiofsCore> {
241        Arc::new(MonoiofsCore::new(PathBuf::new(), worker_threads, 1024))
242    }

244    async fn dispatch_simple(core: Arc<MonoiofsCore>) {
245        let result = core.dispatch(|| async { 42 }).await;
246        assert_eq!(result, 42);
247        let bytes: Vec<u8> = vec![1, 2, 3, 4, 5, 6, 7, 8];
248        let bytes_clone = bytes.clone();
249        let result = core.dispatch(move || async move { bytes }).await;
250        assert_eq!(result, bytes_clone);
251    }
252
253    async fn dispatch_concurrent(core: Arc<MonoiofsCore>) {
254        let (tx, mut rx) = mpsc::unbounded();
255
256        fn spawn_task(core: Arc<MonoiofsCore>, tx: UnboundedSender<u64>, sleep_millis: u64) {
257            tokio::spawn(async move {
258                let result = core
259                    .dispatch(move || async move {
260                        monoio::time::sleep(Duration::from_millis(sleep_millis)).await;
261                        sleep_millis
262                    })
263                    .await;
264                assert_eq!(result, sleep_millis);
265                tx.send(result).unwrap();
266            });
267        }
268
269        spawn_task(core.clone(), tx.clone(), 200);
270        spawn_task(core.clone(), tx.clone(), 20);
271        drop(tx);
272        let first = rx.recv().await.ok();
273        let second = rx.recv().await.ok();
274        let third = rx.recv().await.ok();
275        assert_eq!(first, Some(20));
276        assert_eq!(second, Some(200));
277        assert_eq!(third, None);
278    }
279
280    async fn dispatch_panic(core: Arc<MonoiofsCore>) {
281        core.dispatch(|| async { panic!("BOOM") }).await;
282    }

324    #[tokio::test]
325    async fn test_prepare_path_rejects_parent_dir() {
326        let core = Arc::new(MonoiofsCore::new(PathBuf::from("/data/root"), 1, 1024));
327        for key in ["../etc/passwd", "../../etc/passwd", "a/../../b", "a/.."] {
328            let err = core.prepare_path(key).unwrap_err();
329            assert_eq!(
330                err.kind(),
331                ErrorKind::NotFound,
332                "key should be rejected: {key}"
333            );
334        }
335    }

337    #[tokio::test]
338    async fn test_prepare_path_rejects_absolute_path() {
339        let core = Arc::new(MonoiofsCore::new(PathBuf::from("/data/root"), 1, 1024));
340        // `PathBuf::join` drops the root when the key is absolute.
341        for key in ["/etc/passwd", "//etc/passwd"] {
342            let err = core.prepare_path(key).unwrap_err();
343            assert_eq!(
344                err.kind(),
345                ErrorKind::NotFound,
346                "key should be rejected: {key}"
347            );
348        }
349    }

351    #[tokio::test]
352    async fn test_prepare_path_allows_normal_keys() {
353        let core = Arc::new(MonoiofsCore::new(PathBuf::from("/data/root"), 1, 1024));
354        // Normal keys, `.` (CurDir), and trailing slashes resolve unchanged.
355        assert_eq!(
356            core.prepare_path("a/b.txt").unwrap(),
357            PathBuf::from("/data/root/a/b.txt")
358        );
359        assert_eq!(
360            core.prepare_path("a/b/").unwrap(),
361            PathBuf::from("/data/root/a/b")
362        );
363        assert_eq!(
364            core.prepare_path("./a/b").unwrap(),
365            PathBuf::from("/data/root/a/b")
366        );
367        // A key containing `..` only as a substring of a name is not a traversal.
368        assert_eq!(
369            core.prepare_path("a..b").unwrap(),
370            PathBuf::from("/data/root/a..b")
371        );
372    }
