18use opendal_core::Error;
19use opendal_core::ErrorKind;
20use opendal_core::raw::new_std_io_error;
21use smb::Status;
22
23pub(super) fn parse_smb_error(error: smb::Error) -> Error {
24    if let smb::Error::IoError(error) = error {
25        return new_std_io_error(error);
26    }
27    let kind = match &error {
28        smb::Error::UnexpectedMessageStatus(status)
29        | smb::Error::ReceivedErrorMessage(status, _) => match *status {
30            Status::U32_OBJECT_NAME_NOT_FOUND
31            | Status::U32_OBJECT_PATH_NOT_FOUND
32            // A recently deleted directory can remain pending while another
33            // handle closes; treat it as absent for stat/list/delete.
34            | 0xC0000056 => {
35                ErrorKind::NotFound
36            }
37            Status::U32_ACCESS_DENIED
38            | Status::U32_LOGON_FAILURE
39            | Status::U32_USER_ACCOUNT_LOCKED_OUT => ErrorKind::PermissionDenied,
40            Status::U32_OBJECT_NAME_COLLISION => ErrorKind::AlreadyExists,
41            Status::U32_FILE_IS_A_DIRECTORY => ErrorKind::IsADirectory,
42            // STATUS_NOT_A_DIRECTORY is not exposed by smb::Status.
43            0xC0000103 => ErrorKind::NotADirectory,
44            Status::U32_DIRECTORY_NOT_EMPTY | Status::U32_SHARING_VIOLATION => ErrorKind::Conflict,
45            Status::U32_NOT_SUPPORTED | Status::U32_NOT_IMPLEMENTED => ErrorKind::Unsupported,
46            _ => ErrorKind::Unexpected,
47        },
48        smb::Error::NotFound(_) => ErrorKind::NotFound,
49        smb::Error::MissingPermissions(_) => ErrorKind::PermissionDenied,
50        smb::Error::InvalidConfiguration(_) | smb::Error::InvalidArgument(_) => {
51            ErrorKind::ConfigInvalid
52        }
53        smb::Error::UnsupportedOperation(_) => ErrorKind::Unsupported,
54        _ => ErrorKind::Unexpected,
55    };
56    Error::new(kind, "SMB operation failed").set_source(error)
57}
