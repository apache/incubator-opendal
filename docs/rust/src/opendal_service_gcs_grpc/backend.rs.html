63impl GcsGrpcBuilder {
64    /// Set the working directory root.
65    pub fn root(mut self, root: &str) -> Self {
66        self.config.root = (!root.is_empty()).then(|| root.to_string());
67        self
68    }
69
70    /// Set the bucket name.
71    pub fn bucket(mut self, bucket: &str) -> Self {
72        self.config.bucket = bucket.to_string();
73        self
74    }
75
76    /// Set the gRPC endpoint.
77    pub fn endpoint(mut self, endpoint: &str) -> Self {
78        self.config.endpoint = (!endpoint.is_empty()).then(|| endpoint.to_string());
79        self
80    }
81
82    /// Set the Google OAuth 2.0 scope.
83    pub fn scope(mut self, scope: &str) -> Self {
84        self.config.scope = (!scope.is_empty()).then(|| scope.to_string());
85        self
86    }
87
88    /// Set the service account used by the GCE metadata server.
89    pub fn service_account(mut self, service_account: &str) -> Self {
90        self.config.service_account =
91            (!service_account.is_empty()).then(|| service_account.to_string());
92        self
93    }
94
95    /// Set a base64-encoded service account credential.
96    pub fn credential(mut self, credential: &str) -> Self {
97        self.config.credential = (!credential.is_empty()).then(|| credential.to_string());
98        self
99    }
100
101    /// Set the path to a service account credential file.
102    pub fn credential_path(mut self, path: &str) -> Self {
103        self.config.credential_path = (!path.is_empty()).then(|| path.to_string());
104        self
105    }
106
107    /// Set a custom Google credential provider.
108    ///
109    /// Implement the provider against [`opendal_core::reqsign_core`] and
110    /// [`crate::reqsign_google`] so that its types match the ones this service is
111    /// built with.
112    pub fn credential_provider(
113        mut self,
114        provider: impl ProvideCredential<Credential = Credential> + 'static,
115    ) -> Self {
116        self.credential_provider_chain = Some(
117            self.credential_provider_chain
118                .unwrap_or_default()
119                .push_front(provider),
120        );
121        self
122    }
123
124    /// Set a custom Google credential provider chain.
125    ///
126    /// Build the chain from [`opendal_core::reqsign_core`] and
127    /// [`crate::reqsign_google`] so that its types match the ones this service is
128    /// built with.
129    pub fn credential_provider_chain(mut self, chain: ProvideCredentialChain<Credential>) -> Self {
130        self.credential_provider_chain = Some(chain);
131        self
132    }
133
134    /// Set an OAuth 2.0 access token.
135    pub fn token(mut self, token: String) -> Self {
136        self.config.token = Some(token);
137        self
138    }
139
140    /// Disable the GCE metadata credential provider.
141    pub fn disable_vm_metadata(mut self) -> Self {
142        self.config.disable_vm_metadata = true;
143        self
144    }
145
146    /// Disable environment and well-known credential loading.
147    pub fn disable_config_load(mut self) -> Self {
148        self.config.disable_config_load = true;
149        self
150    }
151
152    /// Send requests without authentication.
153    pub fn skip_signature(mut self) -> Self {
154        self.config.skip_signature = true;
155        self
156    }

159impl Builder for GcsGrpcBuilder {
160    type Config = GcsGrpcConfig;
161
162    fn build(self) -> Result<impl Service> {
163        debug!("backend build started: {self:?}");
164        let root = normalize_root(&self.config.root.unwrap_or_default());
165        if self.config.bucket.is_empty() {
166            return Err(
167                Error::new(ErrorKind::ConfigInvalid, "The bucket is misconfigured")
168                    .with_operation("Builder::build")
169                    .with_context("service", GCS_GRPC_SCHEME),
170            );
171        }
172
173        let endpoint = self
174            .config
175            .endpoint
176            .clone()
177            .unwrap_or_else(|| DEFAULT_GCS_GRPC_ENDPOINT.to_string());
178        let channel_endpoint = build_endpoint(&endpoint)?;
179        let scope = self
180            .config
181            .scope
182            .clone()
183            .unwrap_or_else(|| DEFAULT_GCS_SCOPE.to_string());
184
185        let os_env = OsEnv;
186        let mut envs = os_env.vars();
187        envs.insert("GOOGLE_SCOPE".to_string(), scope.clone());
188        let ctx = Context::new()
189            .with_file_read(TokioFileRead)
190            .with_env(StaticEnv {
191                home_dir: os_env.home_dir(),
192                envs,
193            });
194
195        let mut default_credential = DefaultCredentialProvider::builder();
196        if self.config.disable_config_load {
197            default_credential = default_credential.no_env().no_well_known();
198        }
199        if self.config.disable_vm_metadata || self.config.service_account.is_some() {
200            default_credential = default_credential.no_vm_metadata();
201        }
202        let mut credential_chain = ProvideCredentialChain::new().push(default_credential.build());
203        if !self.config.disable_vm_metadata
204            && let Some(service_account) = self.config.service_account.as_deref()
205        {
206            credential_chain = credential_chain.push(
207                VmMetadataCredentialProvider::new()
208                    .with_scope(&scope)
209                    .with_service_account(service_account),
210            );
211        }
212        if let Some(path) = self.config.credential_path.as_deref() {
213            credential_chain =
214                credential_chain.push_front(FileCredentialProvider::new(path).with_scope(&scope));
215        }
216        if let Some(content) = self.config.credential.as_deref()
217            && let Ok(provider) = StaticCredentialProvider::from_base64(content)
218        {
219            credential_chain = credential_chain.push_front(provider.with_scope(&scope));
220        }
221        if let Some(token) = self.config.token.as_deref() {
222            credential_chain = credential_chain.push_front(TokenCredentialProvider::new(token));
223        }
224        if let Some(custom) = self.credential_provider_chain {
225            credential_chain = credential_chain.push_front(custom);
226        }
227
228        let signer = Signer::new(
229            ctx.clone(),
230            credential_chain,
231            RequestSigner::new("storage").with_scope(&scope),
232        );
233        let capability = Capability {
234            stat: true,
235            stat_with_version: true,
236            read: true,
237            read_with_version: true,
238            read_with_suffix: true,
239            write: true,
240            write_can_empty: true,
241            write_can_multi: true,
242            write_with_content_type: true,
243            write_with_content_disposition: true,
244            write_with_content_encoding: true,
245            write_with_cache_control: true,
246            write_with_if_not_exists: true,
247            write_with_user_metadata: true,
248            delete: true,
249            delete_with_version: true,
250            copy: true,
251            copy_with_if_not_exists: true,
252            copy_with_source_version: true,
253            list: true,
254            list_with_limit: true,
255            list_with_start_after: true,
256            list_with_recursive: true,
257            shared: true,
258            ..Default::default()
259        };
260        let bucket = self.config.bucket;
261        Ok(GcsGrpcBackend {
262            core: Arc::new(GcsGrpcCore {
263                info: ServiceInfo::new(GCS_GRPC_SCHEME, &root, &bucket),
264                capability,
265                endpoint,
266                bucket,
267                root,
268                channel_endpoint,
269                channel: Default::default(),
270                signer,
271                sign_ctx: ctx,
272                skip_signature: self.config.skip_signature,
273            }),
274        })
275    }
276}

278fn build_endpoint(endpoint: &str) -> Result<Endpoint> {
279    let mut endpoint_builder = Endpoint::from_shared(endpoint.to_string()).map_err(|err| {
280        Error::new(ErrorKind::ConfigInvalid, "invalid GCS gRPC endpoint").set_source(err)
281    })?;
282    match endpoint_builder.uri().scheme() {
283        Some(scheme) if scheme == &http::uri::Scheme::HTTPS => {
284            endpoint_builder = endpoint_builder
285                .tls_config(ClientTlsConfig::new().with_webpki_roots())
286                .map_err(|err| {
287                    Error::new(
288                        ErrorKind::ConfigInvalid,
289                        "invalid GCS gRPC TLS configuration",
290                    )
291                    .set_source(err)
292                })?;
293        }
294        Some(scheme) if scheme == &http::uri::Scheme::HTTP => {}
295        _ => {
296            return Err(Error::new(
297                ErrorKind::ConfigInvalid,
298                "GCS gRPC endpoint must use http or https",
299            ));
300        }
301    }
302    Ok(endpoint_builder)
303}

311impl Service for GcsGrpcBackend {
312    type Reader = oio::Reader;
313    type Writer = oio::Writer;
314    type Lister = oio::Lister;
315    type Deleter = oio::Deleter;
316    type Copier = oio::Copier;
317    type Composer = ();
318
319    fn info(&self) -> ServiceInfo {
320        self.core.info.clone()
321    }
322
323    fn capability(&self) -> Capability {
324        self.core.capability
325    }
326
327    async fn create_dir(
328        &self,
329        _ctx: &OperationContext,
330        _path: &str,
331        _args: OpCreateDir,
332    ) -> Result<RpCreateDir> {
333        Err(Error::new(
334            ErrorKind::Unsupported,
335            "operation is not supported",
336        ))
337    }
338
339    async fn stat(&self, ctx: &OperationContext, path: &str, args: OpStat) -> Result<RpStat> {
340        if path == "/" {
341            return Ok(RpStat::new(MetadataBuilder::dir().build()));
342        }
343        let object = self.core.object_name(path);
344        let request = GetObjectRequest {
345            bucket: self.core.bucket_resource(),
346            object,
347            generation: parse_generation(args.version())?,
348        };
349        let request = self
350            .core
351            .request(ctx, request, &[("bucket", &self.core.bucket_resource())])
352            .await?;
353        let response = self
354            .core
355            .client()
356            .get_object(request)
357            .await
358            .map_err(|status| {
359                parse_status(ErrorContext::new(ServiceOperation("GetObject")), status)
360            })?;
361        Ok(RpStat::new(parse_object(response.get_ref())))
362    }
363
364    fn read(&self, ctx: &OperationContext, path: &str, args: OpRead) -> Result<Self::Reader> {
365        Ok(Box::new(GcsGrpcReader::new(
366            self.core.clone(),
367            ctx.clone(),
368            path,
369            args,
370        )))
371    }
372
373    fn write(&self, ctx: &OperationContext, path: &str, args: OpWrite) -> Result<Self::Writer> {
374        Ok(Box::new(GcsGrpcWriter::new(
375            self.core.clone(),
376            ctx.clone(),
377            path,
378            args,
379        )))
380    }
381
382    fn delete(&self, ctx: &OperationContext) -> Result<Self::Deleter> {
383        Ok(Box::new(GcsGrpcDeleter::new(
384            self.core.clone(),
385            ctx.clone(),
386        )))
387    }
388
389    fn list(&self, ctx: &OperationContext, path: &str, args: OpList) -> Result<Self::Lister> {
390        Ok(Box::new(GcsGrpcLister::new(
391            self.core.clone(),
392            ctx.clone(),
393            path,
394            args,
395        )))
396    }
397
398    fn copy(
399        &self,
400        ctx: &OperationContext,
401        from: &str,
402        to: &str,
403        args: OpCopy,
404    ) -> Result<Self::Copier> {
405        Ok(Box::new(new_gcs_grpc_copier(
406            self.core.clone(),
407            ctx.clone(),
408            from,
409            to,
410            args,
411        )))
412    }
413
414    async fn rename(
415        &self,
416        _ctx: &OperationContext,
417        _from: &str,
418        _to: &str,
419        _args: OpRename,
420    ) -> Result<RpRename> {
421        Err(Error::new(
422            ErrorKind::Unsupported,
423            "operation is not supported",
424        ))
425    }

427    async fn presign(
428        &self,
429        _ctx: &OperationContext,
430        _path: &str,
431        _args: OpPresign,
432    ) -> Result<RpPresign> {

434            ErrorKind::Unsupported,
435            "operation is not supported",
436        ))
437    }
