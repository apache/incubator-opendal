112    /// Endpoint must be full uri, e.g.
113    ///

211    /// Enable server side encryption with customer key.
212    ///
213    /// As known as: CPK
214    ///
215    /// # Args
216    ///
217    /// `key`: Raw AES-256 key containing exactly 32 bytes.
218    /// Building an operator returns [`ErrorKind::ConfigInvalid`] if the key
219    /// has a different length.
220    ///
221    /// # Note
222    ///
223    /// Function that helps the user to set the server-side customer-provided encryption key, the key's SHA256, and the algorithm.
224    /// See [Server-side encryption with customer-provided keys (CPK)](https://learn.microsoft.com/en-us/azure/storage/blobs/encryption-customer-provided-keys)
225    /// for more info.
226    pub fn server_side_encryption_with_customer_key(mut self, key: &[u8]) -> Self {
227        // Only AES256 is supported for now
228        self.config.encryption_algorithm = Some("AES256".to_string());
229        self.config.encryption_key = Some(BASE64_STANDARD.encode(key));
230        let key_sha256 = Sha256::digest(key);
231        self.config.encryption_key_sha256 = Some(BASE64_STANDARD.encode(key_sha256));
232        self
233    }

275    /// from_connection_string will make a builder from connection string
276    ///
277    /// connection string looks like:
278    ///
279    /// ```txt
280    /// DefaultEndpointsProtocol=http;AccountName=devstoreaccount1;
281    /// AccountKey=Eby8vdM02xNOcqFlqUwJPLlmEtlCDXJ1OUzFT50uSRZ6IFsuFq2UVErCz4I6tq/K1SZFPTOtr/KBHBeksoGMGw==;
282    /// BlobEndpoint=http://127.0.0.1:10000/devstoreaccount1;
283    /// QueueEndpoint=http://127.0.0.1:10001/devstoreaccount1;
284    /// TableEndpoint=http://127.0.0.1:10002/devstoreaccount1;
285    /// ```
286    ///
287    /// Or
288    ///
289    /// ```txt
290    /// DefaultEndpointsProtocol=https;
291    /// AccountName=storagesample;
292    /// AccountKey=<account-key>;
293    /// EndpointSuffix=core.chinacloudapi.cn;
294    /// ```
295    ///
296    /// For reference: [Configure Azure Storage connection strings](https://learn.microsoft.com/en-us/azure/storage/common/storage-configure-connection-string)
297    ///
298    /// # Note
299    ///
300    /// Connection strings can only configure the endpoint, account name and
301    /// authentication information. Users still need to configure container name.
302    pub fn from_connection_string(conn: &str) -> Result<Self> {
303        let config = azure_config_from_connection_string(conn, AzureStorageService::Blob)?;
304
305        Ok(AzblobConfig::from(config).into_builder())
306    }

309impl Builder for AzblobBuilder {
310    type Config = AzblobConfig;
311
312    fn build(self) -> Result<impl Service> {
313        debug!("backend build started: {:?}", self);
314
315        let root = normalize_root(&self.config.root.unwrap_or_default());
316        debug!("backend use root {root}");
317
318        // Handle endpoint, region and container name.
319        let container = match self.config.container.is_empty() {
320            false => Ok(&self.config.container),
321            true => Err(Error::new(ErrorKind::ConfigInvalid, "container is empty")
322                .with_operation("Builder::build")
323                .with_context("service", AZBLOB_SCHEME)),
324        }?;
325        debug!("backend use container {}", container);
326
327        let endpoint = match &self.config.endpoint {
328            Some(endpoint) => Ok(endpoint.clone()),
329            None => Err(Error::new(ErrorKind::ConfigInvalid, "endpoint is empty")
330                .with_operation("Builder::build")
331                .with_context("service", AZBLOB_SCHEME)),
332        }?;
333        debug!("backend use endpoint {}", container);
334
335        let account_name = self
336            .config
337            .account_name
338            .clone()
339            .or_else(|| azure_account_name_from_endpoint(endpoint.as_str()));
340
341        if let Some(v) = &self.config.account_key {
342            // Validate that account_key can be decoded as base64
343            if let Err(e) = BASE64_STANDARD.decode(v) {
344                return Err(Error::new(
345                    ErrorKind::ConfigInvalid,
346                    format!("invalid account_key: cannot decode as base64: {e}"),
347                )
348                .with_operation("Builder::build")
349                .with_context("service", AZBLOB_SCHEME)
350                .with_context("key", "account_key"));
351            }
352        }
353
354        let encryption_key = match &self.config.encryption_key {
355            None => None,
356            Some(v) => {
357                let key = BASE64_STANDARD.decode(v).map_err(|_| {
358                    Error::new(
359                        ErrorKind::ConfigInvalid,
360                        "encryption_key must be valid Base64",
361                    )
362                    .with_operation("Builder::build")
363                    .with_context("service", AZBLOB_SCHEME)
364                    .with_context("key", "encryption_key")
365                })?;
366                if key.len() != 32 {
367                    return Err(Error::new(
368                        ErrorKind::ConfigInvalid,
369                        "encryption_key must decode to exactly 32 bytes",
370                    )
371                    .with_operation("Builder::build")
372                    .with_context("service", AZBLOB_SCHEME)
373                    .with_context("key", "encryption_key"));
374                }
375                Some(build_header_value(v).map_err(|err| {
376                    err.with_context("key", "server_side_encryption_customer_key")
377                })?)
378            }
379        };
380
381        let encryption_key_sha256 = match &self.config.encryption_key_sha256 {
382            None => None,
383            Some(v) => Some(build_header_value(v).map_err(|err| {
384                err.with_context("key", "server_side_encryption_customer_key_sha256")
385            })?),
386        };
387
388        let encryption_algorithm = match &self.config.encryption_algorithm {
389            None => None,
390            Some(v) => {
391                if v == "AES256" {
392                    Some(build_header_value(v).map_err(|err| {
393                        err.with_context("key", "server_side_encryption_customer_algorithm")
394                    })?)
395                } else {
396                    return Err(Error::new(
397                        ErrorKind::ConfigInvalid,
398                        "encryption_algorithm value must be AES256",
399                    ));
400                }
401            }
402        };
403
404        let ctx = Context::new().with_file_read(TokioFileRead).with_env(OsEnv);
405
406        let mut credential_providers =
407            ProvideCredentialChain::new().push(DefaultCredentialProvider::new());
408
409        if let (Some(account_name), Some(account_key)) =
410            (account_name.as_deref(), self.config.account_key.as_deref())
411        {
412            credential_providers = credential_providers.push_front(
413                StaticCredentialProvider::new_shared_key(account_name, account_key),
414            );
415        }
416
417        if let Some(sas_token) = self.config.sas_token.as_deref() {
418            credential_providers =
419                credential_providers.push_front(StaticCredentialProvider::new_sas_token(sas_token));
420        }
421
422        if let Some(customized_credential_chain) = self.credential_providers {
423            credential_providers = customized_credential_chain;
424        }
425
426        let signer = Signer::new(
427            ctx,
428            credential_providers,
429            RequestSigner::new().with_service_sas_permissions("racwd"),
430        );
431
432        let info = ServiceInfo::new(AZBLOB_SCHEME, &root, container);
433        let capability = Capability {
434            stat: true,
435            stat_with_if_match: true,
436            stat_with_if_none_match: true,
437
438            read: true,
439
440            read_with_if_match: true,
441            read_with_if_none_match: true,
442            read_with_override_content_disposition: true,
443            read_with_if_modified_since: true,
444            read_with_if_unmodified_since: true,
445
446            write: true,
447            write_can_append: true,
448            write_can_empty: true,
449            write_can_multi: true,
450            write_with_cache_control: true,
451            write_with_content_type: true,
452            write_with_if_match: true,
453            write_with_if_not_exists: true,
454            write_with_if_none_match: true,
455            write_with_user_metadata: true,
456
457            delete: true,
458            delete_with_if_match: true,
459            delete_with_if_none_match: true,
460            delete_max_size: Some(AZBLOB_BATCH_LIMIT),
461
462            copy: true,
463            copy_with_if_not_exists: true,
464            copy_with_if_match: true,
465            copy_with_if_none_match: true,
466            copy_can_multi: true,
467            copy_multi_min_size: Some(AZBLOB_COPY_MIN_BLOCK_SIZE),
468            copy_multi_max_size: Some(AZBLOB_COPY_MAX_BLOCK_SIZE),
469
470            list: true,
471            list_with_recursive: true,
472
473            presign: self.config.sas_token.is_some(),
474            presign_stat: self.config.sas_token.is_some(),
475            presign_read: self.config.sas_token.is_some(),
476            presign_write: self.config.sas_token.is_some(),
477
478            shared: true,
479
480            ..Default::default()
481        };
482
483        Ok(AzblobBackend {
484            core: Arc::new(AzblobCore {
485                info,
486                capability,
487                root,
488                endpoint,
489                encryption_key,
490                encryption_key_sha256,
491                encryption_algorithm,
492                container: self.config.container.clone(),
493                skip_signature: self.config.skip_signature,
494                signer,
495            }),
496        })
497    }
498}

500/// Backend for azblob services.
501#[derive(Debug, Clone)]
502pub struct AzblobBackend {
503    pub(crate) core: Arc<AzblobCore>,
504}
505
506impl Service for AzblobBackend {
507    type Reader = oio::StreamReader<AzblobReader>;
508    type Writer = AzblobWriters;
509    type Lister = oio::PageLister<AzblobLister>;
510    type Deleter = oio::BatchDeleter<AzblobDeleter>;
511    type Copier = AzblobCopiers;
512    type Composer = ();
513
514    fn info(&self) -> ServiceInfo {
515        self.core.info.clone()
516    }
517
518    fn capability(&self) -> Capability {
519        self.core.capability
520    }
521
522    async fn create_dir(
523        &self,
524        _ctx: &OperationContext,
525        _path: &str,
526        _args: OpCreateDir,
527    ) -> Result<RpCreateDir> {
528        Err(Error::new(
529            ErrorKind::Unsupported,
530            "operation is not supported",
531        ))
532    }
533
534    async fn stat(&self, ctx: &OperationContext, path: &str, args: OpStat) -> Result<RpStat> {
535        let error_ctx = ErrorContext::new(ServiceOperation("GetBlobProperties"))
536            .with_caller_condition(args.is_conditional());
537        let resp = self
538            .core
539            .azblob_get_blob_properties(ctx, path, &args)
540            .await?;
541
542        let status = resp.status();
543
544        match status {
545            StatusCode::OK => {
546                let headers = resp.headers();
547                let mut meta = parse_into_metadata(path, headers)?.into_builder();
548                if let Some(version_id) = parse_header_to_str(headers, X_MS_VERSION_ID)? {
549                    meta.version(version_id);
550                }
551
552                let user_meta = parse_prefixed_headers(headers, X_MS_META_PREFIX);
553                if !user_meta.is_empty() {
554                    meta.user_metadata(user_meta);
555                }
556
557                Ok(RpStat::new(meta.build()))
558            }
559            _ => Err(parse_error(error_ctx, resp)),
560        }
561    }
562    fn read(&self, ctx: &OperationContext, path: &str, args: OpRead) -> Result<Self::Reader> {
563        let output: oio::StreamReader<AzblobReader> = {
564            Ok(oio::StreamReader::new(AzblobReader::new(
565                self.clone(),
566                ctx.clone(),
567                path,
568                args,
569            )))
570        }?;
571
572        Ok(output)
573    }
574
575    fn write(&self, ctx: &OperationContext, path: &str, args: OpWrite) -> Result<Self::Writer> {
576        let output: AzblobWriters = {
577            let w = AzblobWriter::new(
578                self.core.clone(),
579                ctx.clone(),
580                args.clone(),
581                path.to_string(),
582            );
583            let w = if args.append() {
584                AzblobWriters::Two(oio::AppendWriter::new(w))
585            } else {
586                AzblobWriters::One(oio::BlockWriter::new(
587                    ctx.executor().clone(),
588                    w,
589                    args.concurrent(),
590                ))
591            };
592
593            Ok(w)
594        }?;
595
596        Ok(output)
597    }
598
599    fn delete(&self, ctx: &OperationContext) -> Result<Self::Deleter> {
600        let output: oio::BatchDeleter<AzblobDeleter> = {
601            Ok(oio::BatchDeleter::new(
602                AzblobDeleter::new(self.core.clone(), ctx.clone()),
603                self.core.capability.delete_max_size,
604            ))
605        }?;
606
607        Ok(output)
608    }
609
610    fn list(&self, ctx: &OperationContext, path: &str, args: OpList) -> Result<Self::Lister> {
611        let output: oio::PageLister<AzblobLister> = {
612            let l = AzblobLister::new(
613                self.core.clone(),
614                ctx.clone(),
615                path.to_string(),
616                args.recursive(),
617                args.limit(),
618            );
619
620            Ok(oio::PageLister::new(l))
621        }?;
622
623        Ok(output)
624    }
625
626    fn copy(
627        &self,
628        ctx: &OperationContext,
629        from: &str,
630        to: &str,
631        args: OpCopy,
632    ) -> Result<Self::Copier> {
633        let output: AzblobCopiers = {
634            let copier = new_azblob_copier(self.core.clone(), ctx, from, to, args)?;
635            Ok(copier)
636        }?;
637
638        Ok(output)
639    }
640
641    async fn rename(
642        &self,
643        _ctx: &OperationContext,
644        _from: &str,
645        _to: &str,
646        _args: OpRename,
647    ) -> Result<RpRename> {
648        Err(Error::new(
649            ErrorKind::Unsupported,
650            "operation is not supported",
651        ))
652    }
653
654    async fn presign(
655        &self,
656        ctx: &OperationContext,
657        path: &str,
658        args: OpPresign,
659    ) -> Result<RpPresign> {
660        let req = match args.operation() {
661            PresignOperation::Stat(v) => self.core.azblob_head_blob_request(path, v),
662            PresignOperation::Read(range, v) => self.core.azblob_get_blob_request(path, *range, v),
663            PresignOperation::Write(v) => {
664                self.core
665                    .azblob_put_blob_request(path, None, v, Buffer::new())
666            }
667            PresignOperation::Delete(_) => Err(Error::new(
668                ErrorKind::Unsupported,
669                "operation is not supported",
670            )),
671            _ => Err(Error::new(
672                ErrorKind::Unsupported,
673                "presign operation is not supported",
674            )),
675        };
676
677        let req = req?;
678        let req = self.core.sign_query(ctx, req).await?;
679
680        let (parts, _) = req.into_parts();
681
682        Ok(RpPresign::new(PresignedRequest::new(
683            parts.method,
684            parts.uri,
685            parts.headers,
686        )))
687    }
688}
