41/// [`FoyerKey`] is a key for the foyer cache. It implements foyer's [`Code`] trait
42/// directly, so the layer does not depend on foyer's `serde` feature (and its bincode
43/// dependency).
44///
45/// It's possible to specify a version in the [`OpRead`] args:
46///
47/// - If a version is given, the object is cached under that versioned key.
48/// - If version is not supplied, the object is cached exactly as returned by the backend,
49///   We do NOT interpret `None` as "latest" and we do not promote it to any other version.
50#[derive(Debug, Clone, PartialEq, Eq, Hash)]
51pub struct FoyerKey {
52    /// Object path used as the cache key.
53    pub path: String,
54    /// Object version, when the read targets a specific version.
55    pub version: Option<String>,
56}
57
58impl Code for FoyerKey {
59    fn encode(&self, writer: &mut impl std::io::Write) -> FoyerResult<()> {
60        write_bytes(writer, self.path.as_bytes())?;
61        match &self.version {
62            None => writer.write_all(&[0u8])?,
63            Some(version) => {
64                writer.write_all(&[1u8])?;
65                write_bytes(writer, version.as_bytes())?;
66            }
67        }
68        Ok(())
69    }
70
71    fn decode(reader: &mut impl std::io::Read) -> FoyerResult<Self> {
72        let path = read_string(reader)?;
73        let mut tag = [0u8; 1];
74        reader.read_exact(&mut tag)?;
75        let version = match tag[0] {
76            0 => None,
77            1 => Some(read_string(reader)?),
78            other => {
79                return Err(std::io::Error::new(
80                    std::io::ErrorKind::InvalidData,
81                    format!("invalid FoyerKey version tag: {other}"),
82                )
83                .into());
84            }
85        };
86        Ok(FoyerKey { path, version })
87    }
88
89    fn estimated_size(&self) -> usize {
90        8 + self.path.len() + 1 + self.version.as_ref().map_or(0, |v| 8 + v.len())
91    }
92}
93
94fn write_bytes(writer: &mut impl std::io::Write, bytes: &[u8]) -> FoyerResult<()> {
95    writer.write_all(&(bytes.len() as u64).to_le_bytes())?;
96    writer.write_all(bytes)?;
97    Ok(())
98}
99
100fn read_string(reader: &mut impl std::io::Read) -> FoyerResult<String> {
101    let mut len = [0u8; 8];
102    reader.read_exact(&mut len)?;
103    let len = u64::from_le_bytes(len) as usize;
104    let mut buf = vec![0u8; len];
105    reader.read_exact(&mut buf)?;
106    String::from_utf8(buf)
107        .map_err(|e| std::io::Error::new(std::io::ErrorKind::InvalidData, e).into())
108}

110/// [`FoyerValue`] stores buffered object data in Foyer.
111#[derive(Debug)]
112pub struct FoyerValue(
113    /// Buffered object data.
114    pub Buffer,
115);
116
117impl Deref for FoyerValue {
118    type Target = Buffer;
119
120    fn deref(&self) -> &Self::Target {
121        &self.0
122    }
123}
124
125impl Code for FoyerValue {
126    fn encode(&self, writer: &mut impl std::io::Write) -> FoyerResult<()> {
127        let len = self.0.len() as u64;
128        writer.write_all(&len.to_le_bytes())?;
129        std::io::copy(&mut self.0.clone(), writer)?;
130        Ok(())
131    }
132
133    fn decode(reader: &mut impl std::io::Read) -> FoyerResult<Self>
134    where
135        Self: Sized,
136    {
137        let mut len_bytes = [0u8; 8];
138        reader.read_exact(&mut len_bytes)?;
139        let len = u64::from_le_bytes(len_bytes) as usize;
140        let mut buffer = vec![0u8; len];
141        reader.read_exact(&mut buffer[..len])?;
142        Ok(FoyerValue(buffer.into()))
143    }
144
145    fn estimated_size(&self) -> usize {
146        8 + self.0.len()
147    }
148}

150/// `FoyerLayer` caches OpenDAL data with [foyer](https://github.com/foyer-rs/foyer).
151///
152/// # Operation Behavior
153/// - `write`: [`FoyerLayer`] caches data after the service completes the write.
154/// - `read`: [`FoyerLayer`] checks the cache first. On a cache miss, it reads
155///   and caches the entire object if its size is within the configured
156///   [`FoyerLayer::with_size_limit`] range, even when only a byte range is requested.
157/// - `delete`: [`FoyerLayer`] removes cached data after a successful delete when the deleter
158///   closes. A failed delete is not invalidated. Cache invalidation happens before the underlying
159///   deleter is closed, so the data remains invalidated if closing the deleter fails.
160/// - Other operations: [`FoyerLayer`] passes operations such as `list`, `copy`,
161///   and `rename` to the service without caching their results.
162///
163/// # Range Reads
164///
165/// [`FoyerLayer`] caches entire objects, not individual byte ranges. For example,
166/// `operator.read_with("some/path").range(0..1024)` returns only the first 1024
167/// bytes, but on a cache miss the layer fetches and buffers the entire object
168/// before returning that range if the object is eligible for caching. On a cache
169/// hit, the layer returns the requested range from the cached object.
170///
171/// Reading a small range from a large object can therefore transfer and buffer
172/// much more data than requested, increasing read latency and memory usage.
173/// Use [`FoyerLayer::with_size_limit`] to exclude large objects from caching.
174/// On a cache miss, objects outside the configured size range are read directly
175/// from the service using the requested byte range.
176///
177/// # Examples
178///
179/// ```no_run
180/// use opendal_core::{Operator, services::Memory};
181/// use opendal_layer_foyer::FoyerLayer;
182/// use foyer::HybridCacheBuilder;
183///
184/// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
185/// let cache = HybridCacheBuilder::new()
186///     .memory(64 * 1024 * 1024) // 64MB memory cache
187///     .with_shards(4)
188///     .storage()
189///     .build()
190///     .await?;
191///
192/// let op = Operator::new(Memory::default())?
193///     .layer(FoyerLayer::new(cache));
194/// # Ok(())
195/// # }
196/// ```
197///
198/// # Note
199///
200/// When object versioning is enabled, `FoyerLayer` treats objects with the same
201/// path but different versions as different objects.
202#[derive(Debug)]
203pub struct FoyerLayer {
204    cache: HybridCache<FoyerKey, FoyerValue>,
205    size_limit: Range<usize>,
206    deleted_keys: Arc<Mutex<HashSet<FoyerKey>>>,
207}
208
209impl FoyerLayer {
210    /// Creates a new `FoyerLayer` with the given foyer hybrid cache.
211    pub fn new(cache: HybridCache<FoyerKey, FoyerValue>) -> Self {
212        FoyerLayer {
213            cache,
214            size_limit: 0..usize::MAX,
215            deleted_keys: Arc::default(),
216        }
217    }
218
219    /// Set the object-size range eligible for caching.
220    ///
221    /// The layer reads through or writes through objects outside this range
222    /// without retaining them in Foyer. The default range accepts every
223    /// representable object size below `usize::MAX`.
224    pub fn with_size_limit<R: RangeBounds<usize>>(mut self, size_limit: R) -> Self {
225        let start = match size_limit.start_bound() {
226            Bound::Included(v) => *v,
227            Bound::Excluded(v) => *v + 1,
228            Bound::Unbounded => 0,
229        };
230        let end = match size_limit.end_bound() {
231            Bound::Included(v) => *v + 1,
232            Bound::Excluded(v) => *v,
233            Bound::Unbounded => usize::MAX,
234        };
235        self.size_limit = start..end;
236        self
237    }
238}

775    #[tokio::test]
776    async fn test_size_limit() {
777        let dir = tempfile::tempdir().unwrap();
778
779        let cache = HybridCacheBuilder::new()
780            .memory(1024 * 1024)
781            .with_shards(1)
782            .storage()
783            .with_engine_config(
784                BlockEngineConfig::new(
785                    FsDeviceBuilder::new(dir.path())
786                        .with_capacity(16 * MiB as usize)
787                        .build()
788                        .unwrap(),
789                )
790                .with_block_size(MiB as usize),
791            )
792            .with_recover_mode(RecoverMode::None)
793            .build()
794            .await
795            .unwrap();
796
797        // Set size limit: only cache files between 1KB and 10KB
798        let op = Operator::new(Memory::default())
799            .unwrap()
800            .layer(FoyerLayer::new(cache.clone()).with_size_limit(1024..10 * 1024));
801
802        let small_data = vec![1u8; 5 * 1024]; // 5KB - should be cached
803        let large_data = vec![2u8; 20 * 1024]; // 20KB - should NOT be cached
804        let tiny_data = vec![3u8; 512]; // 512B - below size limit, should NOT be cached
805
806        // Write all files
807        op.write("small.txt", small_data.clone()).await.unwrap();
808        op.write("large.txt", large_data.clone()).await.unwrap();
809        op.write("tiny.txt", tiny_data.clone()).await.unwrap();
810
811        // All should be readable
812        let read_small = op.read("small.txt").await.unwrap();
813        assert_eq!(read_small.to_vec(), small_data);
814
815        let read_large = op.read("large.txt").await.unwrap();
816        assert_eq!(read_large.to_vec(), large_data);
817
818        let read_tiny = op.read("tiny.txt").await.unwrap();
819        assert_eq!(read_tiny.to_vec(), tiny_data);
820
821        // Clear the cache to test read-through behavior
822        cache.clear().await.unwrap();
823
824        // All files should still be readable from underlying storage
825        let read_small = op.read("small.txt").await.unwrap();
826        assert_eq!(read_small.to_vec(), small_data);
827
828        let read_large = op.read("large.txt").await.unwrap();
829        assert_eq!(read_large.to_vec(), large_data);
830
831        let read_tiny = op.read("tiny.txt").await.unwrap();
832        assert_eq!(read_tiny.to_vec(), tiny_data);
833
834        // After reading, small file should be cached, but large and tiny should not
835        // We can verify this by reading with range - cached files should support range reads
836        let read_small_range = op.read_with("small.txt").range(0..1024).await.unwrap();
837        assert_eq!(read_small_range.len(), 1024);
838        assert_eq!(read_small_range.to_vec(), small_data[0..1024]);
839    }
