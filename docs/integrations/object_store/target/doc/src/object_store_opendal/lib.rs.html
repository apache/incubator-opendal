18//! object_store_opendal is an object store implementation using opendal.
19//!
20//! This crate can help you to access 30 more storage services with the same object_store API.
21//!
22//! ```no_run
23//! use std::sync::Arc;
24//!
25//! use bytes::Bytes;
26//! use object_store::path::Path;
27//! use object_store::ObjectStore;
28//! use object_store::ObjectStoreExt;
29//! use object_store_opendal::OpendalStore;
30//! use opendal::services::S3;
31//! use opendal::{Builder, Operator};
32//!
33//! #[tokio::main]
34//! async fn main() {
35//!    let builder = S3::default()
36//!     .access_key_id("my_access_key")
37//!     .secret_access_key("my_secret_key")
38//!     .endpoint("my_endpoint")
39//!     .region("my_region");
40//!
41//!     // Create a new operator
42//!     let operator = Operator::new(builder).unwrap();
43//!
44//!     // Create a new object store
45//!     let object_store = Arc::new(OpendalStore::new(operator));
46//!
47//!     let path = Path::from("data/nested/test.txt");
48//!     let bytes = Bytes::from_static(b"hello, world! I am nested.");
49//!
50//!     object_store.put(&path, bytes.clone().into()).await.unwrap();
51//!
52//!     let content = object_store
53//!         .get(&path)
54//!         .await
55//!         .unwrap()
56//!         .bytes()
57//!         .await
58//!         .unwrap();
59//!
60//!     assert_eq!(content, bytes);
61//! }
62//! ```
63//!
64//! Use the re-exported future helper when an integration needs the same OpenDAL
65//! future wrapper used by this crate:
66//!
67//! ```no_run
68//! use object_store_opendal::IntoSendFuture;
69//! use opendal::Operator;
70//!
71//! async fn rename(op: Operator) -> opendal::Result<()> {
72//!     op.rename("from", "to").into_send().await
73//! }
74//! ```

89// Make sure `send_wrapper` works as expected
90#[cfg(all(feature = "send_wrapper", test))]
91mod assert_send {
92    use object_store::{ObjectStore, ObjectStoreExt, PutPayload};
93    use opendal::Operator;
94
95    #[allow(dead_code)]
96    fn assert_send<T: Send>(_: T) {}
97
98    #[allow(dead_code)]
99    fn assertion() {
100        let op = Operator::new(opendal::services::Memory::default()).unwrap();
101        let store = super::OpendalStore::new(op);
102        assert_send(store.put(&"test".into(), PutPayload::new()));
103        assert_send(store.get(&"test".into()));
104        assert_send(store.get_range(&"test".into(), 0..1));
105        assert_send(store.head(&"test".into()));
106        assert_send(store.delete(&"test".into()));
107        assert_send(store.list(None));
108        assert_send(store.list_with_offset(None, &"test".into()));
109        assert_send(store.list_with_delimiter(None));
110    }
111}

113fn timestamp_to_datetime(ts: opendal::raw::Timestamp) -> Option<chrono::DateTime<chrono::Utc>> {
114    let jiff_ts = ts.into_inner();
115    chrono::DateTime::<chrono::Utc>::from_timestamp(
116        jiff_ts.as_second(),
117        jiff_ts.subsec_nanosecond() as u32,
118    )
119}
120
121fn datetime_to_timestamp(dt: chrono::DateTime<chrono::Utc>) -> Option<opendal::raw::Timestamp> {
122    opendal::raw::Timestamp::new(dt.timestamp(), dt.timestamp_subsec_nanos() as i32).ok()
