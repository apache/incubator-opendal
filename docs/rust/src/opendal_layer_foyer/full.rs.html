39pub struct FullReader {
40    inner: Arc<Inner>,
41    size_limit: std::ops::Range<usize>,
42    path: String,
43    args: OpRead,
44}
45
46impl FullReader {
47    pub(crate) fn new(
48        inner: Arc<Inner>,
49        size_limit: std::ops::Range<usize>,
50        path: String,
51        args: OpRead,
52    ) -> Self {
53        Self {
54            inner,
55            size_limit,
56            path,
57            args,
58        }
59    }
60
61    fn cache_key(&self) -> FoyerKey {
62        FoyerKey {
63            path: self.path.clone(),
64            version: self.args.version().map(|v| v.to_string()),
65        }
66    }
67
68    fn full_object_rp(buffer: &Buffer) -> RpRead {
69        RpRead::new({
70            let metadata = MetadataBuilder::file(buffer.len() as _);
71            metadata.build()
72        })
73    }
74
75    fn slice_full_object(buffer: &Buffer, range: BytesRange) -> Buffer {
76        buffer.slice(range.to_range_as_usize())
77    }
78
79    fn is_deleted(&self, key: &FoyerKey) -> bool {
80        self.inner
81            .deleted_keys
82            .lock()
83            .expect("deleted keys lock poisoned")
84            .contains(key)
85    }
86
87    /// Derive the `OpStat` used for cache fill from the reader's `OpRead`, so the
88    /// stat observes the same conditions (version, if-match, ...) as the request.
89    fn stat_args(&self) -> OpStat {
90        options::StatOptions {
91            version: self.args.version().map(str::to_owned),
92            if_match: self.args.if_match().map(str::to_owned),
93            if_none_match: self.args.if_none_match().map(str::to_owned),
94            if_version_match: self.args.if_version_match().map(str::to_owned),
95            if_version_not_match: self.args.if_version_not_match().map(str::to_owned),
96            if_modified_since: self.args.if_modified_since(),
97            if_unmodified_since: self.args.if_unmodified_since(),
98            ..Default::default()
99        }
100        .into()
101    }
102
103    async fn fallback_open(
104        &self,
105        range: BytesRange,
106    ) -> Result<(RpRead, Box<dyn oio::ReadStreamDyn>)> {
107        let reader = self
108            .inner
109            .srv
110            .read(&self.inner.ctx, &self.path, self.args.clone())?;
111        reader.open(range).await
112    }
113
114    async fn fallback_read(&self, range: BytesRange) -> Result<(RpRead, Buffer)> {
115        let reader = self
116            .inner
117            .srv
118            .read(&self.inner.ctx, &self.path, self.args.clone())?;
119        reader.read(range).await
120    }
121
122    async fn read_full_object(&self) -> Result<Option<Buffer>> {
123        let path_str = self.path.clone();
124
125        // Use get_or_fetch to read data from cache or fallback to remote. It can automatically
126        // handle the thundering herd problem by ensuring only one request is made for a given
127        // key.
128        //
129        // Please note that we only cache the object if it's smaller than size_limit. And we'll
130        // fetch the ENTIRE object from remote to put it into cache, then slice it to the requested
131        // range.
132        let key = self.cache_key();
133
134        if self.is_deleted(&key) {
135            return Ok(None);
136        }
137
138        let result = self
139            .inner
140            .cache
141            .get_or_fetch(&key, || {
142                let inner = self.inner.clone();
143                let size_limit = self.size_limit.clone();
144                let path_clone = path_str.clone();
145                let stat_args = self.stat_args();
146                let read_args = self.args.clone();
147                async move {
148                    // read the metadata first, if it's too large, do not cache
149                    let metadata = inner
150                        .srv
151                        .stat(&inner.ctx, &path_clone, stat_args)
152                        .await
153                        .map_err(FetchError::from_error)?
154                        .into_metadata();
155
156                    let size = metadata.content_length() as usize;
157                    if !size_limit.contains(&size) {
158                        return Err(FetchError::SizeTooLarge);
159                    }
160
161                    // fetch the ENTIRE object from remote.
162                    let reader = inner
163                        .srv
164                        .read(&inner.ctx, &path_clone, read_args)
165                        .map_err(FetchError::from_error)?;
166                    let (_, mut stream) = reader
167                        .open(BytesRange::new(0, None))
168                        .await
169                        .map_err(FetchError::from_error)?;
170                    let buffer = stream.read_all().await.map_err(FetchError::from_error)?;
171
172                    Ok(FoyerValue(buffer))
173                }
174            })
175            .await;
176
177        match result {
178            Ok(entry) => Ok(Some(entry.0.clone())),
179            Err(e) => match e.downcast_ref::<FetchError>() {
180                Some(FetchError::SizeTooLarge) => Ok(None),
181                Some(FetchError::Source { kind, message }) => {
182                    Err(Error::new(*kind, message.clone()))
183                }
184                None => Err(extract_err(e)),
185            },
186        }
187    }
188
189    /// Read data from cache or underlying storage.
190    /// Caches the ENTIRE object, then slices to requested range.
191    async fn read_range(&self, range: BytesRange) -> Result<(RpRead, Buffer)> {
192        match self.read_full_object().await? {
193            Some(buffer) => {
194                let rp = Self::full_object_rp(&buffer);
195                let buffer = Self::slice_full_object(&buffer, range);
196                Ok((rp, buffer))
197            }
198            None => self.fallback_read(range).await,
199        }
200    }
201}
202
203impl oio::Read for FullReader {
204    async fn open(&self, range: BytesRange) -> Result<(RpRead, Box<dyn oio::ReadStreamDyn>)> {
205        match self.read_full_object().await? {
206            Some(buffer) => {
207                let rp = Self::full_object_rp(&buffer);
208                let buffer = Self::slice_full_object(&buffer, range);
209                Ok((rp, Box::new(buffer) as Box<dyn oio::ReadStreamDyn>))
210            }
211            None => self.fallback_open(range).await,
212        }
213    }
214
215    async fn read(&self, range: BytesRange) -> Result<(RpRead, Buffer)> {
216        self.read_range(range).await
217    }
