18use std::sync::Arc;
19
20use super::core::GoosefsCore;
21use opendal_core::ErrorKind;
22use opendal_core::MetadataBuilder;
23use opendal_core::Result;
24use opendal_core::raw::oio::Entry;
25use opendal_core::raw::*;
26
27pub struct GoosefsLister {
28    core: Arc<GoosefsCore>,
29    path: String,
30}
31
32impl GoosefsLister {
33    pub(super) fn new(core: Arc<GoosefsCore>, path: &str) -> Self {
34        GoosefsLister {
35            core,
36            path: path.to_string(),
37        }
38    }
39}
40
41impl oio::PageList for GoosefsLister {
42    async fn next_page(&self, ctx: &mut oio::PageContext) -> Result<()> {
43        // OpenDAL's list contract (see behavior tests `list_empty_dir` /
44        // `list_nested_dir`) requires that when a caller lists an
45        // **existing** directory, the directory itself is the first
46        // entry returned. GooseFS `listStatus(dir)` — like HDFS / Alluxio
47        // — only returns the directory's children, not the directory
48        // node itself, so we synthesize the self-entry here.
49        //
50        // Importantly, the self-entry must be omitted when the target
51        // directory does NOT exist (see behavior test
52        // `list_non_exist_dir` which asserts an empty result). So we
53        // first attempt the list; only on success do we prepend the
54        // synthetic self-entry.
55        let result = self.core.list_status(&self.path).await;
56
57        match result {
58            Ok(file_infos) => {
59                ctx.done = true;
60
61                // Synthesize the self-entry for non-empty directory
62                // paths. Listing the backend root ("") is valid but has
63                // no "self" entry in OpenDAL semantics, so we skip the
64                // synthesis for it.
65                if !self.path.is_empty() && self.path.ends_with('/') {
66                    ctx.entries
67                        .push_back(Entry::new(&self.path, MetadataBuilder::dir().build()));
68                }
69
70                for file_info in file_infos {
71                    let (rel_path, metadata) = self.core.file_info_to_entry(&file_info)?;
72                    // Guard against double-emitting the self-entry if
73                    // GooseFS ever decides to include `self.path` in the
74                    // listing response (it currently doesn't, but the
75                    // semantics are not documented as stable).
76                    if rel_path == self.path {
77                        continue;
78                    }
79                    ctx.entries.push_back(Entry::new(&rel_path, metadata));
80                }
81
82                Ok(())
83            }
84            Err(e) => {
85                if e.kind() == ErrorKind::NotFound {
86                    // Non-existent directory: return an empty page
87                    // without a synthetic self-entry.
88                    ctx.done = true;
89                    return Ok(());
90                }
91                Err(e)
92            }
93        }
94    }
95}
