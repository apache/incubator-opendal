18use std::sync::Arc;
19
20use opendal_core::raw::*;
21use opendal_core::*;
22
23use super::core::RocksdbCore;
24
25pub struct RocksdbDeleter {
26    core: Arc<RocksdbCore>,
27    root: String,
28}
29
30impl RocksdbDeleter {
31    pub fn new(core: Arc<RocksdbCore>, root: String) -> Self {
32        Self { core, root }
33    }
34}
35
36impl oio::OneShotDelete for RocksdbDeleter {
37    async fn delete_once(&self, path: String, _: OpDelete) -> Result<()> {
38        let p = build_abs_path(&self.root, &path);
39        self.core.delete(&p)?;
40        Ok(())
41    }
42}
