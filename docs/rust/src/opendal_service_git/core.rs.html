122    fn get_or_create_repo(&self) -> Result<gix::Repository> {
123        let mut cell = self
124            .repo_cell
125            .lock()
126            .map_err(|_| Error::new(ErrorKind::Unexpected, "repository lock poisoned"))?;
127
128        if cell.is_none() {
129            let temp_dir = tempfile::tempdir().map_err(|_| {
130                Error::new(
131                    ErrorKind::Unexpected,
132                    "failed to create temporary git directory",
133                )
134            })?;
135
136            let username = self.username.clone();
137            let password = self.password.clone();
138            let mut prepare = gix::prepare_clone(self.repository.as_str(), temp_dir.path())
139                .map_err(|_| Error::new(ErrorKind::Unexpected, "failed to prepare git clone"))?;
140
141            if self.reference == GIT_HEAD_REF {
142                let depth = NonZeroU32::new(1).expect("1 is non-zero");
143                prepare = prepare.with_shallow(gix::remote::fetch::Shallow::DepthAtRemote(depth));
144            } else if gix::hash::ObjectId::from_hex(self.reference.as_bytes()).is_err() {
145                prepare = prepare
146                    .with_ref_name(Some(self.reference.as_str()))
147                    .map_err(|_| {
148                        Error::new(ErrorKind::ConfigInvalid, "git reference is invalid")
149                    })?;
150            }
151
152            prepare = prepare.configure_connection(
153                move |conn| -> std::result::Result<(), Box<dyn std::error::Error + Send + Sync>> {
154                    if let (Some(username), Some(password)) = (username.clone(), password.clone()) {
155                        #[expect(
156                            clippy::result_large_err,
157                            reason = "gix fixes the credential callback's error type"
158                        )]
159                        let credentials = move |action| match action {
160                            gix::credentials::helper::Action::Get(ctx) => {
161                                Ok(Some(gix::credentials::protocol::Outcome {
162                                    identity: gix::sec::identity::Account {
163                                        username: username.clone(),
164                                        password: password.clone(),
165                                        oauth_refresh_token: None,
166                                    },
167                                    next: ctx.into(),
168                                }))
169                            }
170                            _ => Ok(None),
171                        };
172                        conn.set_credentials(credentials);
173                    }
174                    Ok(())
175                },
176            );
177
178            let (repo, _) = prepare
179                .fetch_only(gix::progress::Discard, &gix::interrupt::IS_INTERRUPTED)
180                .map_err(|_| Error::new(ErrorKind::Unexpected, "failed to fetch git repository"))?;
181
182            *cell = Some(RepoHolder {
183                repo,
184                _tempdir: Arc::new(temp_dir),
185            });
186        }
187
188        cell.as_ref()
189            .map(|holder| holder.repo.clone())
190            .ok_or_else(|| Error::new(ErrorKind::Unexpected, "repository not initialized"))
191    }

193    fn resolve_reference(&self, repo: &gix::Repository) -> Result<gix::hash::ObjectId> {
194        if let Some(oid) = *self
195            .commit_oid_cache
196            .lock()
197            .map_err(|_| Error::new(ErrorKind::Unexpected, "commit cache lock poisoned"))?
198        {
199            return Ok(oid);
200        }
201
202        let object = repo
203            .rev_parse_single(self.reference.as_str())
204            .map_err(|_| Error::new(ErrorKind::NotFound, "git reference not found"))?
205            .object()
206            .map_err(|_| Error::new(ErrorKind::NotFound, "git reference object not found"))?;
207        let commit_oid = object
208            .peel_to_commit()
209            .map_err(|_| Error::new(ErrorKind::Unexpected, "git reference is not a commit"))?
210            .id()
211            .detach();
212
213        *self
214            .commit_oid_cache
215            .lock()
216            .map_err(|_| Error::new(ErrorKind::Unexpected, "commit cache lock poisoned"))? =
217            Some(commit_oid);
218        Ok(commit_oid)
219    }

221    fn get_commit_tree<'a>(
222        &self,
223        repo: &'a gix::Repository,
224        commit_oid: gix::hash::ObjectId,
225    ) -> Result<(gix::Tree<'a>, i64)> {
226        if let Some((tree_oid, commit_time)) = *self
227            .commit_info_cache
228            .lock()
229            .map_err(|_| Error::new(ErrorKind::Unexpected, "commit info cache lock poisoned"))?
230        {
231            let tree = repo
232                .find_object(tree_oid)
233                .map_err(|_| Error::new(ErrorKind::NotFound, "tree not found"))?
234                .try_into_tree()
235                .map_err(|_| Error::new(ErrorKind::Unexpected, "cached object is not a tree"))?;
236            return Ok((tree, commit_time));
237        }
238
239        let commit = repo
240            .find_object(commit_oid)
241            .map_err(|_| Error::new(ErrorKind::NotFound, "commit not found"))?
242            .peel_to_commit()
243            .map_err(|_| Error::new(ErrorKind::Unexpected, "object is not a commit"))?;
244        let commit_time = commit
245            .time()
246            .map_err(|_| Error::new(ErrorKind::Unexpected, "failed to read commit time"))?
247            .seconds;
248        let tree = commit
249            .tree()
250            .map_err(|_| Error::new(ErrorKind::Unexpected, "failed to read commit tree"))?;
251
252        *self
253            .commit_info_cache
254            .lock()
255            .map_err(|_| Error::new(ErrorKind::Unexpected, "commit info cache lock poisoned"))? =
256            Some((tree.id().detach(), commit_time));
257        Ok((tree, commit_time))
258    }

260    pub(super) fn stat_file_blocking(&self, path: &str) -> Result<(u64, bool, i64)> {
261        let repo = self.get_or_create_repo()?;
262        let commit_oid = self.resolve_reference(&repo)?;
263        let (tree, commit_time) = self.get_commit_tree(&repo, commit_oid)?;
264
265        let rooted_path = build_rooted_abs_path(&self.root, path);
266        let path = rooted_path.trim_matches('/');
267        if path.is_empty() {
268            return Ok((0, true, commit_time));
269        }
270        let entry = tree
271            .lookup_entry_by_path(path)
272            .map_err(|_| Error::new(ErrorKind::Unexpected, "failed to look up git path"))?
273            .ok_or_else(|| Error::new(ErrorKind::NotFound, "path not found"))?;
274
275        if entry.mode().is_tree() {
276            return Ok((0, true, commit_time));
277        }
278        if entry.mode().is_commit() {
279            return Err(Error::new(
280                ErrorKind::Unsupported,
281                "submodule entries are not supported",
282            ));
283        }
284
285        let blob = entry
286            .object()
287            .map_err(|_| Error::new(ErrorKind::Unexpected, "failed to read git object"))?
288            .try_into_blob()
289            .map_err(|_| Error::new(ErrorKind::Unsupported, "git entry is not a blob"))?;
290        let mut size = blob.data.len() as u64;
291        if self.resolve_lfs
292            && Self::is_lfs_pointer(&blob.data)
293            && let Ok(pointer) = Self::parse_lfs_pointer(&blob.data)
294        {
295            size = pointer.size;
296        }
297
298        Ok((size, false, commit_time))
299    }

301    pub(super) fn list_dir_blocking(&self, path: &str) -> Result<Vec<(String, bool, u64, i64)>> {
302        let repo = self.get_or_create_repo()?;
303        let commit_oid = self.resolve_reference(&repo)?;
304        let (mut tree, commit_time) = self.get_commit_tree(&repo, commit_oid)?;
305
306        let rooted_path = build_rooted_abs_path(&self.root, path);
307        let path = rooted_path.trim_matches('/');
308        if !path.is_empty() {
309            let Some(entry) = tree
310                .lookup_entry_by_path(path)
311                .map_err(|_| Error::new(ErrorKind::Unexpected, "failed to look up git path"))?
312            else {
313                return Ok(Vec::new());
314            };
315            if !entry.mode().is_tree() {
316                return Err(Error::new(
317                    ErrorKind::NotADirectory,
318                    "path is not a directory",
319                ));
320            }
321            tree = entry
322                .object()
323                .map_err(|_| Error::new(ErrorKind::Unexpected, "failed to read git object"))?
324                .try_into_tree()
325                .map_err(|_| Error::new(ErrorKind::Unexpected, "git entry is not a tree"))?;
326        }
327
328        let mut entries = Vec::new();
329        for entry in tree.iter() {
330            let entry = entry
331                .map_err(|_| Error::new(ErrorKind::Unexpected, "failed to read tree entry"))?;
332            let name = entry.filename().to_string();
333
334            if entry.mode().is_tree() {
335                entries.push((name, true, 0, commit_time));
336                continue;
337            }
338            if entry.mode().is_commit() {
339                return Err(Error::new(
340                    ErrorKind::Unsupported,
341                    "submodule entries are not supported",
342                ));
343            }
344
345            let blob = entry
346                .object()
347                .map_err(|_| Error::new(ErrorKind::Unexpected, "failed to read git object"))?
348                .try_into_blob()
349                .map_err(|_| Error::new(ErrorKind::Unsupported, "git entry is not a blob"))?;
350            let mut size = blob.data.len() as u64;
351            if self.resolve_lfs
352                && Self::is_lfs_pointer(&blob.data)
353                && let Ok(pointer) = Self::parse_lfs_pointer(&blob.data)
354            {
355                size = pointer.size;
356            }
357            entries.push((name, false, size, commit_time));
358        }
359
360        Ok(entries)
361    }

393    pub(super) fn is_lfs_pointer(content: &[u8]) -> bool {
394        let prefix_len = content.len().min(256);
395        std::str::from_utf8(&content[..prefix_len])
396            .map(|text| text.starts_with(LFS_VERSION_PREFIX))
397            .unwrap_or(false)
398    }

400    pub(super) fn parse_lfs_pointer(content: &[u8]) -> Result<LfsPointer> {
401        let text = std::str::from_utf8(content)
402            .map_err(|_| Error::new(ErrorKind::Unexpected, "invalid LFS pointer"))?;
403        let mut lines = text.lines();
404
405        if lines.next() != Some(LFS_VERSION_PREFIX) {
406            return Err(Error::new(ErrorKind::Unexpected, "invalid LFS pointer"));
407        }
408        let oid_line = lines
409            .next()
410            .ok_or_else(|| Error::new(ErrorKind::Unexpected, "invalid LFS pointer"))?;
411        let size_line = lines
412            .next()
413            .ok_or_else(|| Error::new(ErrorKind::Unexpected, "invalid LFS pointer"))?;
414        if lines.next().is_some() {
415            return Err(Error::new(ErrorKind::Unexpected, "invalid LFS pointer"));
416        }
417
418        let oid = oid_line
419            .strip_prefix(LFS_OID_PREFIX)
420            .filter(|oid| {
421                oid.len() == 64
422                    && oid
423                        .bytes()
424                        .all(|byte| byte.is_ascii_hexdigit() && !byte.is_ascii_uppercase())
425            })
426            .ok_or_else(|| Error::new(ErrorKind::Unexpected, "invalid LFS pointer"))?;
427        let size = size_line
428            .strip_prefix(LFS_SIZE_PREFIX)
429            .and_then(|size| size.parse::<u64>().ok())
430            .ok_or_else(|| Error::new(ErrorKind::Unexpected, "invalid LFS pointer"))?;
431
432        Ok(LfsPointer {
433            oid: oid.to_string(),
434            size,
435        })
436    }

438    pub(super) fn slice_blob(content: Bytes, range: BytesRange) -> Result<(RpRead, Bytes)> {
439        let size = content.len() as u64;
440        let rp = RpRead::new(MetadataBuilder::file(size).build());
441        if range.is_full() {
442            return Ok((rp, content));
443        }
444
445        let offset = range.offset();
446        if offset > size {
447            return Err(Error::new(
448                ErrorKind::RangeNotSatisfied,
449                "range start offset exceeds content length",
450            ));
451        }
452        let end = range
453            .size()
454            .map(|len| offset.saturating_add(len).min(size))
455            .unwrap_or(size);
456        Ok((rp, content.slice(offset as usize..end as usize)))
457    }

471    fn lfs_batch_url(&self) -> Result<Url> {
472        let mut url = self.repository_url()?;
473        if url.scheme() != "https" {
474            return Err(Error::new(
475                ErrorKind::ConfigInvalid,
476                "LFS resolution requires an https repository URL",
477            ));
478        }
479        if url.query().is_some() || url.fragment().is_some() {
480            return Err(Error::new(
481                ErrorKind::ConfigInvalid,
482                "repository URL must not contain a query or fragment",
483            ));
484        }
485
486        let path = url.path().trim_end_matches('/');
487        let path = if path.ends_with(".git") {
488            format!("{path}/info/lfs/objects/batch")
489        } else {
490            format!("{path}.git/info/lfs/objects/batch")
491        };
492        url.set_path(&path);
493        url.set_query(None);
494        url.set_fragment(None);
495        Ok(url)
496    }

498    pub(super) async fn resolve_lfs_action(
499        &self,
500        ctx: &OperationContext,
501        oid: &str,
502        size: u64,
503    ) -> Result<LfsAction> {
504        let batch_url = self.lfs_batch_url()?;
505        let body = serde_json::json!({
506            "operation": "download",
507            "transfers": ["basic"],
508            "objects": [{"oid": oid, "size": size}],
509        });
510        let body = serde_json::to_vec(&body)
511            .map_err(|_| Error::new(ErrorKind::Unexpected, "failed to encode LFS request"))?;
512
513        let mut builder = http::Request::builder()
514            .method(http::Method::POST)
515            .uri(batch_url.as_str())
516            .header(header::CONTENT_TYPE, LFS_CONTENT_TYPE)
517            .header(header::ACCEPT, LFS_CONTENT_TYPE);
518        if let Some(auth) = self.basic_authorization_for(&batch_url) {
519            builder = builder.header(header::AUTHORIZATION, auth);
520        }
521        let request = builder
522            .body(Buffer::from(body))
523            .map_err(|_| Error::new(ErrorKind::Unexpected, "failed to build LFS batch request"))?;
524
525        let response = ctx.http_transport().fetch(request).await?;
526        if !response.status().is_success() {
527            return Err(Error::new(
528                ErrorKind::Unexpected,
529                format!("LFS batch request failed with status {}", response.status()),
530            ));
531        }
532
533        let mut body = response.into_body();
534        let mut bufs = Vec::new();
535        let mut total = 0;
536        loop {
537            let buf = body.read().await?;
538            if buf.is_empty() {
539                break;
540            }
541            total += buf.len();
542            if total > LFS_BATCH_BODY_LIMIT {
543                return Err(Error::new(
544                    ErrorKind::Unexpected,
545                    "LFS batch response too large",
546                ));
547            }
548            bufs.push(buf);
549        }
550        let buffer: Buffer = bufs.into_iter().flatten().collect();
551        Self::parse_lfs_action(&buffer.to_bytes(), oid, size)
552    }

554    fn parse_lfs_action(body: &[u8], oid: &str, size: u64) -> Result<LfsAction> {
555        let document: Value = serde_json::from_slice(body)
556            .map_err(|_| Error::new(ErrorKind::Unexpected, "invalid LFS batch response"))?;
557        let objects = document
558            .get("objects")
559            .and_then(Value::as_array)
560            .ok_or_else(|| Error::new(ErrorKind::Unexpected, "invalid LFS batch response"))?;
561        if objects.len() != 1 {
562            return Err(Error::new(
563                ErrorKind::Unexpected,
564                "invalid LFS batch response",
565            ));
566        }
567
568        let object = &objects[0];
569        if object.get("oid").and_then(Value::as_str) != Some(oid)
570            || object.get("size").and_then(Value::as_u64) != Some(size)
571        {
572            return Err(Error::new(
573                ErrorKind::Unexpected,
574                "LFS batch response object mismatch",
575            ));
576        }
577
578        let action = object
579            .get("actions")
580            .and_then(Value::as_object)
581            .and_then(|actions| actions.get("download"))
582            .and_then(Value::as_object)
583            .ok_or_else(|| Error::new(ErrorKind::Unexpected, "LFS download action is missing"))?;
584        let href = action
585            .get("href")
586            .and_then(Value::as_str)
587            .ok_or_else(|| Error::new(ErrorKind::Unexpected, "invalid LFS download URL"))?;
588        let url = Url::parse(href)
589            .map_err(|_| Error::new(ErrorKind::Unexpected, "invalid LFS download URL"))?;
590        if url.scheme() != "https"
591            || url.host_str().is_none()
592            || !url.username().is_empty()
593            || url.password().is_some()
594        {
595            return Err(Error::new(
596                ErrorKind::Unexpected,
597                "invalid LFS download URL",
598            ));
599        }
600
601        let mut headers = HeaderMap::new();
602        match action.get("header") {
603            None | Some(Value::Null) => {}
604            Some(value) => {
605                let values = value.as_object().ok_or_else(|| {
606                    Error::new(ErrorKind::Unexpected, "invalid LFS action headers")
607                })?;
608                for (name, value) in values {
609                    let name =
610                        http::header::HeaderName::from_bytes(name.as_bytes()).map_err(|_| {
611                            Error::new(ErrorKind::Unexpected, "invalid LFS action headers")
612                        })?;
613                    let value = value.as_str().ok_or_else(|| {
614                        Error::new(ErrorKind::Unexpected, "invalid LFS action headers")
615                    })?;
616                    let value = HeaderValue::from_str(value).map_err(|_| {
617                        Error::new(ErrorKind::Unexpected, "invalid LFS action headers")
618                    })?;
619                    headers.insert(name, value);
620                }
621            }
622        }
623
624        let expires_in = action.get("expires_in").and_then(|value| {
625            value
626                .as_u64()
627                .or_else(|| value.as_str().and_then(|value| value.parse().ok()))
628        });
629        let expires_at = action.get("expires_at").and_then(Value::as_str);
630        let now = Instant::now();
631        let reuse_until = compute_reuse_until(expires_in, expires_at, now, SystemTime::now());
632
633        Ok(LfsAction {
634            url,
635            headers,
636            reuse_until,
637        })
638    }

640    pub(super) fn action_headers(&self, action: &LfsAction) -> HeaderMap {
641        let mut headers = action.headers.clone();
642        if !headers.contains_key(header::AUTHORIZATION)
643            && let Some(auth) = self.basic_authorization_for(&action.url)
644            && let Ok(value) = HeaderValue::from_str(&auth)
645        {
646            headers.insert(header::AUTHORIZATION, value);
647        }
648        headers
649    }
650
651    fn basic_authorization_for(&self, target: &Url) -> Option<String> {
652        let username = self.username.as_ref()?;
653        let password = self.password.as_ref()?;
654        let repository = self.repository_url().ok()?;
655        if target.origin() != repository.origin() {
656            return None;
657        }
658        format_authorization_by_basic(username, password).ok()
659    }
660}
661
662fn compute_reuse_until(
663    expires_in: Option<u64>,
664    expires_at: Option<&str>,
665    now: Instant,
666    now_system: SystemTime,
667) -> Instant {
668    let mut advertised = expires_in.map(Duration::from_secs);
669    if let Some(expires_at) = expires_at
670        && let Ok(timestamp) = expires_at.parse::<Timestamp>()
671    {
672        let expires_at: SystemTime = timestamp.into();
673        let duration = expires_at
674            .duration_since(now_system)
675            .unwrap_or(Duration::ZERO);
676        advertised = Some(
677            advertised
678                .map(|current| current.min(duration))
679                .unwrap_or(duration),
680        );
681    }
682
683    let lifetime = advertised
684        .map(|duration| duration.saturating_sub(LFS_ACTION_EXPIRY_MARGIN))
685        .unwrap_or(LFS_ACTION_REUSE_CAP);
686    now + lifetime.min(LFS_ACTION_REUSE_CAP)
687}
