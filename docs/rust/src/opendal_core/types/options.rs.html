25/// Options for delete operations.
26///
27/// Each condition checks the file currently stored at the delete path, and
28/// every condition must hold for the delete to proceed:
29///
30/// - A false condition fails the delete with
31///   [`crate::ErrorKind::ConditionNotMatch`].
32/// - A condition the service does not advertise through its capability fails
33///   the delete with [`crate::ErrorKind::Unsupported`]; OpenDAL never
34///   silently drops a condition. A service that cannot preserve a
35///   combination of conditions also returns `Unsupported`.
36/// - A service-side conflict unrelated to these conditions surfaces as
37///   [`crate::ErrorKind::Conflict`].
38///
39/// Once every condition holds, deleting a missing file remains a successful
40/// no-op.
41///
42/// See the [conditional operation specification][crate::docs::specs::conditional_operations]
43/// for the complete cross-operation contract.
44#[derive(Debug, Clone, Default, Eq, PartialEq)]
45pub struct DeleteOptions {
46    /// Delete the given version of the file instead of the current one.
47    ///
48    /// This selects which stored version the delete removes; it is not a
49    /// condition on the file.
50    ///
51    /// Check [`crate::Capability::delete_with_version`] before using this
52    /// option.
53    pub version: Option<String>,
54    /// Whether to delete the target recursively.
55    ///
56    /// - If `false`, behaves like the traditional single-object delete.
57    /// - If `true`, all entries under the path (or sharing the prefix for file-like paths)
58    ///   will be removed.
59    pub recursive: bool,
60    /// Delete only when the file at the delete path has this exact ETag.
61    ///
62    /// The condition succeeds when the file exists and its ETag equals this
63    /// value. A file with a different ETag, or a missing file, fails the
64    /// delete with [`crate::ErrorKind::ConditionNotMatch`]. Only concrete
65    /// ETag values are portable; a wildcard such as `"*"` has no portable
66    /// meaning here.
67    ///
68    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
69    /// service does not advertise [`crate::Capability::delete_with_if_match`].
70    pub if_match: Option<String>,
71    /// Delete only when the file at the delete path does not have this ETag.
72    ///
73    /// With a concrete ETag value, the condition succeeds when the file has
74    /// a different ETag or does not exist; deleting a missing file is then a
75    /// successful no-op. A file whose ETag equals this value fails the
76    /// delete with [`crate::ErrorKind::ConditionNotMatch`]. Only concrete
77    /// ETag values are portable; a wildcard such as `"*"` has no portable
78    /// meaning here.
79    ///
80    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
81    /// service does not advertise
82    /// [`crate::Capability::delete_with_if_none_match`].
83    pub if_none_match: Option<String>,
84    /// Delete only when the file at the delete path has this exact version.
85    ///
86    /// The condition succeeds when the file exists and its version equals
87    /// this value. A file with a different version, or a missing file, fails
88    /// the delete with [`crate::ErrorKind::ConditionNotMatch`].
89    ///
90    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
91    /// service does not advertise
92    /// [`crate::Capability::delete_with_if_version_match`].
93    pub if_version_match: Option<String>,
94    /// Delete only when the file at the delete path does not have this
95    /// version.
96    ///
97    /// The condition succeeds when the file exists with a different version.
98    /// A file whose version equals this value fails the delete with
99    /// [`crate::ErrorKind::ConditionNotMatch`]. When no file exists at the
100    /// delete path, no portable behavior is defined.
101    ///
102    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
103    /// service does not advertise
104    /// [`crate::Capability::delete_with_if_version_not_match`].
105    pub if_version_not_match: Option<String>,
106    /// Delete only when the file at the delete path still has the identity
107    /// recorded in this metadata.
108    ///
109    /// Pass metadata previously returned by OpenDAL for the same path.
110    /// OpenDAL derives a version match when the service supports version
111    /// conditions and the metadata contains a version. Otherwise it derives
112    /// an ETag match when possible. A changed or missing file fails the delete
113    /// with [`crate::ErrorKind::ConditionNotMatch`], as does combining this
114    /// option with a conflicting `if_match` or `if_version_match` value.
115    ///
116    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
117    /// service does not support the derived primitive condition, and
118    /// [`crate::ErrorKind::ConfigInvalid`] when the metadata contains neither
119    /// a version nor an ETag.
120    pub if_not_changed: Option<Metadata>,
121}

123/// Options for list operations.
124///
125/// # Groups
126/// - Traversal: `recursive`.
127/// - Pagination: `limit`, `start_after`.
128/// - Versioning: `versions`, `deleted` (effective on version-aware backends).
129
130#[derive(Debug, Clone, Default, Eq, PartialEq)]
131pub struct ListOptions {
132    /// Maximum results per request (backend hint) to control memory and throttling.
133    pub limit: Option<usize>,
134    /// The start_after passes to underlying service to specify the specified key
135    /// to start listing from.
136    pub start_after: Option<String>,
137    /// Whether to list recursively under the prefix; default `false`.
138    pub recursive: bool,
139    /// Include object versions when supported by the backend; default `false`.
140    pub versions: bool,
141    /// Include delete markers when supported by version-aware backends; default `false`.
142    pub deleted: bool,
143}

145/// Options for read operations.
146///
147/// Each condition checks the file being read, and every condition must hold
148/// for the read to proceed. A missing file fails the read with
149/// [`crate::ErrorKind::NotFound`] no matter which conditions are set. An
150/// existing file that fails a condition returns
151/// [`crate::ErrorKind::ConditionNotMatch`], and a condition the service does
152/// not advertise through its capability returns
153/// [`crate::ErrorKind::Unsupported`]; OpenDAL never silently drops a
154/// condition.

156/// See the [conditional operation specification][crate::docs::specs::conditional_operations]
157/// for the complete cross-operation contract.
158#[derive(Debug, Clone, Default, Eq, PartialEq)]
159pub struct ReadOptions {
160    /// Set `range` for this operation.
161    ///
162    /// If we have a file with size `n`.
163    ///
164    /// - `..` means read bytes in range `[0, n)` of file.
165    /// - `0..1024` and `..1024` means read bytes in range `[0, 1024)` of file
166    /// - `1024..` means read bytes in range `[1024, n)` of file
167    /// - `BytesRange::suffix(1024)` means read the last `min(1024, n)` bytes of file
168    ///
169    /// The type implements `From<RangeBounds<u64>>`, so users can use `(1024..).into()` instead.
170    pub range: BytesRange,
171    /// Read the given version of the file instead of the current one.

173    /// This selects which stored version the read returns; it is not a
174    /// condition on the file. A version that does not exist returns
175    /// [`crate::ErrorKind::NotFound`].
176    ///
177    /// Check [`crate::Capability::read_with_version`] before using this
178    /// option.
179    pub version: Option<String>,
180
181    /// Read only when the file has this exact ETag.
182    ///
183    /// The condition succeeds when the file exists and its ETag equals this
184    /// value. A file with a different ETag returns
185    /// [`crate::ErrorKind::ConditionNotMatch`]; a missing file returns
186    /// [`crate::ErrorKind::NotFound`]. Only concrete ETag values are
187    /// portable; a wildcard such as `"*"` has no portable meaning here.
188    ///
189    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
190    /// service does not advertise [`crate::Capability::read_with_if_match`].
191    pub if_match: Option<String>,
192    /// Read only when the file does not have this ETag.
193    ///
194    /// With a concrete ETag value, the condition succeeds when the file
195    /// exists with a different ETag. A file whose ETag equals this value
196    /// returns [`crate::ErrorKind::ConditionNotMatch`]; a missing file
197    /// returns [`crate::ErrorKind::NotFound`]. Only concrete ETag values
198    /// are portable; a wildcard such as `"*"` has no portable meaning here.
199    ///
200    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
201    /// service does not advertise
202    /// [`crate::Capability::read_with_if_none_match`].
203    pub if_none_match: Option<String>,
204    /// Read only when the file has this exact version.
205    ///
206    /// The condition succeeds when the file exists and its version equals
207    /// this value. A file with a different version returns
208    /// [`crate::ErrorKind::ConditionNotMatch`]; a missing file returns
209    /// [`crate::ErrorKind::NotFound`].
210    ///
211    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
212    /// service does not advertise
213    /// [`crate::Capability::read_with_if_version_match`].
214    pub if_version_match: Option<String>,
215    /// Read only when the file does not have this version.
216    ///
217    /// The condition succeeds when the file exists with a different version.
218    /// A file whose version equals this value returns
219    /// [`crate::ErrorKind::ConditionNotMatch`]; a missing file returns
220    /// [`crate::ErrorKind::NotFound`].
221    ///
222    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
223    /// service does not advertise
224    /// [`crate::Capability::read_with_if_version_not_match`].
225    pub if_version_not_match: Option<String>,
226    /// Read only when the file was modified after this timestamp.
227    ///
228    /// The condition succeeds when the file exists and was modified after
229    /// this time. A file not modified since this time returns
230    /// [`crate::ErrorKind::ConditionNotMatch`]; a missing file returns
231    /// [`crate::ErrorKind::NotFound`].
232    ///
233    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
234    /// service does not advertise
235    /// [`crate::Capability::read_with_if_modified_since`].
236    pub if_modified_since: Option<Timestamp>,
237    /// Read only when the file was not modified after this timestamp.
238    ///
239    /// The condition succeeds when the file exists and was not modified
240    /// after this time. A file modified after this time returns
241    /// [`crate::ErrorKind::ConditionNotMatch`]; a missing file returns
242    /// [`crate::ErrorKind::NotFound`].
243    ///
244    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
245    /// service does not advertise
246    /// [`crate::Capability::read_with_if_unmodified_since`].
247    pub if_unmodified_since: Option<Timestamp>,
248
249    /// Known content length of the object.
250    ///
251    /// This is an execution hint that allows OpenDAL to avoid extra metadata
252    /// requests while planning reads. It must not be used as an object identity
253    /// or consistency condition.
254    pub content_length_hint: Option<u64>,
255
256    /// Set `concurrent` for the operation.
257    ///
258    /// OpenDAL by default to read file without concurrent. This is not efficient for cases when users
259    /// read large chunks of data. By setting `concurrent`, opendal will reading files concurrently
260    /// on support storage services.
261    ///
262    /// By setting `concurrent`, opendal will fetch chunks concurrently with
263    /// the give chunk size.
264    ///
265    /// Refer to the [performance guide](https://github.com/apache/opendal/tree/main/core/core/src/docs/performance)
266    /// for more details.
267    pub concurrent: usize,
268    /// Set `chunk` for the operation.
269    ///
270    /// OpenDAL will use services' preferred chunk size by default. Users can set chunk based on their own needs.
271    ///
272    /// Refer to the [performance guide](https://github.com/apache/opendal/tree/main/core/core/src/docs/performance)
273    /// for more details.
274    pub chunk: Option<usize>,
275    /// Controls the optimization strategy for range reads in [`crate::Reader::fetch`].
276    ///
277    /// When performing range reads, if the gap between two requested ranges is less than or
278    /// equal to the configured `gap` size, OpenDAL will merge these ranges into a single read request
279    /// and discard the unrequested data in between. This helps reduce the number of API calls
280    /// to remote storage services.
281    ///
282    /// Set to `0` to disable merging ranges separated by a gap. Overlapping or adjacent ranges
283    /// are still merged.
284    ///
285    /// This optimization is particularly useful when performing multiple small range reads
286    /// that are close to each other, as it reduces the overhead of multiple network requests
287    /// at the cost of transferring some additional data.
288    ///
289    /// Refer to the [performance guide](https://github.com/apache/opendal/tree/main/core/core/src/docs/performance)
290    /// for more details.
291    pub gap: Option<usize>,
292
293    /// Specify the content-type header that should be sent back by the operation.
294    ///
295    /// This option is only meaningful when used along with presign.
296    pub override_content_type: Option<String>,
297    /// Specify the `cache-control` header that should be sent back by the operation.
298    ///
299    /// This option is only meaningful when used along with presign.
300    pub override_cache_control: Option<String>,
301    /// Specify the `content-disposition` header that should be sent back by the operation.
302    ///
303    /// This option is only meaningful when used along with presign.
304    pub override_content_disposition: Option<String>,
305}

567/// Options for write operations.
568///
569/// Each condition checks the file currently stored at the write path, and
570/// the service evaluates it atomically with the write's visible commit:
571///
572/// - A false condition fails the write with
573///   [`crate::ErrorKind::ConditionNotMatch`] and leaves the previously
574///   visible file unchanged. Depending on the service, the error may surface
575///   when the write starts, while writing, or when closing the writer.
576/// - A condition the service does not advertise through its capability fails
577///   the write with [`crate::ErrorKind::Unsupported`]; OpenDAL never
578///   silently drops a condition. A service that cannot preserve a
579///   combination of conditions also returns `Unsupported`.
580/// - A service-side conflict unrelated to these conditions surfaces as
581///   [`crate::ErrorKind::Conflict`].
582///
583/// See the [conditional operation specification][crate::docs::specs::conditional_operations]
584/// for the complete cross-operation contract.
585#[derive(Debug, Clone, Default, Eq, PartialEq)]
586pub struct WriteOptions {
587    /// Sets append mode for this operation.
588    ///
589    /// ### Capability
590    ///
591    /// Check [`crate::Capability::write_can_append`] before using this option.
592    ///
593    /// ### Behavior
594    ///
595    /// - By default, write operations overwrite existing files
596    /// - When append is set to true:
597    ///   - New data will be appended to the end of existing file
598    ///   - If file doesn't exist, it will be created
599    /// - If not supported, will return an error
600    ///
601    /// This operation allows adding data to existing files instead of overwriting them.
602    pub append: bool,
603
604    /// Sets Cache-Control header for this write operation.
605    ///
606    /// ### Capability
607    ///
608    /// Check [`crate::Capability::write_with_cache_control`] before using this feature.
609    ///
610    /// ### Behavior
611    ///
612    /// - If supported, sets Cache-Control as system metadata on the target file
613    /// - The value should follow HTTP Cache-Control header format
614    /// - If not supported, the value will be ignored
615    ///
616    /// This operation allows controlling caching behavior for the written content.
617    ///
618    /// ### Use Cases
619    ///
620    /// - Setting browser cache duration
621    /// - Configuring CDN behavior
622    /// - Optimizing content delivery
623    /// - Managing cache invalidation
624    ///
625    /// ### References
626    ///
627    /// - [MDN Cache-Control](https://developer.mozilla.org/en-US/docs/Web/HTTP/Headers/Cache-Control)
628    /// - [RFC 7234 Section 5.2](https://tools.ietf.org/html/rfc7234#section-5.2)
629    pub cache_control: Option<String>,
630    /// Sets `Content-Type` header for this write operation.
631    ///
632    /// ## Capability
633    ///
634    /// Check [`crate::Capability::write_with_content_type`] before using this feature.
635    ///
636    /// ### Behavior
637    ///
638    /// - If supported, sets Content-Type as system metadata on the target file
639    /// - The value should follow MIME type format (e.g. "text/plain", "image/jpeg")
640    /// - If not supported, the value will be ignored
641    ///
642    /// This operation allows specifying the media type of the content being written.
643    pub content_type: Option<String>,
644    /// Sets Content-Disposition header for this write request.
645    ///
646    /// ### Capability
647    ///
648    /// Check [`crate::Capability::write_with_content_disposition`] before using this feature.
649    ///
650    /// ### Behavior
651    ///
652    /// - If supported, sets Content-Disposition as system metadata on the target file
653    /// - The value should follow HTTP Content-Disposition header format
654    /// - Common values include:
655    ///   - `inline` - Content displayed within browser
656    ///   - `attachment` - Content downloaded as file
657    ///   - `attachment; filename="example.jpg"` - Downloaded with specified filename
658    /// - If not supported, the value will be ignored
659    ///
660    /// This operation allows controlling how the content should be displayed or downloaded.
661    pub content_disposition: Option<String>,
662    /// Sets Content-Encoding header for this write request.
663    ///
664    /// ### Capability
665    ///
666    /// Check [`crate::Capability::write_with_content_encoding`] before using this feature.
667    ///
668    /// ### Behavior
669    ///
670    /// - If supported, sets Content-Encoding as system metadata on the target file
671    /// - The value should follow HTTP Content-Encoding header format
672    /// - Common values include:
673    ///   - `gzip` - Content encoded using gzip compression
674    ///   - `deflate` - Content encoded using deflate compression
675    ///   - `br` - Content encoded using Brotli compression
676    ///   - `identity` - No encoding applied (default value)
677    /// - If not supported, the value will be ignored
678    ///
679    /// This operation allows specifying the encoding applied to the content being written.
680    pub content_encoding: Option<String>,
681    /// Sets user metadata for this write request.
682    ///
683    /// ### Capability
684    ///
685    /// Check [`crate::Capability::write_with_user_metadata`] before using this feature.
686    ///
687    /// ### Behavior
688    ///
689    /// - If supported, the user metadata will be attached to the object during write
690    /// - Accepts key-value pairs where both key and value are strings
691    /// - Keys are case-insensitive in most services
692    /// - Services may have limitations for user metadata, for example:
693    ///   - Key length is typically limited (e.g., 1024 bytes)
694    ///   - Value length is typically limited (e.g., 4096 bytes)
695    ///   - Total metadata size might be limited
696    ///   - Some characters might be forbidden in keys
697    /// - If not supported, the metadata will be ignored
698    ///
699    /// User metadata provides a way to attach custom metadata to objects during write operations.
700    /// This metadata can be retrieved later when reading the object.
701    pub user_metadata: Option<HashMap<String, String>>,
702
703    /// Write only when the file at the write path has this exact ETag.
704    ///
705    /// The condition succeeds when the file exists and its ETag equals this
706    /// value. A file with a different ETag, or a missing file, fails the
707    /// write with [`crate::ErrorKind::ConditionNotMatch`]. Only concrete
708    /// ETag values are portable; a wildcard such as `"*"` has no portable
709    /// meaning here.
710    ///
711    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
712    /// service does not advertise [`crate::Capability::write_with_if_match`].
713    pub if_match: Option<String>,
714    /// Write only when the file at the write path does not have this ETag.
715    ///
716    /// With a concrete ETag value, the condition succeeds when the file has
717    /// a different ETag or does not exist; the write then replaces or
718    /// creates the file. A file whose ETag equals this value fails the write
719    /// with [`crate::ErrorKind::ConditionNotMatch`]. Only concrete ETag
720    /// values are portable; a wildcard such as `"*"` has no portable meaning
721    /// here.
722    ///
723    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
724    /// service does not advertise
725    /// [`crate::Capability::write_with_if_none_match`]. Some services, such
726    /// as `s3`, support `if_not_exists` but not `if_none_match`; use
727    /// [`WriteOptions::if_not_exists`] when you only need to guard against
728    /// an existing file.
729    pub if_none_match: Option<String>,
730    /// Write only when the file at the write path has this exact version.
731    ///
732    /// The condition succeeds when the file exists and its version equals
733    /// this value. A file with a different version, or a missing file, fails
734    /// the write with [`crate::ErrorKind::ConditionNotMatch`].
735    ///
736    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
737    /// service does not advertise
738    /// [`crate::Capability::write_with_if_version_match`].
739    pub if_version_match: Option<String>,
740    /// Write only when the file at the write path does not have this
741    /// version.
742    ///
743    /// The condition succeeds when the file exists with a different version.
744    /// A file whose version equals this value fails the write with
745    /// [`crate::ErrorKind::ConditionNotMatch`]. When no file exists at the
746    /// write path, no portable behavior is defined.
747    ///
748    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
749    /// service does not advertise
750    /// [`crate::Capability::write_with_if_version_not_match`].
751    pub if_version_not_match: Option<String>,
752    /// Write only when no file exists at the write path.
753    ///
754    /// The condition succeeds when the path is empty, so the write creates
755    /// the file. An existing file fails the write with
756    /// [`crate::ErrorKind::ConditionNotMatch`].
757    ///
758    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
759    /// service does not advertise
760    /// [`crate::Capability::write_with_if_not_exists`].
761    pub if_not_exists: bool,
762    /// Write only when the file at the write path still has the identity
763    /// recorded in this metadata.
764    ///
765    /// Pass metadata previously returned by OpenDAL for the same path.
766    /// OpenDAL derives a version match when the service supports version
767    /// conditions and the metadata contains a version. Otherwise it derives
768    /// an ETag match when possible. A changed or missing file fails the write
769    /// with [`crate::ErrorKind::ConditionNotMatch`], as does combining this
770    /// option with a conflicting `if_match` or `if_version_match` value.
771    ///
772    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
773    /// service does not support the derived primitive condition, and
774    /// [`crate::ErrorKind::ConfigInvalid`] when the metadata contains neither
775    /// a version nor an ETag.
776    pub if_not_changed: Option<Metadata>,
777
778    /// Sets concurrent write operations for this writer.
779    ///
780    /// ## Behavior
781    ///
782    /// - By default, OpenDAL writes files sequentially
783    /// - When concurrent is set:
784    ///   - Multiple write operations can execute in parallel
785    ///   - Write operations return immediately without waiting if tasks space are available
786    ///   - Close operation ensures all writes complete in order
787    ///   - Memory usage increases with concurrency level
788    /// - If not supported, falls back to sequential writes
789    ///
790    /// This feature significantly improves performance when:
791    /// - Writing large files
792    /// - Network latency is high
793    /// - Storage service supports concurrent uploads like multipart uploads
794    ///
795    /// ## Performance Impact
796    ///
797    /// Setting appropriate concurrency can:
798    /// - Increase write throughput
799    /// - Reduce total write time
800    /// - Better utilize available bandwidth
801    /// - Trade memory for performance
802    pub concurrent: usize,
803    /// Sets chunk size for buffered writes.
804    ///
805    /// ### Capability
806    ///
807    /// Check [`crate::Capability::write_multi_min_size`] and [`crate::Capability::write_multi_max_size`] for size limits.
808    ///
809    /// ### Behavior
810    ///
811    /// - By default, OpenDAL sets optimal chunk size based on service capabilities
812    /// - When chunk size is set:
813    ///   - Data will be buffered until reaching chunk size
814    ///   - One API call will be made per chunk
815    ///   - Last chunk may be smaller than chunk size
816    /// - Important considerations:
817    ///   - Some services require minimum chunk sizes (e.g. S3's EntityTooSmall error)
818    ///   - Smaller chunks increase API calls and costs
819    ///   - Larger chunks increase memory usage, but improve performance and reduce costs
820    ///
821    /// ### Performance Impact
822    ///
823    /// Setting appropriate chunk size can:
824    /// - Reduce number of API calls
825    /// - Improve overall throughput
826    /// - Lower operation costs
827    /// - Better utilize network bandwidth
828    pub chunk: Option<usize>,
829}

904/// Options for copy operations.
905///
906/// Each condition checks the destination file, never the source, and every
907/// condition must hold for the copy to proceed:
908///
909/// - A false condition fails the copy with
910///   [`crate::ErrorKind::ConditionNotMatch`].
911/// - A condition the service does not advertise through its capability fails
912///   the copy with [`crate::ErrorKind::Unsupported`]; OpenDAL never silently
913///   drops a condition.
914/// - A missing source fails the copy with [`crate::ErrorKind::NotFound`];
915///   when a condition error applies at the same time, which error the copy
916///   returns is unspecified.
917/// - A service-side conflict unrelated to these conditions surfaces as
918///   [`crate::ErrorKind::Conflict`].
919///
920/// See the [conditional operation specification][crate::docs::specs::conditional_operations]
921/// for the complete cross-operation contract.
922#[derive(Debug, Clone, Default, Eq, PartialEq)]
923pub struct CopyOptions {
924    /// Copy only when no file exists at the destination path.
925    ///
926    /// The condition succeeds when the destination path is empty, so the
927    /// copy creates the destination file. An existing destination file fails
928    /// the copy with [`crate::ErrorKind::ConditionNotMatch`].
929    ///
930    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
931    /// service does not advertise
932    /// [`crate::Capability::copy_with_if_not_exists`].
933    pub if_not_exists: bool,
934
935    /// Copy only when the destination file has this exact ETag.
936    ///
937    /// The condition succeeds when the destination file exists and its ETag
938    /// equals this value. A destination file with a different ETag, or a
939    /// missing destination file, fails the copy with
940    /// [`crate::ErrorKind::ConditionNotMatch`]. Only concrete ETag values
941    /// are portable; a wildcard such as `"*"` has no portable meaning here.
942    ///
943    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
944    /// service does not advertise [`crate::Capability::copy_with_if_match`].
945    pub if_match: Option<String>,
946    /// Copy only when the destination file does not have this ETag.
947    ///
948    /// With a concrete ETag value, the condition succeeds when the
949    /// destination file has a different ETag or does not exist; the copy
950    /// then replaces or creates the destination file. A destination file
951    /// whose ETag equals this value fails the copy with
952    /// [`crate::ErrorKind::ConditionNotMatch`]. Only concrete ETag values
953    /// are portable; a wildcard such as `"*"` has no portable meaning here.
954    ///
955    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
956    /// service does not advertise
957    /// [`crate::Capability::copy_with_if_none_match`].
958    pub if_none_match: Option<String>,
959    /// Copy only when the destination file has this exact version.
960    ///
961    /// The condition succeeds when the destination file exists and its
962    /// version equals this value. A destination file with a different
963    /// version, or a missing destination file, fails the copy with
964    /// [`crate::ErrorKind::ConditionNotMatch`].
965    ///
966    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
967    /// service does not advertise
968    /// [`crate::Capability::copy_with_if_version_match`].
969    pub if_version_match: Option<String>,
970    /// Copy only when the destination file does not have this version.
971    ///
972    /// The condition succeeds when the destination file exists with a
973    /// different version. A destination file whose version equals this value
974    /// fails the copy with [`crate::ErrorKind::ConditionNotMatch`]. When no
975    /// destination file exists, no portable behavior is defined.
976    ///
977    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
978    /// service does not advertise
979    /// [`crate::Capability::copy_with_if_version_not_match`].
980    pub if_version_not_match: Option<String>,
981    /// Copy only when the destination file still has the identity recorded
982    /// in this metadata.
983    ///
984    /// Pass metadata previously returned by OpenDAL for the destination
985    /// path. OpenDAL derives a version match when the service supports version
986    /// conditions and the metadata contains a version. Otherwise it derives
987    /// an ETag match when possible. A changed or missing destination file
988    /// fails the copy with [`crate::ErrorKind::ConditionNotMatch`], as does
989    /// combining this option with a conflicting `if_match` or
990    /// `if_version_match` value.
991    ///
992    /// The operation returns [`crate::ErrorKind::Unsupported`] when the
993    /// service does not support the derived primitive condition, and
994    /// [`crate::ErrorKind::ConfigInvalid`] when the metadata contains neither
995    /// a version nor an ETag.
996    pub if_not_changed: Option<Metadata>,
997
998    /// Copy from a specific source file version.
999    ///
1000    /// This selects which stored version of the source the copy reads; it is
1001    /// not a condition on the destination file. Destination behavior follows
1002    /// normal copy semantics.
1003    ///
1004    /// Check [`crate::Capability::copy_with_source_version`] before using
1005    /// this option.
1006    pub source_version: Option<String>,
1007
1008    /// Asserted complete content length of the source object.
1009    ///
1010    /// OpenDAL may trust this value without reading source metadata. It can use
1011    /// the value to plan copied ranges and to construct the returned metadata.
1012    /// An incorrect value can therefore cause incomplete copy planning, failed
1013    /// requests, or incorrect result metadata.
1014    ///
1015    /// A service can ignore this value when its copy operation already reports
1016    /// an authoritative copied size.
1017    ///
1018    /// This option does not pin the source object or protect the copy from a
1019    /// concurrent source change. Set it only when the caller can guarantee that
1020    /// the value describes the object that the service will copy, for example by
1021    /// selecting an immutable [`Self::source_version`]. Otherwise omit it if the
1022    /// source can change.
1023    pub source_content_length_hint: Option<u64>,
1024
1025    /// Sets concurrent copy operations for this copier.
1026    ///
1027    /// This is a best-effort execution option. Services that cannot split copy
1028    /// into concurrent server-side tasks can ignore it.
1029    pub concurrent: usize,
1030
1031    /// Sets chunk size for segmented copy operations.
1032    ///
1033    /// ### Capability
1034    ///
1035    /// Check [`crate::Capability::copy_can_multi`],
1036    /// [`crate::Capability::copy_multi_min_size`] and
1037    /// [`crate::Capability::copy_multi_max_size`] before using this feature.
1038    ///
1039    /// This is a best-effort execution option. Services that support
1040    /// server-side segmented copy can use it as the target size for each copy
1041    /// step. Services that cannot split copy operations can ignore it.
1042    pub chunk: Option<usize>,
1043}
