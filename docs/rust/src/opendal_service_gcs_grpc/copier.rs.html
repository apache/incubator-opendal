18use std::sync::Arc;
19
20use opendal_core::raw::*;
21use opendal_core::*;
22
23use crate::core::{ErrorContext, GcsGrpcCore, parse_generation, parse_object, parse_status};
24use crate::generated::google::storage::v2::RewriteObjectRequest;
25
26pub(super) fn new_gcs_grpc_copier(
27    core: Arc<GcsGrpcCore>,
28    ctx: OperationContext,
29    from: &str,
30    to: &str,
31    args: OpCopy,
32) -> oio::OneShotCopier {
33    let from = from.to_string();
34    let to = to.to_string();
35    oio::OneShotCopier::new_with(move || {
36        let core = core.clone();
37        let ctx = ctx.clone();
38        let from = from.clone();
39        let to = to.clone();
40        let args = args.clone();
41        async move { copy_object(core, ctx, from, to, args).await }
42    })
43}
44
45async fn copy_object(
46    core: Arc<GcsGrpcCore>,
47    ctx: OperationContext,
48    from: String,
49    to: String,
50    args: OpCopy,
51) -> Result<Metadata> {
52    let mut request = RewriteObjectRequest {
53        destination_name: core.object_name(&to),
54        destination_bucket: core.bucket_resource(),
55        source_bucket: core.bucket_resource(),
56        source_object: core.object_name(&from),
57        source_generation: parse_generation(args.source_version())?,
58        if_generation_match: args.if_not_exists().then_some(0),
59        ..Default::default()
60    };
61    let bucket = core.bucket_resource();
62    loop {
63        let tonic_request = core
64            .request(
65                &ctx,
66                request.clone(),
67                &[("source_bucket", &bucket), ("bucket", &bucket)],
68            )
69            .await?;
70        let response = core
71            .client()
72            .rewrite_object(tonic_request)
73            .await
74            .map_err(|status| {
75                parse_status(
76                    ErrorContext::new(ServiceOperation("RewriteObject"))
77                        .with_if_not_exists(args.if_not_exists()),
78                    status,
79                )
80            })?
81            .into_inner();
82        if response.done {
83            return response.resource.as_ref().map(parse_object).ok_or_else(|| {
84                Error::new(
85                    ErrorKind::Unexpected,
86                    "GCS rewrite completed without object metadata",
87                )
88            });
89        }
90        if response.rewrite_token.is_empty() {
91            return Err(Error::new(
92                ErrorKind::Unexpected,
93                "GCS rewrite requires continuation but returned no token",
94            ));
95        }
96        request.rewrite_token = response.rewrite_token;
97    }
98}
