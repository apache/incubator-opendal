36pub type CosWriters = TwoWays<oio::MultipartWriter<CosWriter>, oio::AppendWriter<CosWriter>>;
37
38pub struct CosWriter {
39    core: Arc<CosCore>,
40    ctx: OperationContext,
41
42    op: OpWrite,
43    path: String,
44}
45
46impl CosWriter {
47    pub fn new(core: Arc<CosCore>, ctx: OperationContext, path: &str, op: OpWrite) -> Self {
48        CosWriter {
49            core,
50            ctx,
51            path: path.to_string(),
52            op,
53        }
54    }
55
56    fn parse_metadata(headers: &HeaderMap<HeaderValue>) -> Result<Metadata> {
57        let mut meta = MetadataBuilder::unknown();
58        if let Some(etag) = parse_etag(headers)? {
59            meta.etag(etag);
60        }
61        if let Some(md5) = parse_content_md5(headers)? {
62            meta.content_md5(md5);
63        }
64        if let Some(version) = parse_header_to_str(headers, constants::X_COS_VERSION_ID)?
65            && version != "null"
66        {
67            meta.version(version);
68        }
69
70        Ok(meta.build())
71    }
72}
73
74impl oio::MultipartWrite for CosWriter {
75    async fn write_once(&self, size: u64, body: Buffer) -> Result<Metadata> {
76        let req = self
77            .core
78            .cos_put_object_request(&self.path, Some(size), &self.op, body)?;
79
80        let req = self.core.sign(&self.ctx, req).await?;
81
82        let resp = self.core.send(&self.ctx, req).await?;
83
84        let meta = Self::parse_metadata(resp.headers())?;
85
86        let status = resp.status();
87
88        match status {
89            StatusCode::CREATED | StatusCode::OK => Ok(meta),
90            _ => Err(parse_error(
91                ErrorContext::new(ServiceOperation("PutObject"))
92                    .with_if_not_exists(self.op.if_not_exists()),
93                resp,
94            )),
95        }
96    }
97
98    async fn initiate_part(&self) -> Result<String> {
99        let resp = self
100            .core
101            .cos_initiate_multipart_upload(&self.ctx, &self.path, &self.op)
102            .await?;
103
104        let status = resp.status();
105
106        match status {
107            StatusCode::OK => {
108                let bs = resp.into_body();
109
110                let result: InitiateMultipartUploadResult =
111                    quick_xml::de::from_reader(bytes::Buf::reader(bs))
112                        .map_err(new_xml_deserialize_error)?;
113
114                Ok(result.upload_id)
115            }
116            _ => Err(parse_error(
117                ErrorContext::new(ServiceOperation("InitiateMultipartUpload"))
118                    .with_if_not_exists(self.op.if_not_exists()),
119                resp,
120            )),
121        }
122    }
123
124    async fn write_part(
125        &self,
126        upload_id: &str,
127        part_number: usize,
128        size: u64,
129        body: Buffer,
130    ) -> Result<oio::MultipartPart> {
131        // COS requires part number must between [1..=10000]
132        let part_number = part_number + 1;
133
134        let resp = self
135            .core
136            .cos_upload_part_request(&self.ctx, &self.path, upload_id, part_number, size, body)
137            .await?;
138
139        let status = resp.status();
140
141        match status {
142            StatusCode::OK => {
143                let etag = parse_etag(resp.headers())?
144                    .ok_or_else(|| {
145                        Error::new(
146                            ErrorKind::Unexpected,
147                            "ETag not present in returning response",
148                        )
149                    })?
150                    .to_string();
151
152                Ok(oio::MultipartPart {
153                    part_number,
154                    etag,
155                    checksum: None,
156                    size: None,
157                })
158            }
159            _ => Err(parse_error(
160                ErrorContext::new(ServiceOperation("UploadPart")),
161                resp,
162            )),
163        }
164    }
165
166    async fn complete_part(
167        &self,
168        upload_id: &str,
169        parts: &[oio::MultipartPart],
170    ) -> Result<Metadata> {
171        let parts = parts
172            .iter()
173            .map(|p| CompleteMultipartUploadRequestPart {
174                part_number: p.part_number,
175                etag: p.etag.clone(),
176            })
177            .collect();
178
179        let mut resp = self
180            .core
181            .cos_complete_multipart_upload(&self.ctx, &self.path, upload_id, parts, &self.op)
182            .await?;
183
184        let mut meta = Self::parse_metadata(resp.headers())?.into_builder();
185
186        let status = resp.status();
187        if status != StatusCode::OK {
188            return Err(parse_error(
189                ErrorContext::new(ServiceOperation("CompleteMultipartUpload"))
190                    .with_if_not_exists(self.op.if_not_exists()),
191                resp,
192            ));
193        }
194
195        let result: CompleteMultipartUploadResult =
196            quick_xml::de::from_reader(resp.body_mut().reader())
197                .map_err(new_xml_deserialize_error)?;
198        meta.etag(&result.etag);
199        Ok(meta.build())
200    }
201
202    async fn abort_part(&self, upload_id: &str) -> Result<()> {
203        let resp = self
204            .core
205            .cos_abort_multipart_upload(&self.ctx, &self.path, upload_id)
206            .await?;
207        match resp.status() {
208            // cos returns code 204 if abort succeeds.
209            // Reference: https://www.tencentcloud.com/document/product/436/7740
210            StatusCode::NO_CONTENT => Ok(()),
211            _ => Err(parse_error(
212                ErrorContext::new(ServiceOperation("AbortMultipartUpload")),
213                resp,
214            )),
215        }
216    }
217}

219impl oio::AppendWrite for CosWriter {
220    async fn offset(&self) -> Result<u64> {
221        let resp = self
222            .core
223            .cos_head_object(&self.ctx, &self.path, &OpStat::default())
224            .await?;
225
226        let status = resp.status();
227        match status {
228            StatusCode::OK => {
229                let content_length = parse_content_length(resp.headers())?.ok_or_else(|| {
230                    Error::new(
231                        ErrorKind::Unexpected,
232                        "Content-Length not present in returning response",
233                    )
234                })?;
235                Ok(content_length)
236            }
237            StatusCode::NOT_FOUND => Ok(0),
238            _ => Err(parse_error(
239                ErrorContext::new(ServiceOperation("HeadObject")),
240                resp,
241            )),
242        }
243    }
244
245    async fn append(&self, offset: u64, size: u64, body: Buffer) -> Result<Metadata> {
246        let req = self
247            .core
248            .cos_append_object_request(&self.path, offset, size, &self.op, body)?;
249
250        let req = self.core.sign(&self.ctx, req).await?;
251
252        let resp = self.core.send(&self.ctx, req).await?;
253
254        let meta = Self::parse_metadata(resp.headers())?;
255
256        let status = resp.status();
257
258        match status {
259            StatusCode::OK => Ok(meta),
260            _ => Err(parse_error(
261                ErrorContext::new(ServiceOperation("AppendObject")),
262                resp,
263            )),
264        }
265    }
