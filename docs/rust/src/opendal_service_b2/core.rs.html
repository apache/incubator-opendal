79    /// [b2_authorize_account](https://www.backblaze.com/apidocs/b2-authorize-account)
80    pub async fn get_auth_info(&self, ctx: &OperationContext) -> Result<AuthInfo> {
81        {
82            let signer = self.signer.read().await;
83
84            if !signer.auth_info.authorization_token.is_empty()
85                && signer.auth_info.expires_in > Timestamp::now()
86            {
87                let auth_info = signer.auth_info.clone();
88                return Ok(auth_info);
89            }
90        }
91
92        {
93            let mut signer = self.signer.write().await;
94            let req = Request::get("https://api.backblazeb2.com/b2api/v2/b2_authorize_account")
95                .header(
96                    header::AUTHORIZATION,
97                    format_authorization_by_basic(
98                        &signer.application_key_id,
99                        &signer.application_key,
100                    )?,
101                )
102                .body(Buffer::new())
103                .map_err(new_request_build_error)?;
104
105            let resp = ctx.http_transport().send(req).await?;
106            let status = resp.status();
107
108            match status {
109                StatusCode::OK => {
110                    let resp_body = resp.into_body();
111                    let token: AuthorizeAccountResponse =
112                        serde_json::from_reader(resp_body.reader())
113                            .map_err(new_json_deserialize_error)?;
114                    signer.auth_info = AuthInfo {
115                        authorization_token: token.authorization_token.clone(),
116                        api_url: token.api_url.clone(),
117                        download_url: token.download_url.clone(),
118                        // This authorization token is valid for at most 24 hours.
119                        expires_in: Timestamp::now() + Duration::from_secs(20 * 60 * 60),
120                    };
121                }
122                _ => {
123                    return Err(parse_error(
124                        ErrorContext::new(ServiceOperation("AuthorizeAccount")),
125                        resp,
126                    ));
127                }
128            }
129            Ok(signer.auth_info.clone())
130        }
131    }

137        ctx: &OperationContext,
138        path: &str,
139        range: BytesRange,
140        _args: &OpRead,

145
146        // Construct headers to add to the request

156        req = req.header(header::AUTHORIZATION, auth_info.authorization_token);
157
158        if !range.is_full() {

160        }
161
162        let req = req
163            .extension(Operation::Read)
164            .extension(ServiceOperation("DownloadFileByName"));
165
166        let req = req.body(Buffer::new()).map_err(new_request_build_error)?;
167
168        ctx.http_transport().fetch(req).await
169    }

257    pub async fn upload_file(
258        &self,
259        ctx: &OperationContext,
260        path: &str,
261        size: Option<u64>,
262        args: &OpWrite,
263        body: Buffer,
264    ) -> Result<Response<Buffer>> {
265        let resp = self.get_upload_url(ctx).await?;
266
267        let p = build_abs_path(&self.root, path);
268
269        let mut req = Request::post(resp.upload_url);
270
271        req = req.header(X_BZ_FILE_NAME, percent_encode_path(&p));
272
273        req = req.header(header::AUTHORIZATION, resp.authorization_token);
274
275        req = req.header(X_BZ_CONTENT_SHA1, "do_not_verify");
276
277        if let Some(size) = size {
278            req = req.header(header::CONTENT_LENGTH, size.to_string())
279        }
280
281        if let Some(mime) = args.content_type() {
282            req = req.header(header::CONTENT_TYPE, mime)
283        } else {
284            req = req.header(header::CONTENT_TYPE, "b2/x-auto")
285        }
286
287        if let Some(pos) = args.content_disposition() {
288            req = req.header(header::CONTENT_DISPOSITION, pos)
289        }
290
291        // Set user metadata headers.
292        // B2 uses X-Bz-Info-* prefix for custom file info.
293        if let Some(user_metadata) = args.user_metadata() {
294            for (key, value) in user_metadata {
295                req = req.header(
296                    format!("{}{}", constants::X_BZ_INFO_PREFIX, key),
297                    percent_encode_path(value),
298                );
299            }
300        }
301
302        let req = req
303            .extension(Operation::Write)
304            .extension(ServiceOperation("UploadFile"));
305
306        // Set body
307        let req = req.body(body).map_err(new_request_build_error)?;
308
309        self.send(ctx, req).await
310    }

312    pub async fn start_large_file(
313        &self,
314        ctx: &OperationContext,
315        path: &str,
316        args: &OpWrite,
317    ) -> Result<Response<Buffer>> {
318        let p = build_abs_path(&self.root, path);
319
320        let auth_info = self.get_auth_info(ctx).await?;
321
322        let url = format!("{}/b2api/v2/b2_start_large_file", auth_info.api_url);
323
324        let mut req = Request::post(&url);
325
326        req = req.header(header::AUTHORIZATION, auth_info.authorization_token);
327
328        let mut start_large_file_request = StartLargeFileRequest {
329            bucket_id: self.bucket_id.clone(),
330            file_name: percent_encode_path(&p),
331            content_type: "b2/x-auto".to_owned(),
332            file_info: None,
333        };
334
335        if let Some(mime) = args.content_type() {
336            mime.clone_into(&mut start_large_file_request.content_type)
337        }
338
339        // Set user metadata in file_info.
340        // B2 uses fileInfo field for custom file info in start_large_file API.
341        if let Some(user_metadata) = args.user_metadata() {
342            let file_info: HashMap<String, String> = user_metadata
343                .into_iter()
344                .map(|(k, v)| (k.to_string(), percent_encode_path(v)))
345                .collect();
346            start_large_file_request.file_info = Some(file_info);
347        }
348
349        let req = req
350            .extension(Operation::Write)
351            .extension(ServiceOperation("StartLargeFile"));
352
353        let body =
354            serde_json::to_vec(&start_large_file_request).map_err(new_json_serialize_error)?;
355        let body = bytes::Bytes::from(body);
356
357        let req = req
358            .body(Buffer::from(body))
359            .map_err(new_request_build_error)?;
360
361        self.send(ctx, req).await
362    }

498    pub async fn get_file_info(
499        &self,
500        ctx: &OperationContext,
501        path: &str,
502        delimiter: Option<&str>,
503    ) -> Result<File> {
504        let resp = self
505            .list_file_names_raw(ctx, Some(path), delimiter, None, None, Operation::Stat)
506            .await?;
507
508        let status = resp.status();
509        match status {
510            StatusCode::OK => {
511                let bs = resp.into_body();
512                let mut resp: ListFileNamesResponse =
513                    serde_json::from_reader(bs.reader()).map_err(new_json_deserialize_error)?;
514
515                if resp.files.is_empty() {
516                    return Err(Error::new(ErrorKind::NotFound, "no such file or directory"));
517                }
518                Ok(resp.files.swap_remove(0))
519            }
520            _ => Err(parse_error(
521                ErrorContext::new(ServiceOperation("ListFileNames")),
522                resp,
523            )),
524        }
525    }

539    async fn list_file_names_raw(
540        &self,
541        ctx: &OperationContext,
542        prefix: Option<&str>,
543        delimiter: Option<&str>,
544        limit: Option<usize>,
545        start_after: Option<String>,
546        operation: Operation,
547    ) -> Result<Response<Buffer>> {
548        let auth_info = self.get_auth_info(ctx).await?;
549
550        let url = format!("{}/b2api/v2/b2_list_file_names", auth_info.api_url);
551
552        let mut url = QueryPairsWriter::new(&url);
553        url = url.push("bucketId", &self.bucket_id);
554
555        if let Some(prefix) = prefix {
556            let prefix = build_abs_path(&self.root, prefix);
557            if !prefix.is_empty() {
558                url = url.push("prefix", &percent_encode_path(&prefix));
559            }
560        }
561
562        if let Some(limit) = limit {
563            url = url.push("maxFileCount", &limit.to_string());
564        }
565
566        if let Some(start_after) = start_after {
567            url = url.push("startFileName", &percent_encode_path(&start_after));
568        }
569
570        if let Some(delimiter) = delimiter {
571            url = url.push("delimiter", delimiter);
572        }
573
574        let mut req = Request::get(url.finish());
575
576        req = req.header(header::AUTHORIZATION, auth_info.authorization_token);
577
578        req = req
579            .extension(operation)
580            .extension(ServiceOperation("ListFileNames"));
581
582        // Set body
583        let req = req.body(Buffer::new()).map_err(new_request_build_error)?;
584
585        self.send(ctx, req).await
586    }

666#[derive(Clone)]
667pub struct AuthInfo {
668    pub authorization_token: String,
669    /// The base URL to use for all API calls except for uploading and downloading files.
670    pub api_url: String,
671    /// The base URL to use for downloading files.
672    pub download_url: String,
673    /// The time when the authorization token expires.
674    pub expires_in: Timestamp,
675}

784#[derive(Debug, Clone, Deserialize)]
785#[serde(rename_all = "camelCase")]
786pub struct File {
787    pub file_id: Option<String>,
788    pub content_length: u64,
789    pub content_md5: Option<String>,
790    pub content_type: Option<String>,
791    pub file_name: String,
792    /// Custom file info (user metadata) stored with the file.
793    /// Keys are the original names without the `X-Bz-Info-` prefix.
794    /// Values are URL-encoded when stored, decoded when read.
795    #[serde(default)]
796    pub file_info: HashMap<String, String>,
797}
798
799pub(super) fn parse_file_info(file: &File) -> Metadata {
800    if file.file_name.ends_with('/') {
801        return MetadataBuilder::dir().build();
802    }
803
804    let mut metadata = MetadataBuilder::file(file.content_length);
805
806    if let Some(content_md5) = &file.content_md5 {
807        metadata.content_md5(content_md5);
808    }
809
810    if let Some(content_type) = &file.content_type {
811        metadata.content_type(content_type);
812    }
813
814    // Parse user metadata from file_info
815    // B2 stores user metadata with keys stripped of the "X-Bz-Info-" prefix
816    // and values are URL-encoded
817    if !file.file_info.is_empty() {
818        let user_metadata: HashMap<String, String> = file
819            .file_info
820            .iter()
821            .map(|(k, v)| (k.to_lowercase(), percent_decode_path(v)))
822            .collect();
823        if !user_metadata.is_empty() {
824            metadata.user_metadata(user_metadata);
825        }
826    }
827
828    metadata.build()
829}

859/// the error response of b2
860#[derive(Default, Debug, Deserialize)]
861#[allow(dead_code)]
862struct B2Error {
863    status: u32,
864    code: String,
865    message: String,
866}
867
868/// Context needed to classify an error from this service.
869#[derive(Clone, Copy, Debug)]
870pub(crate) struct ErrorContext {
871    service_operation: ServiceOperation,
872}
873
874impl ErrorContext {
875    pub(crate) const fn new(service_operation: ServiceOperation) -> Self {
876        Self { service_operation }
877    }
878}
879
880/// Parse an error response using its service request context.
881pub(crate) fn parse_error(ctx: ErrorContext, resp: Response<Buffer>) -> Error {
882    let (parts, body) = resp.into_parts();
883    let bs = body.to_bytes();
884
885    let (mut kind, mut retryable) = match parts.status.as_u16() {
886        403 => (ErrorKind::PermissionDenied, false),
887        404 => (ErrorKind::NotFound, false),
888        304 | 412 => (ErrorKind::ConditionNotMatch, false),
889        // Service b2 could return 403, show the authorization error
890        401 => (ErrorKind::PermissionDenied, true),
891        429 => (ErrorKind::RateLimited, true),
892        500 | 502 | 503 | 504 => (ErrorKind::Unexpected, true),
893        _ => (ErrorKind::Unexpected, false),
894    };
895
896    let (message, b2_err) = serde_json::from_reader::<_, B2Error>(bs.clone().reader())
897        .map(|b2_err| (format!("{b2_err:?}"), Some(b2_err)))
898        .unwrap_or_else(|_| (String::from_utf8_lossy(&bs).into_owned(), None));
899
900    if let Some(b2_err) = b2_err {
901        (kind, retryable) = parse_b2_error_code(b2_err.code.as_str()).unwrap_or((kind, retryable));
902    };
903
904    let mut err = Error::new(kind, message);
905
906    err = err.with_context("service_operation", ctx.service_operation.0);
907    err = with_error_response_context(err, parts);
908
909    if retryable {
910        err = err.set_temporary();
911    }
912
913    err
914}
915
916/// Returns the `Error kind` of this code and whether the error is retryable.
917pub(crate) fn parse_b2_error_code(code: &str) -> Option<(ErrorKind, bool)> {
918    match code {
919        "already_hidden" => Some((ErrorKind::AlreadyExists, false)),
920        "no_such_file" => Some((ErrorKind::NotFound, false)),
921        _ => None,
922    }
923}
