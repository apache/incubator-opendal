43/// Builder for Volcengine TOS service.
44#[doc = include_str!("docs.md")]
45#[derive(Default)]
46pub struct TosBuilder {
47    pub(super) config: TosConfig,
48}
49
50impl Debug for TosBuilder {
51    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
52        f.debug_struct("TosBuilder")
53            .field("config", &self.config)
54            .finish_non_exhaustive()
55    }
56}

58impl TosBuilder {
59    /// Set root of this backend.
60    ///
61    /// All operations will happen under this root.
62    pub fn root(mut self, root: &str) -> Self {
63        self.config.root = if root.is_empty() {
64            None
65        } else {
66            Some(root.to_string())
67        };
68        self
69    }
70
71    /// Set bucket name of this backend.
72    pub fn bucket(mut self, bucket: &str) -> Self {
73        self.config.bucket = bucket.to_string();
74        self
75    }
76
77    /// Set endpoint of this backend.
78    ///
79    /// Endpoint must be full uri, e.g.
80    /// - TOS: `https://tos-cn-beijing.volces.com`
81    /// - TOS with region: `https://tos-{region}.volces.com`
82    ///
83    /// If user inputs endpoint without scheme like "tos-cn-beijing.volces.com", we
84    /// will prepend "https://" before it.
85    pub fn endpoint(mut self, endpoint: &str) -> Self {
86        if !endpoint.is_empty() {
87            self.config.endpoint = Some(endpoint.trim_end_matches('/').to_string());
88        }
89        self
90    }
91
92    /// Set region of this backend.
93    ///
94    /// Region represent the signing region of this endpoint.
95    ///
96    /// If region is not set, we will try to load it from environment.
97    /// If still not set, default to `cn-beijing`.
98    pub fn region(mut self, region: &str) -> Self {
99        if !region.is_empty() {
100            self.config.region = Some(region.to_string());
101        }
102        self
103    }
104
105    /// Set access_key_id of this backend.
106    ///
107    /// - If access_key_id is set, we will take user's input first.
108    /// - If not, we will try to load it from environment.
109    pub fn access_key_id(mut self, v: &str) -> Self {
110        self.config.access_key_id = Some(v.to_string());
111        self
112    }
113
114    /// Set secret_access_key of this backend.
115    ///
116    /// - If secret_access_key is set, we will take user's input first.
117    /// - If not, we will try to load it from environment.
118    pub fn secret_access_key(mut self, v: &str) -> Self {
119        self.config.secret_access_key = Some(v.to_string());
120        self
121    }
122
123    /// Set security_token of this backend.
124    pub fn security_token(mut self, v: &str) -> Self {
125        self.config.security_token = Some(v.to_string());
126        self
127    }
128
129    /// Skip signature will skip loading credentials and signing requests.
130    pub fn skip_signature(mut self) -> Self {
131        self.config.skip_signature = true;
132        self
133    }
134}

136impl Builder for TosBuilder {
137    type Config = TosConfig;
138
139    fn build(self) -> Result<impl Service> {
140        let mut config = self.config;
141        let region = config
142            .region
143            .clone()
144            .unwrap_or_else(|| "cn-beijing".to_string());
145
146        if config.endpoint.is_none() {
147            config.endpoint = Some(format!("https://tos-{}.volces.com", region));
148        }
149
150        let endpoint = config.endpoint.clone().unwrap();
151        let bucket = config.bucket.clone();
152        let root = normalize_root(&config.root.clone().unwrap_or_default());
153
154        let ctx = Context::new().with_file_read(TokioFileRead).with_env(OsEnv);
155
156        let mut provider = ProvideCredentialChain::new().push(EnvCredentialProvider::new());
157
158        if let (Some(ak), Some(sk)) = (&config.access_key_id, &config.secret_access_key) {
159            let static_provider = if let Some(token) = config.security_token.as_deref() {
160                StaticCredentialProvider::new(ak, sk).with_security_token(token)
161            } else {
162                StaticCredentialProvider::new(ak, sk)
163            };
164            provider = provider.push_front(static_provider);
165        }
166
167        let request_signer = RequestSigner::new(&region);
168        let signer = Signer::new(ctx, provider, request_signer);
169
170        let info = ServiceInfo::new(TOS_SCHEME, &root, &bucket);
171        let capability = Capability {
172            stat_with_version: true,
173
174            read: true,
175            read_with_suffix: true,
176            read_with_if_match: true,
177            read_with_if_none_match: true,
178            read_with_if_modified_since: true,
179            read_with_if_unmodified_since: true,
180            read_with_version: true,
181
182            write: true,
183            write_can_empty: true,
184            write_can_multi: true,
185            write_with_cache_control: true,
186            write_with_content_type: true,
187            write_with_content_encoding: true,
188            write_with_if_match: true,
189            write_with_if_not_exists: true,
190            write_with_user_metadata: true,
191            write_multi_min_size: Some(5 * 1024 * 1024),
192            write_multi_max_size: if cfg!(target_pointer_width = "64") {
193                Some(5 * 1024 * 1024 * 1024)
194            } else {
195                Some(usize::MAX)
196            },
197
198            delete: true,
199            delete_max_size: Some(1000),
200            delete_with_version: true,
201
202            copy: true,
203            copy_can_multi: true,
204            copy_multi_min_size: Some(5 * 1024 * 1024),
205            copy_multi_max_size: if cfg!(target_pointer_width = "64") {
206                Some(5 * 1024 * 1024 * 1024)
207            } else {
208                Some(usize::MAX)
209            },
210
211            list: true,
212            list_with_limit: true,
213            list_with_start_after: true,
214            list_with_recursive: true,
215            list_with_versions: true,
216            list_with_deleted: true,
217
218            presign: true,
219            presign_stat: true,
220            presign_read: true,
221            presign_write: true,
222            presign_delete: true,
223
224            stat: true,
225            stat_with_if_match: true,
226            stat_with_if_none_match: true,
227            stat_with_if_modified_since: true,
228            stat_with_if_unmodified_since: true,
229
230            shared: true,
231
232            ..Default::default()
233        };
234
235        // Extract domain from endpoint, removing http:// or https:// prefix
236        let endpoint_domain = if let Some(stripped) = endpoint.strip_prefix("http://") {
237            stripped
238        } else if let Some(stripped) = endpoint.strip_prefix("https://") {
239            stripped
240        } else {
241            &endpoint
242        };
243
244        let core = TosCore {
245            info,
246            capability,
247            bucket,
248            endpoint: endpoint.clone(),
249            endpoint_domain: endpoint_domain.to_string(),
250            root,
251            default_storage_class: None,
252            skip_signature: config.skip_signature,
253            signer,
254        };
255
256        Ok(TosBackend {
257            core: Arc::new(core),
258        })
259    }
260}

262#[derive(Debug, Clone)]
263pub struct TosBackend {
264    pub(crate) core: Arc<TosCore>,
265}
266
267impl Service for TosBackend {
268    type Reader = oio::StreamReader<TosReader>;
269    type Writer = oio::MultipartWriter<TosWriter>;
270    type Lister = TosListers;
271    type Deleter = oio::BatchDeleter<TosDeleter>;
272    type Copier = TosCopiers;
273    type Composer = ();
274
275    fn info(&self) -> ServiceInfo {
276        self.core.info.clone()
277    }
278
279    fn capability(&self) -> Capability {
280        self.core.capability
281    }
282
283    async fn create_dir(
284        &self,
285        _ctx: &OperationContext,
286        _path: &str,
287        _args: OpCreateDir,
288    ) -> Result<RpCreateDir> {
289        Err(Error::new(
290            ErrorKind::Unsupported,
291            "operation is not supported",
292        ))
293    }
294
295    async fn stat(&self, ctx: &OperationContext, path: &str, args: OpStat) -> Result<RpStat> {
296        let resp = self.core.tos_head_object(ctx, path, args).await?;
297
298        let status = resp.status();
299
300        match status {
301            StatusCode::OK => {
302                let headers = resp.headers();
303                let mut meta = tos_parse_into_metadata(path, headers)?.into_builder();
304
305                let user_meta = parse_prefixed_headers(headers, X_TOS_META_PREFIX);
306                if !user_meta.is_empty() {
307                    meta.user_metadata(user_meta);
308                }
309
310                if let Some(v) = parse_header_to_str(headers, X_TOS_VERSION_ID)? {
311                    meta.version(v);
312                }
313
314                if parse_header_to_str(headers, X_TOS_DIRECTORY)?
315                    .map(|v| {
316                        v.parse::<bool>().map_err(|e| {
317                            Error::new(ErrorKind::Unexpected, "header value is not valid integer")
318                                .set_source(e)
319                        })
320                    })
321                    .transpose()?
322                    == Some(true)
323                {
324                    meta.set_dir();
325                }
326
327                Ok(RpStat::new(meta.build()))
328            }
329            _ => Err(parse_error(
330                ErrorContext::new(ServiceOperation("HeadObject")),
331                resp,
332            )),
333        }
334    }
335    fn read(&self, ctx: &OperationContext, path: &str, args: OpRead) -> Result<Self::Reader> {
336        let output: oio::StreamReader<TosReader> = {
337            Ok(oio::StreamReader::new(TosReader::new(
338                self.clone(),
339                ctx.clone(),
340                path,
341                args,
342            )))
343        }?;
344
345        Ok(output)
346    }
347
348    fn write(&self, ctx: &OperationContext, path: &str, args: OpWrite) -> Result<Self::Writer> {
349        let output: oio::MultipartWriter<TosWriter> = {
350            let writer = TosWriter::new(self.core.clone(), ctx.clone(), path, args.clone());
351
352            let w = oio::MultipartWriter::new(ctx.executor().clone(), writer, args.concurrent());
353
354            Ok(w)
355        }?;
356
357        Ok(output)
358    }
359
360    fn delete(&self, ctx: &OperationContext) -> Result<Self::Deleter> {
361        let output: oio::BatchDeleter<TosDeleter> = {
362            let deleter = TosDeleter::new(self.core.clone(), ctx.clone());
363            Ok(oio::BatchDeleter::new(
364                deleter,
365                self.core.capability.delete_max_size,
366            ))
367        }?;
368
369        Ok(output)
370    }
371
372    fn list(&self, ctx: &OperationContext, path: &str, args: OpList) -> Result<Self::Lister> {
373        let output: TosListers = {
374            let lister = if args.versions() || args.deleted() {
375                TwoWays::Two(oio::PageLister::new(TosObjectVersionsLister::new(
376                    self.core.clone(),
377                    ctx.clone(),
378                    path,
379                    args,
380                )))
381            } else {
382                TwoWays::One(oio::PageLister::new(TosLister::new(
383                    self.core.clone(),
384                    ctx.clone(),
385                    path,
386                    args,
387                )))
388            };
389            Ok(lister)
390        }?;
391
392        Ok(output)
393    }
394
395    fn copy(
396        &self,
397        ctx: &OperationContext,
398        from: &str,
399        to: &str,
400        args: OpCopy,
401    ) -> Result<Self::Copier> {
402        let output: TosCopiers = {
403            let copier = new_tos_copier(self.core.clone(), ctx, from, to, args)?;
404            Ok(copier)
405        }?;
406
407        Ok(output)
408    }
409
410    async fn rename(
411        &self,
412        _ctx: &OperationContext,
413        _from: &str,
414        _to: &str,
415        _args: OpRename,
416    ) -> Result<RpRename> {
417        Err(Error::new(
418            ErrorKind::Unsupported,
419            "operation is not supported",
420        ))
421    }
422
423    async fn presign(
424        &self,
425        ctx: &OperationContext,
426        path: &str,
427        args: OpPresign,
428    ) -> Result<RpPresign> {
429        let (expire, op) = args.into_parts();
430        let req = match op {
431            PresignOperation::Stat(v) => self.core.tos_head_object_request(path, v),
432            PresignOperation::Read(range, v) => self.core.tos_get_object_request(path, range, &v),
433            PresignOperation::Write(v) => {
434                self.core
435                    .tos_put_object_request(path, None, &v, Buffer::new())
436            }
437            PresignOperation::Delete(v) => self.core.tos_delete_object_request(path, &v),
438            _ => Err(Error::new(
439                ErrorKind::Unsupported,
440                "operation is not supported",
441            )),
442        }?;
443        let req = self.core.sign_query(ctx, req, expire).await?;
444        let (parts, _) = req.into_parts();
445
446        Ok(RpPresign::new(PresignedRequest::new(
447            parts.method,
448            parts.uri,
449            parts.headers,
450        )))
451    }
