34/// AsyncReader implements AsyncFileReader trait by using opendal.
35///
36/// ```no_run
37/// use std::sync::Arc;
38/// use arrow::array::{ArrayRef, Int64Array, RecordBatch};
39///
40/// use futures::StreamExt;
41/// use opendal::{services::S3Config, Operator};
42/// use parquet::arrow::{AsyncArrowWriter, ParquetRecordBatchStreamBuilder};
43/// use parquet_opendal::{AsyncReader, AsyncWriter};
44///
45/// #[tokio::main]
46/// async fn main() {
47///     let mut cfg = S3Config::default();
48///     cfg.access_key_id = Some("my_access_key".to_string());
49///     cfg.secret_access_key = Some("my_secret_key".to_string());
50///     cfg.endpoint = Some("my_endpoint".to_string());
51///     cfg.region = Some("my_region".to_string());
52///     cfg.bucket = "my_bucket".to_string();
53///
54///     // Create a new operator
55///     let operator = Operator::from_config(cfg).unwrap();
56///     let path = "/path/to/file.parquet";
57///
58///     // Create an async writer
59///     let writer = AsyncWriter::new(
60///         operator
61///             .writer_with(path)
62///             .chunk(32 * 1024 * 1024)
63///             .concurrent(8)
64///             .await
65///             .unwrap(),
66///     );
67///
68///     let col = Arc::new(Int64Array::from_iter_values([1, 2, 3])) as ArrayRef;
69///     let to_write = RecordBatch::try_from_iter([("col", col)]).unwrap();
70///     let mut writer = AsyncArrowWriter::try_new(writer, to_write.schema(), None).unwrap();
71///     writer.write(&to_write).await.unwrap();
72///     writer.close().await.unwrap();
73///
74///     // gap: Allow the underlying reader to merge small IOs
75///     // when the gap between multiple IO ranges is less than the threshold.
76///     let reader = operator
77///         .reader_with(path)
78///         .gap(512 * 1024)
79///         .chunk(16 * 1024 * 1024)
80///         .concurrent(16)
81///         .await
82///         .unwrap();
83///     let content_len = operator.stat(path).await.unwrap().content_length();
84///     let reader = AsyncReader::new(reader, content_len).with_prefetch_footer_size(512 * 1024);
85///     let mut stream = ParquetRecordBatchStreamBuilder::new(reader)
86///         .await
87///         .unwrap()
88///         .build()
89///         .unwrap();
90///     let read = stream.next().await.unwrap().unwrap();
91///     assert_eq!(to_write, read);
92/// }
93/// ```
94pub struct AsyncReader {
95    inner: Reader,
96    content_length: u64,
97    // The prefetch size for fetching file footer.
98    prefetch_footer_size: usize,
99}
100
101fn set_prefetch_footer_size(footer_size: usize, content_size: u64) -> usize {
102    let footer_size = max(footer_size, FOOTER_SIZE);
103    min(footer_size as u64, content_size) as usize
104}
105
106impl AsyncReader {
107    /// Create a [`AsyncReader`] by given [`Reader`].
108    pub fn new(reader: Reader, content_length: u64) -> Self {
109        Self {
110            inner: reader,
111            content_length,
112            prefetch_footer_size: set_prefetch_footer_size(PREFETCH_FOOTER_SIZE, content_length),
113        }
114    }
115
116    /// Set prefetch size for fetching file footer.
117    pub fn with_prefetch_footer_size(mut self, footer_size: usize) -> Self {
118        self.prefetch_footer_size = set_prefetch_footer_size(footer_size, self.content_length);
119        self
120    }
121}
122
123impl AsyncFileReader for AsyncReader {
124    fn get_bytes(&mut self, range: Range<u64>) -> BoxFuture<'_, ParquetResult<bytes::Bytes>> {
125        async move {
126            Ok(self
127                .inner
128                .read(range)
129                .await
130                .map_err(|err| ParquetError::External(Box::new(err)))?
131                .to_bytes())
132        }
133        .boxed()
134    }
135
136    fn get_byte_ranges(
137        &mut self,
138        ranges: Vec<Range<u64>>,
139    ) -> BoxFuture<'_, ParquetResult<Vec<bytes::Bytes>>> {
140        async move {
141            Ok(self
142                .inner
143                .fetch(ranges)
144                .await
145                .map_err(|err| ParquetError::External(Box::new(err)))?
146                .into_iter()
147                .map(|buf| buf.to_bytes())
148                .collect::<Vec<_>>())
149        }
150        .boxed()
151    }
152
153    fn get_metadata<'a>(
154        &'a mut self,
155        _options: Option<&'a ArrowReaderOptions>,
156    ) -> BoxFuture<'a, ParquetResult<std::sync::Arc<ParquetMetaData>>> {
157        let content_length = self.content_length;
158        let prefetch_footer_size = self.prefetch_footer_size;
159        async move {
160            let reader =
161                ParquetMetaDataReader::new().with_prefetch_hint(Some(prefetch_footer_size));
162            let meta = reader.load_and_finish(self, content_length).await?;
163
164            Ok(Arc::new(meta))
165        }
166        .boxed()
167    }
168}

186    #[tokio::test]
187    async fn test_async_reader_with_prefetch_footer_size() {
188        let operator = Operator::new(services::Memory::default()).unwrap();
189        let path = "/path/to/file.parquet";
190
191        let reader = AsyncReader::new(operator.reader(path).await.unwrap(), 1024);
192        assert_eq!(reader.prefetch_footer_size, 1024);
193        assert_eq!(reader.content_length, 1024);
194
195        let reader = AsyncReader::new(operator.reader(path).await.unwrap(), 1024 * 1024);
196        assert_eq!(reader.prefetch_footer_size, PREFETCH_FOOTER_SIZE);
197        assert_eq!(reader.content_length, 1024 * 1024);
198
199        let reader = AsyncReader::new(operator.reader(path).await.unwrap(), 1024 * 1024)
200            .with_prefetch_footer_size(2048 * 1024);
201        assert_eq!(reader.prefetch_footer_size, 1024 * 1024);
202        assert_eq!(reader.content_length, 1024 * 1024);
203
204        let reader = AsyncReader::new(operator.reader(path).await.unwrap(), 1024 * 1024)
205            .with_prefetch_footer_size(1);
206        assert_eq!(reader.prefetch_footer_size, 8);
207        assert_eq!(reader.content_length, 1024 * 1024);
208    }

210    fn gen_fixed_string(size: usize) -> String {
211        rand::rng()
212            .sample_iter(&Alphanumeric)
213            .take(size)
214            .map(char::from)
215            .collect()
216    }

218    #[tokio::test]
219    async fn test_async_reader() {
220        let operator = Operator::new(services::Memory::default()).unwrap();
221        let path = "/path/to/file.parquet";
222        let writer = AsyncWriter::new(

249    struct TestCase {
250        metadata_size: usize,
251        prefetch: Option<usize>,

254    #[tokio::test]
255    async fn test_async_reader_with_large_metadata() {
256        for case in [
257            TestCase {
258                metadata_size: 256 * 1024,
259                prefetch: None,
260            },
261            TestCase {
262                metadata_size: 1024 * 1024,
263                prefetch: None,
264            },
265            TestCase {
266                metadata_size: 256 * 1024,
267                prefetch: Some(4),
268            },
269            TestCase {
270                metadata_size: 1024 * 1024,
271                prefetch: Some(4),
272            },
273        ] {
274            let operator = Operator::new(services::Memory::default()).unwrap();
275            let path = "/path/to/file.parquet";
276            let writer = AsyncWriter::new(
277                operator
278                    .writer_with(path)
279                    .chunk(32 * 1024 * 1024)
280                    .concurrent(8)
281                    .await
282                    .unwrap(),
283            );
284
285            let col = Arc::new(Int64Array::from_iter_values([1, 2, 3])) as ArrayRef;
286            let to_write = RecordBatch::try_from_iter([("col", col)]).unwrap();
287
288            let mut writer = AsyncArrowWriter::try_new(
289                writer,
290                to_write.schema(),
291                Some(
292                    WriterProperties::builder()
293                        .set_key_value_metadata(Some(vec![KeyValue {
294                            key: "__metadata".to_string(),
295                            value: Some(gen_fixed_string(case.metadata_size)),
296                        }]))
297                        .build(),
298                ),
299            )
300            .unwrap();
301            writer.write(&to_write).await.unwrap();
302            writer.close().await.unwrap();
303
304            let reader = operator.reader(path).await.unwrap();
305            let content_len = operator.stat(path).await.unwrap().content_length();
306            let mut reader = AsyncReader::new(reader, content_len);
307            if let Some(footer_size) = case.prefetch {
308                reader = reader.with_prefetch_footer_size(footer_size);
309            }
310            let mut stream = ParquetRecordBatchStreamBuilder::new(reader)
311                .await
312                .unwrap()
313                .build()
314                .unwrap();
315            let read = stream.next().await.unwrap().unwrap();
316            assert_eq!(to_write, read);
317        }
318    }
