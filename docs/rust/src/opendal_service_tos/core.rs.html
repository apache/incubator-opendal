103    pub async fn sign_query<T>(
104        &self,
105        ctx: &OperationContext,
106        req: Request<T>,
107        duration: Duration,
108    ) -> Result<Request<T>> {
109        if self.skip_signature {
110            return Ok(req);
111        }
112
113        let (mut parts, body) = req.into_parts();
114        self.signer(ctx)
115            .sign(&mut parts, Some(duration))
116            .await
117            .map_err(|e| new_request_sign_error(e.into()))?;
118
119        parts.headers.remove(HOST);
120
121        Ok(Request::from_parts(parts, body))
122    }
123
124    pub async fn send(
125        &self,
126        ctx: &OperationContext,
127        req: Request<Buffer>,
128    ) -> Result<Response<Buffer>> {
129        let (mut parts, body) = req.into_parts();
130
131        if !self.skip_signature {
132            self.signer(ctx)
133                .sign(&mut parts, None)
134                .await
135                .map_err(|e| new_request_sign_error(e.into()))?;
136        }
137
138        parts.headers.insert(
139            USER_AGENT,
140            HeaderValue::from_str(&format!("opendal/{VERSION}"))
141                .expect("user agent must be valid header value"),
142        );
143
144        let resp = ctx
145            .http_transport()
146            .send(Request::from_parts(parts, body))
147            .await?;
148
149        Ok(resp)
150    }

179    pub fn insert_metadata_headers(
180        &self,
181        mut req: http::request::Builder,
182        size: Option<u64>,
183        args: &OpWrite,
184    ) -> http::request::Builder {
185        if let Some(size) = size {
186            req = req.header(CONTENT_LENGTH, size.to_string());
187        }
188
189        if let Some(mime) = args.content_type() {
190            req = req.header(CONTENT_TYPE, mime);
191        }
192
193        if let Some(cache_control) = args.cache_control() {
194            req = req.header(http::header::CACHE_CONTROL, cache_control);
195        }
196
197        if let Some(content_encoding) = args.content_encoding() {
198            req = req.header(http::header::CONTENT_ENCODING, content_encoding);
199        }
200
201        if let Some(content_disposition) = args.content_disposition() {
202            req = req.header(http::header::CONTENT_DISPOSITION, content_disposition);
203        }
204
205        if let Some(if_match) = args.if_match() {
206            req = req.header(http::header::IF_MATCH, if_match);
207        }
208
209        if args.if_not_exists() {
210            req = req.header(http::header::IF_NONE_MATCH, "*");
211        }
212
213        if let Some(v) = &self.default_storage_class {
214            req = req.header(
215                http::HeaderName::from_static(constants::X_TOS_STORAGE_CLASS),
216                v,
217            );
218        }
219
220        if let Some(user_metadata) = args.user_metadata() {
221            for (key, value) in user_metadata {
222                req = req.header(format!("{}{}", constants::X_TOS_META_PREFIX, key), value);
223            }
224        }
225
226        req
227    }

229    pub fn tos_get_object_request(
230        &self,
231        path: &str,
232        range: BytesRange,
233        args: &OpRead,
234    ) -> Result<Request<Buffer>> {
235        let p = build_abs_path(&self.root, path);
236
237        let mut url = format!(
238            "https://{}.{}/{}",
239            self.bucket,
240            self.endpoint_domain,
241            percent_encode_path(&p)
242        );
243
244        let mut query_args = Vec::new();
245        if let Some(override_content_disposition) = args.override_content_disposition() {
246            query_args.push(format!(
247                "{}={}",
248                constants::RESPONSE_CONTENT_DISPOSITION,
249                percent_encode_query(override_content_disposition)
250            ));
251        }
252        if let Some(override_content_type) = args.override_content_type() {
253            query_args.push(format!(
254                "{}={}",
255                constants::RESPONSE_CONTENT_TYPE,
256                percent_encode_query(override_content_type)
257            ));
258        }
259        if let Some(override_cache_control) = args.override_cache_control() {
260            query_args.push(format!(
261                "{}={}",
262                constants::RESPONSE_CACHE_CONTROL,
263                percent_encode_query(override_cache_control)
264            ));
265        }
266        if let Some(version) = args.version() {
267            query_args.push(format!(
268                "{}={}",
269                constants::TOS_QUERY_VERSION_ID,
270                percent_decode_path(version)
271            ));
272        }
273        if !query_args.is_empty() {
274            url.push_str(&format!("?{}", query_args.join("&")));
275        }
276
277        let mut req = Request::get(&url);
278
279        if !range.is_full() {
280            req = req.header(http::header::RANGE, range.to_header());
281        }
282
283        if let Some(if_none_match) = args.if_none_match() {
284            req = req.header(http::header::IF_NONE_MATCH, if_none_match);
285        }
286
287        if let Some(if_match) = args.if_match() {
288            req = req.header(http::header::IF_MATCH, if_match);
289        }
290
291        if let Some(if_modified_since) = args.if_modified_since() {
292            req = req.header(
293                http::header::IF_MODIFIED_SINCE,
294                if_modified_since.format_http_date(),
295            );
296        }
297
298        if let Some(if_unmodified_since) = args.if_unmodified_since() {
299            req = req.header(
300                http::header::IF_UNMODIFIED_SINCE,
301                if_unmodified_since.format_http_date(),
302            );
303        }
304
305        req = req
306            .extension(Operation::Read)
307            .extension(ServiceOperation("GetObject"));
308
309        let req = req.body(Buffer::new()).map_err(new_request_build_error)?;
310
311        Ok(req)
312    }

485    pub async fn tos_delete_objects(
486        &self,
487        ctx: &OperationContext,
488        paths: &[(String, OpDelete)],
489    ) -> Result<Response<Buffer>> {
490        let url = format!("https://{}.{}?delete", self.bucket, self.endpoint_domain);
491
492        let mut req = Request::post(&url);
493
494        let content = serde_json::to_string(&DeleteObjectsRequest {
495            objects: paths
496                .iter()
497                .map(|(path, op)| DeleteObjectsRequestObject {
498                    key: build_abs_path(&self.root, path),
499                    version_id: op.version().map(|v| v.to_owned()),
500                })
501                .collect(),
502            quiet: false,
503        })
504        .map_err(new_json_serialize_error)?;
505
506        req = req.header(CONTENT_LENGTH, content.len());
507        req = req.header(CONTENT_TYPE, "application/json");
508        req = req.header("CONTENT-MD5", format_content_md5(content.as_bytes()));
509
510        req = req
511            .extension(Operation::Delete)
512            .extension(ServiceOperation("DeleteMultipleObjects"));
513
514        let req = req
515            .body(Buffer::from(content))
516            .map_err(new_request_build_error)?;
517
518        self.send(ctx, req).await
519    }

679    pub async fn tos_list_objects_v2(
680        &self,
681        ctx: &OperationContext,
682        path: &str,
683        continuation_token: &str,
684        delimiter: &str,
685        limit: Option<usize>,
686        start_after: Option<String>,
687    ) -> Result<Response<Buffer>> {
688        let p = build_abs_path(&self.root, path);
689
690        let mut url =
691            QueryPairsWriter::new(&format!("https://{}.{}", self.bucket, self.endpoint_domain));
692        url = url.push("list-type", "2");
693
694        if !p.is_empty() {
695            url = url.push("prefix", &percent_encode_query(&p));
696        }
697        if !delimiter.is_empty() {
698            url = url.push("delimiter", &percent_encode_query(delimiter));
699        }
700        if let Some(limit) = limit {
701            url = url.push("max-keys", &limit.to_string());
702        }
703        if let Some(start_after) = start_after
704            && (path.is_empty() || path == "/" || start_after.starts_with(path))
705        {
706            let start_after = build_abs_path(&self.root, &start_after);
707            url = url.push("start-after", &percent_encode_query(&start_after));
708        }
709        if !continuation_token.is_empty() {
710            url = url.push(
711                "continuation-token",
712                &percent_encode_query(continuation_token),
713            );
714        }
715

885            .extension(Operation::Write)
886            .extension(ServiceOperation("CompleteMultipartUpload"))
887            .body(Buffer::from(content))
888            .map_err(new_request_build_error)?;
889
890        self.send(ctx, req).await
891    }

1098/// Context needed to classify an error from this service.
1099#[derive(Clone, Copy, Debug)]
1100pub(crate) struct ErrorContext {
1101    service_operation: ServiceOperation,
1102}
1103
1104impl ErrorContext {
1105    pub(crate) const fn new(service_operation: ServiceOperation) -> Self {
1106        Self { service_operation }
1107    }
1108}

1110/// Parse an error response using its service request context.
1111pub(crate) fn parse_error(ctx: ErrorContext, resp: Response<Buffer>) -> Error {
1112    let (parts, body) = resp.into_parts();
1113    let bs = body.to_bytes();
1114
1115    let (mut kind, mut retryable) = match parts.status {
1116        StatusCode::FORBIDDEN => (ErrorKind::PermissionDenied, false),
1117        StatusCode::NOT_FOUND => (ErrorKind::NotFound, false),
1118        StatusCode::NOT_MODIFIED | StatusCode::PRECONDITION_FAILED => {
1119            (ErrorKind::ConditionNotMatch, false)
1120        }
1121        StatusCode::CONFLICT => (ErrorKind::ConditionNotMatch, true),
1122        StatusCode::TOO_MANY_REQUESTS => (ErrorKind::RateLimited, true),
1123        code if code.is_server_error() => (ErrorKind::Unexpected, true),
1124        _ => (ErrorKind::Unexpected, false),
1125    };
1126
1127    let (message, tos_err) = parse_tos_error(&bs)
1128        .map(|tos_err| (format!("{tos_err:?}"), Some(tos_err)))
1129        .unwrap_or_else(|| (String::from_utf8_lossy(&bs).into_owned(), None));
1130
1131    if let Some(tos_err) = tos_err {
1132        (kind, retryable) =
1133            parse_tos_error_code(tos_err.code.as_str()).unwrap_or((kind, retryable));
1134    }
1135
1136    let mut err = Error::new(kind, message);
1137
1138    err = err.with_context("service_operation", ctx.service_operation.0);
1139    err = with_error_response_context(err, parts);
1140
1141    if retryable {
1142        err = err.set_temporary();
1143    }
1144
1145    err
1146}
1147
1148fn parse_tos_error(bs: &bytes::Bytes) -> Option<TosError> {
1149    de::from_reader::<_, TosError>(bs.chunk().reader())
1150        .or_else(|_| serde_json::from_slice::<TosError>(bs))
1151        .ok()
1152}
1153
1154/// Returns the `Error kind` of this code and whether the error is retryable.
1155/// TOS error codes: https://www.volcengine.com/docs/6349/74874
1156pub fn parse_tos_error_code(code: &str) -> Option<(ErrorKind, bool)> {
1157    match code {
1158        "NoSuchBucket" => Some((ErrorKind::ConfigInvalid, false)),
1159        "NoSuchKey" => Some((ErrorKind::NotFound, false)),
1160        "DuplicateObject" => Some((ErrorKind::ConditionNotMatch, false)),
1161        "RequestTimeout" => Some((ErrorKind::Unexpected, true)),
1162        "InternalError" => Some((ErrorKind::Unexpected, true)),
1163        "OperationAborted" => Some((ErrorKind::Unexpected, true)),
1164        "SlowDown" => Some((ErrorKind::RateLimited, true)),
1165        "ServiceUnavailable" => Some((ErrorKind::Unexpected, true)),
1166        "TooManyRequests" => Some((ErrorKind::RateLimited, true)),
1167        "ExceedAccountQPSLimit" | "ExceedAccountRateLimit" => Some((ErrorKind::RateLimited, true)),
1168        "ExceedBucketQPSLimit" | "ExceedBucketRateLimit" => Some((ErrorKind::RateLimited, true)),
1169        _ => None,
1170    }
1171}

1237    /// Parse etag from header map.

1239    /// Trim quotes from etag if present (TOS returns etag with quotes)
1240    pub fn tos_parse_etag(headers: &HeaderMap) -> opendal_core::Result<Option<&str>> {
1241        let etag = parse_header_to_str(headers, ETAG)?.map(|etag| etag.trim_matches('"'));
1242        Ok(etag)
1243    }

1245    pub(crate) fn tos_parse_into_metadata(
1246        path: &str,
1247        headers: &HeaderMap,
1248    ) -> opendal_core::Result<Metadata> {
1249        let mut metadata = parse_into_metadata(path, headers)?.into_builder();
1250
1251        if let Some(etag) = tos_parse_etag(headers)? {
1252            metadata.etag(etag);
1253        }
1254
1255        Ok(metadata.build())
1256    }
