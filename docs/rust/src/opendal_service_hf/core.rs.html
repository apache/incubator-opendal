255impl PathInfo {
256    pub fn entry_mode(&self) -> EntryMode {
257        match self.type_.as_str() {
258            "directory" => EntryMode::DIR,
259            "file" => EntryMode::FILE,
260            _ => EntryMode::Unknown,
261        }
262    }
263
264    pub fn metadata(&self) -> Result<Metadata> {
265        let mode = self.entry_mode();
266        let mut meta = match mode {
267            EntryMode::FILE => MetadataBuilder::file(self.size),
268            EntryMode::DIR => MetadataBuilder::dir(),
269            EntryMode::Unknown => MetadataBuilder::unknown(),
270        };
271
272        if let Some(commit_info) = self.last_commit.as_ref() {
273            meta.last_modified(commit_info.date.parse::<Timestamp>()?);
274        }
275
276        if mode == EntryMode::FILE {
277            // For buckets, oid may be None; for regular repos, prefer lfs.oid then oid
278            if let Some(lfs) = &self.lfs {
279                meta.etag(&lfs.oid);
280            } else if let Some(oid) = &self.oid {
281                meta.etag(oid);
282            }
283        }
284
285        Ok(meta.build())
286    }

317// Bound entries, paths, and HTTP URIs. Busy entries are never evicted:
318// recreating their slots would allow two concurrent resolutions for one path.
319const RESOLVED_FILE_MAX_ENTRIES: usize = 512;
320const RESOLVED_FILE_MAX_PATH_BYTES: usize = 4096;
321const HTTP_DOWNLOAD_MAX_URI_BYTES: usize = 8192;
322const HTTP_DOWNLOAD_EXPIRY_MARGIN: Duration = Duration::from_secs(30);
323
324#[derive(Default)]
325struct ResolvedFiles {
326    entries: StdMutex<HashMap<String, ResolvedFileEntry>>,
327}
328
329struct ResolvedFileEntry {
330    accessed: Instant,
331    state: Arc<Mutex<ResolvedFile>>,
332}
333
334#[derive(Default)]
335enum ResolvedFile {
336    #[default]
337    Empty,
338    Http(Arc<HttpDownload>),
339    Xet(XetFileResponse),
340    // The path is not XET-backed and has no reusable HTTP destination.
341    NotXet,
342    // Wake existing waiters onto the uncached path when a response cannot be
343    // admitted. Its slot leaves the table so later reads can try again.
344    Bypass,
345}
346
347struct HttpDownload {
348    redirect: HttpRedirect,
349    valid_until: Instant,
350}
351
352impl ResolvedFiles {
353    fn entry(&self, path: &str) -> Option<Arc<Mutex<ResolvedFile>>> {
354        if path.len() > RESOLVED_FILE_MAX_PATH_BYTES {
355            return None;
356        }
357
358        let mut entries = self
359            .entries
360            .lock()
361            .expect("resolved file cache lock poisoned");
362        if entries.len() == RESOLVED_FILE_MAX_ENTRIES && !entries.contains_key(path) {
363            let oldest = entries
364                .iter()
365                .filter(|(_, entry)| Arc::strong_count(&entry.state) == 1)
366                .min_by_key(|(_, entry)| entry.accessed)
367                .map(|(path, _)| path.clone())?;
368            entries.remove(&oldest);
369        }
370        let entry = entries
371            .entry(path.to_string())
372            .or_insert_with(|| ResolvedFileEntry {
373                accessed: Instant::now(),
374                state: Arc::default(),
375            });
376        entry.accessed = Instant::now();
377        Some(entry.state.clone())
378    }
379}
380
381impl HttpDownload {
382    fn from_response(resp: &Response<HttpBody>) -> Option<Self> {
383        let redirect = resp.extensions().get::<HttpRedirect>()?;
384        let uri = redirect.uri().original_uri().parse::<http::Uri>().ok()?;
385        if uri.scheme_str() != Some("https")
386            || uri.authority()?.as_str().contains('@')
387            || redirect.uri().original_uri().len() > HTTP_DOWNLOAD_MAX_URI_BYTES
388        {
389            return None;
390        }
391
392        // HF's signed CDN destinations advertise their authorization deadline
393        // as Expires. Unknown signature formats fall back to fresh resolution.
394        let mut expirations = uri.query()?.split('&').filter_map(|pair| {
395            let (key, value) = pair.split_once('=')?;
396            (key == "Expires").then_some(value)
397        });
398        let expires = expirations.next()?.parse::<u64>().ok()?;
399        if expirations.next().is_some() {
400            return None;
401        }
402        let now = SystemTime::now().duration_since(UNIX_EPOCH).ok()?;
403        let lifetime = Duration::from_secs(expires)
404            .checked_sub(now)?
405            .checked_sub(HTTP_DOWNLOAD_EXPIRY_MARGIN)?;
406        if lifetime.is_zero() {
407            return None;
408        }
409
410        Some(Self {
411            redirect: redirect.clone(),
412            valid_until: Instant::now().checked_add(lifetime)?,
413        })
414    }
415}

515    /// Create a new XET upload commit, seeded with a cached CAS write token

545    /// Get a still-valid cached token for `scope`, fetching and caching a
546    /// fresh one if missing or close to expiry. The lock is held across the
547    /// refresh request so concurrent callers single-flight onto one fetch.
548    ///
549    /// On a clock failure, `now` falls back to `u64::MAX` so the token is
550    /// treated as stale (extra `/api` traffic) rather than `0`, which would
551    /// pin a stale token as valid forever.
552    async fn cached_xet_token(
553        &self,
554        ctx: &OperationContext,
555        scope: XetTokenScope,
556    ) -> Result<XetToken> {
557        let cache = match scope {
558            XetTokenScope::Read => &self.xet_read_token,
559            XetTokenScope::Write => &self.xet_write_token,
560        };
561        let mut cached = cache.lock().await;
562
563        let now = SystemTime::now()
564            .duration_since(UNIX_EPOCH)
565            .map(|d| d.as_secs())
566            .unwrap_or(u64::MAX);
567        if let Some(token) = cached.as_ref()
568            && token.expires_at > now.saturating_add(XET_TOKEN_REFRESH_BUFFER_SECS)
569        {
570            return Ok(token.clone());
571        }
572
573        let url = self.repo.xet_token_url(&self.endpoint, scope);
574        let req = self
575            .request(http::Method::GET, &url, scope.operation(), "XetToken")?
576            .body(Buffer::new())
577            .map_err(new_request_build_error)?;
578        let resp = self.send(ctx, req).await?;
579        if !resp.status().is_success() {
580            let (parts, _) = resp.into_parts();
581            return Err(parse_error(
582                ErrorContext::new(ServiceOperation("XetToken")),
583                parts,
584            ));
585        }
586        let (_, mut body) = resp.into_parts();
587        let buffer = body.to_buffer().await?;
588        let info: XetTokenResponse =
589            serde_json::from_reader(buffer.reader()).map_err(new_json_deserialize_error)?;
590        let fresh = XetToken {
591            cas_url: info.cas_url,
592            access_token: info.access_token,
593            expires_at: info.exp,
594        };
595        *cached = Some(fresh.clone());
596        Ok(fresh)
597    }

638    /// Build an authenticated HTTP request.
639    ///
640    /// Returns `PermissionDenied` for write operations when no token
641    /// is configured.
642    pub(super) fn request(
643        &self,
644        method: http::Method,
645        url: &str,
646        op: Operation,
647        service_operation: &'static str,
648    ) -> Result<http::request::Builder> {
649        let url = HttpUri::new(url);
650        debug!(
651            "hf request: service_operation={service_operation} operation={op} method={method} url={}",
652            url.redacted_uri()
653        );
654        let mut req = Request::builder()
655            .method(method)
656            .uri(url.original_uri())
657            .extension(op)
658            .extension(ServiceOperation(service_operation));
659        match &self.token {
660            Some(token) => {
661                if let Ok(auth) = format_authorization_by_bearer(token) {
662                    req = req.header(header::AUTHORIZATION, auth);
663                }
664            }
665            None if matches!(op, Operation::Write | Operation::Delete) => {
666                return Err(Error::new(
667                    ErrorKind::PermissionDenied,
668                    "token is required for write operations",
669                ));
670            }
671            None => {}
672        }
673        Ok(req)
674    }

676    /// Send `req`, following one same-endpoint `307`/`308` redirect.
677    ///
678    /// HF resolves repo ids case-insensitively but answers any request for
679    /// a non-canonically cased id (`user/repo` for `user/Repo`) with a 307
680    /// to the canonical URL. Transports follow that for `GET`s but hand a
681    /// bodied `POST` (commit, paths-info) back as the bare 307, which would
682    /// otherwise surface as an error while reads silently succeed.
683    /// Re-issuing the request here makes the outcome independent of the
684    /// transport's redirect policy. `send` returns a redirect to any other
685    /// host as-is, so it never sends the bearer token off the configured
686    /// endpoint itself; the caller then reports it via [`parse_error`].
687    pub(super) async fn send(
688        &self,
689        ctx: &OperationContext,
690        req: Request<Buffer>,
691    ) -> Result<Response<HttpBody>> {
692        let retry = req.clone();
693
694        // The response is confined to this block so it is gone before the
695        // retry runs: its body is never read, so the connection cannot be
696        // pooled and is better closed than held across another round trip.
697        let target = {
698            let resp = ctx.http_transport().fetch(req).await?;
699            if !matches!(
700                resp.status(),
701                StatusCode::TEMPORARY_REDIRECT | StatusCode::PERMANENT_REDIRECT
702            ) {
703                return Ok(resp);
704            }
705
706            // Accept a path-only `Location`, which is what HF sends, or one
707            // already on the endpoint (a bare origin, normalized by
708            // `HfBuilder`). What remains after dropping the endpoint must be
709            // a single-slash path, which also refuses a protocol-relative
710            // `//host/...` naming another host.
711            let Some(location) = parse_location(resp.headers())? else {
712                return Ok(resp);
713            };
714            let path = location
715                .strip_prefix(self.endpoint.as_str())
716                .unwrap_or(location);
717            if !path.starts_with('/') || path.starts_with("//") {
718                return Ok(resp);
719            }
720            format!("{}{path}", self.endpoint)
721        };
722
723        let target = HttpUri::new(target);
724        debug!("hf request redirected: url={}", target.redacted_uri());
725        let (mut parts, body) = retry.into_parts();
726        parts.uri = target
727            .original_uri()
728            .parse()
729            .map_err(new_http_uri_invalid_error)?;
730        ctx.http_transport()
731            .fetch(Request::from_parts(parts, body))
732            .await
733    }

743    pub(super) async fn path_info(&self, ctx: &OperationContext, path: &str) -> Result<PathInfo> {
744        let uri = self.repo.uri(&self.root, path);
745        let url = uri.paths_info_url(&self.endpoint);
746        let form_body = format!("paths={}&expand=True", percent_encode_path(&uri.path));
747
748        let req = self
749            .request(http::Method::POST, &url, Operation::Stat, "PathInfo")?
750            .header(header::CONTENT_TYPE, "application/x-www-form-urlencoded")
751            .body(Buffer::from(Bytes::from(form_body)))
752            .map_err(new_request_build_error)?;
753        let resp = self.send(ctx, req).await?;
754        if !resp.status().is_success() {
755            let (parts, _) = resp.into_parts();
756            return Err(parse_error(
757                ErrorContext::new(ServiceOperation("PathInfo")),
758                parts,
759            ));
760        }
761        let (_, mut body) = resp.into_parts();
762        let buffer = body.to_buffer().await?;
763        let mut files: Vec<PathInfo> =
764            serde_json::from_reader(buffer.reader()).map_err(new_json_deserialize_error)?;
765
766        // NOTE: if the file is not found, the server will return 200 with an empty array
767        if files.is_empty() {
768            return Err(Error::new(ErrorKind::NotFound, "path not found"));
769        }
770
771        Ok(files.remove(0))
772    }

774    pub(super) async fn read(
775        &self,
776        ctx: &OperationContext,
777        path: &str,
778        range: BytesRange,
779    ) -> Result<HfReadResponse> {
780        let entry = if self.enable_resolve_cache {
781            self.resolved_files.entry(path)
782        } else {
783            None
784        };
785        let Some(entry) = entry else {
786            let resp = self.resolve(ctx, path, range, self.download_mode).await?;
787            return HfReadResponse::from_response(resp, self.download_mode).await;
788        };
789
790        let mut rejected: Option<Arc<HttpDownload>> = None;
791        loop {
792            let destination = {
793                let mut state = entry.lock().await;
794                match &*state {
795                    ResolvedFile::Xet(info) => return Ok(HfReadResponse::Xet(info.clone())),
796                    ResolvedFile::NotXet | ResolvedFile::Bypass => {
797                        drop(state);
798                        let resp = self.resolve(ctx, path, range, HfDownloadMode::Http).await?;
799                        return Ok(HfReadResponse::Http(resp));
800                    }
801                    ResolvedFile::Http(current)
802                        if current.valid_until > Instant::now()
803                            && !rejected
804                                .as_ref()
805                                .is_some_and(|old| Arc::ptr_eq(old, current)) =>
806                    {
807                        current.clone()
808                    }
809                    _ => {
810                        let mode = if matches!(&*state, ResolvedFile::Http(_)) {
811                            HfDownloadMode::Http
812                        } else {
813                            self.download_mode
814                        };
815                        // Serialize resolution and refresh. XET metadata is
816                        // independent of the range; a one-byte probe bounds the
817                        // body discarded when the file turns out to use HTTP.
818                        *state = ResolvedFile::Empty;
819                        let resolve_range = if mode == HfDownloadMode::Xet {
820                            BytesRange::new(0, Some(1))
821                        } else {
822                            range
823                        };
824                        let resp = self.resolve(ctx, path, resolve_range, mode).await?;
825                        let resp = match HfReadResponse::from_response(resp, mode).await? {
826                            HfReadResponse::Xet(info) => {
827                                *state = ResolvedFile::Xet(info.clone());
828                                return Ok(HfReadResponse::Xet(info));
829                            }
830                            HfReadResponse::Http(resp) => resp,
831                        };
832                        *state = match HttpDownload::from_response(&resp) {
833                            Some(download) => ResolvedFile::Http(Arc::new(download)),
834                            None if self.download_mode == HfDownloadMode::Xet => {
835                                ResolvedFile::NotXet
836                            }
837                            None => {
838                                // This live slot cannot have been evicted or
839                                // replaced while we hold an Arc to it.
840                                self.resolved_files
841                                    .entries
842                                    .lock()
843                                    .expect("resolved file cache lock poisoned")
844                                    .remove(path);
845                                ResolvedFile::Bypass
846                            }
847                        };
848                        if mode == HfDownloadMode::Http {
849                            // The response already contains the requested range.
850                            return Ok(HfReadResponse::Http(resp));
851                        }
852                        // The XET probe returned one byte of HTTP content.
853                        // Reuse its destination to fetch the caller's range.
854                        continue;
855                    }
856                }
857            };
858
859            let url = self
860                .repo
861                .uri(&self.root, path)
862                .resolve_url(&self.endpoint, self.repo.revision());
863            let mut req = self
864                .request(http::Method::GET, &url, Operation::Read, "Download")?
865                .extension(destination.redirect.clone());
866            if !range.is_full() {
867                req = req.header(header::RANGE, range.to_header());
868            }
869            let req = req.body(Buffer::new()).map_err(new_request_build_error)?;
870            let resp = ctx.http_transport().fetch(req).await?;
871            if resp.status().is_success() {
872                return Ok(HfReadResponse::Http(resp));
873            }
874            if rejected.is_none()
875                && matches!(
876                    resp.status(),
877                    StatusCode::UNAUTHORIZED | StatusCode::FORBIDDEN
878                )
879            {
880                // A late rejection must not discard another caller's refreshed
881                // destination. The next iteration compares the Arc generation.
882                rejected = Some(destination);
883                continue;
884            }
885            let (parts, _) = resp.into_parts();
886            return Err(parse_error(
887                ErrorContext::new(ServiceOperation("Download")),
888                parts,
889            ));
890        }
891    }

893    /// Send `GET /resolve` and return the raw streaming response.
894    ///
895    /// In `Xet` mode adds `Accept: application/vnd.xet-fileinfo+json` so the
896    /// server returns XET metadata instead of redirecting; in `Http` mode the
897    /// redirect is followed and the file bytes are streamed directly. Either
898    /// way, the response may carry an `x-xet-hash` header -- interpreting
899    /// that is up to the caller, not this method's concern.
900    pub(super) async fn resolve(
901        &self,
902        ctx: &OperationContext,
903        path: &str,
904        range: BytesRange,
905        mode: HfDownloadMode,
906    ) -> Result<Response<HttpBody>> {
907        let uri = self.repo.uri(&self.root, path);
908        let url = uri.resolve_url(&self.endpoint, self.repo.revision());
909
910        let mut req = self.request(http::Method::GET, &url, Operation::Read, "Resolve")?;
911
912        if mode == HfDownloadMode::Xet {
913            req = req.header(header::ACCEPT, "application/vnd.xet-fileinfo+json");
914        }
915
916        if !range.is_full() {
917            req = req.header(header::RANGE, range.to_header());
918        }
919
920        let req = req.body(Buffer::new()).map_err(new_request_build_error)?;
921        let resp = self.send(ctx, req).await?;
922
923        if !resp.status().is_success() {
924            let status = resp.status();
925            // Drop the streaming body without reading it — parse_error reads
926            // only response headers, so there is no need to buffer the body
927            // (which may be a large HTML error page).
928            let (parts, _) = resp.into_parts();
929            let mut err = parse_error(ErrorContext::new(ServiceOperation("Resolve")), parts);
930            if status == http::StatusCode::NOT_FOUND && self.path_info(ctx, path).await.is_ok() {
931                err = err.set_temporary();
932            }
933            return Err(err);
934        }
935
936        Ok(resp)
937    }

1352    /// A scripted transport mirroring how HF serves a repo whose configured
1353    /// id is not canonically cased: it answers that URL with `status` and a
1354    /// `Location` of `location`, and only the canonically cased URL accepts
1355    /// the commit. Tests set both, so one transport covers
1356    /// a followed redirect, a refused one, and a non-307 status.
1357    #[derive(Clone)]
1358    struct RedirectingTransport {
1359        status: StatusCode,
1360        location: &'static str,
1361        requests: Arc<Mutex<Vec<SeenRequest>>>,
1362    }

1364    impl HttpTransport for RedirectingTransport {
1365        async fn fetch(&self, req: Request<Buffer>) -> Result<Response<HttpBody>> {
1366            let uri = req.uri().to_string();
1367            let auth = req
1368                .headers()
1369                .get(header::AUTHORIZATION)
1370                .and_then(|v| v.to_str().ok())
1371                .unwrap_or_default()
1372                .to_string();
1373            let body = String::from_utf8(req.body().to_bytes().to_vec()).unwrap();
1374            self.requests.lock().unwrap().push(SeenRequest {
1375                method: req.method().to_string(),
1376                uri: uri.clone(),
1377                auth,
1378                body,
1379            });
1380
1381            let resp = match uri.as_str() {
1382                "https://huggingface.co/api/models/test-user/uppercase-repo/commit/main" => {
1383                    Response::builder()
1384                        .status(self.status)
1385                        .header(header::LOCATION, self.location)
1386                        .header(header::CONTENT_LENGTH, 0)
1387                        .body(HttpBody::new(futures::stream::empty(), Some(0)))
1388                }
1389                "https://huggingface.co/api/models/test-user/Uppercase-Repo/commit/main" => {
1390                    Response::builder()
1391                        .status(StatusCode::OK)
1392                        .header(header::CONTENT_LENGTH, 2)
1393                        .body(HttpBody::new(
1394                            futures::stream::iter(vec![Ok(Buffer::from(Bytes::from_static(
1395                                b"{}",
1396                            )))]),
1397                            Some(2),
1398                        ))
1399                }
1400                other => panic!("unexpected request to {other}"),
1401            };
1402            Ok(resp.unwrap())
1403        }
1404    }

1866pub(crate) fn parse_error(ctx: ErrorContext, mut parts: http::response::Parts) -> Error {
1867    let location = HttpUri::from_response_location(&mut parts).cloned();
1868    // HF sets x-error-message on every error response with a short human-readable
1869    // description. Using the header avoids reading the response body, which can be
1870    // a large HTML error page (e.g. 52 KB on 404s from the /resolve/ endpoint).
1871    let error_message = parts
1872        .headers
1873        .get("x-error-message")
1874        .and_then(|v| v.to_str().ok());
1875    let message = match (error_message, location) {
1876        (Some(message), _) => message.to_string(),
1877        (None, Some(location)) if parts.status.is_redirection() => {
1878            let location = location.redacted_uri();
1879            format!("redirect to {location} not followed")
1880        }
1881        _ => "unknown error".to_string(),
1882    };
1883
1884    // HF git-style commit APIs reject stale branch snapshots with 412.
1885    // Treat this specific conflict as temporary so RetryLayer can replay
1886    // the whole write/delete close sequence on a fresh branch head.
1887    let branch_updated_conflict = parts.status == StatusCode::PRECONDITION_FAILED
1888        && message
1889            .to_ascii_lowercase()
1890            .contains("branch was updated since you opened this page");
1891
1892    let (kind, retryable) = match parts.status {
1893        StatusCode::NOT_FOUND => (ErrorKind::NotFound, false),
1894        StatusCode::UNAUTHORIZED | StatusCode::FORBIDDEN => (ErrorKind::PermissionDenied, false),
1895        StatusCode::PRECONDITION_FAILED if branch_updated_conflict => (ErrorKind::Conflict, true),
1896        StatusCode::PRECONDITION_FAILED => (ErrorKind::Unexpected, false),
1897        StatusCode::TOO_MANY_REQUESTS => (ErrorKind::RateLimited, true),
1898        StatusCode::INTERNAL_SERVER_ERROR
1899        | StatusCode::BAD_GATEWAY
1900        | StatusCode::SERVICE_UNAVAILABLE
1901        | StatusCode::GATEWAY_TIMEOUT => (ErrorKind::Unexpected, true),
1902        _ => (ErrorKind::Unexpected, false),
1903    };
1904
1905    let mut err = Error::new(kind, message);
1906
1907    err = err.with_context("service_operation", ctx.service_operation.0);
1908    err = with_error_response_context(err, parts);
1909
1910    if retryable {
1911        err = err.set_temporary();
1912    }
1913
1914    err
1915}

2026    ///

2036        /// Parse a Hugging Face path into its components.
2037        /// Path format: `[<repo_type>/]<repo_id>[@<revision>][/<path_in_repo>]`
2038        pub fn parse(path: &str) -> opendal_core::Result<Self> {
2039            if path.is_empty() {
2040                return Err(opendal_core::Error::new(
2041                    opendal_core::ErrorKind::ConfigInvalid,
2042                    "repo_id is required in uri path",
2043                )
2044                .with_context("service", HUGGINGFACE_SCHEME));
2045            }
2046
2047            let mut path = path.to_string();
2048
2049            // Strip repo_type prefix if present (e.g. "datasets/user/repo" → "user/repo")
2050            let repo_type = if let Some((first, rest)) = path.split_once('/') {
2051                if let Ok(rt) = HfRepoType::parse(first) {
2052                    path = rest.to_string();
2053                    rt
2054                } else {
2055                    HfRepoType::Model
2056                }
2057            } else if HfRepoType::parse(&path).is_ok() {
2058                return Err(opendal_core::Error::new(
2059                    opendal_core::ErrorKind::ConfigInvalid,
2060                    "repository name is required in uri path",
2061                )
2062                .with_context("service", HUGGINGFACE_SCHEME));
2063            } else {
2064                HfRepoType::Model
2065            };
2066
2067            // Parse repo_id, revision, and path_in_repo.
2068            // Path is now: <repo_id>[@<revision>][/<path_in_repo>]
2069            let (repo_id, revision, path_in_repo) = if path.contains('/') {
2070                // Check if @ appears in the first two segments (the repo_id portion).
2071                // This distinguishes "user/repo@rev/file" from "user/repo/path/to/@file".
2072                let first_two: String = path.splitn(3, '/').take(2).collect::<Vec<_>>().join("/");
2073
2074                if first_two.contains('@') {
2075                    let (repo_id, rev_and_path) = path.split_once('@').unwrap();
2076                    let rev_and_path = rev_and_path.replace("%2F", "/");
2077                    let (revision, path_in_repo) = Self::parse_revision(&rev_and_path);
2078                    (repo_id.to_string(), Some(revision), path_in_repo)
2079                } else {
2080                    let segments: Vec<_> = path.splitn(3, '/').collect();
2081                    let repo_id = format!("{}/{}", segments[0], segments[1]);
2082                    let path_in_repo = segments.get(2).copied().unwrap_or("").to_string();
2083                    (repo_id, None, path_in_repo)
2084                }
2085            } else if let Some((repo_id, rev)) = path.split_once('@') {
2086                let rev = rev.replace("%2F", "/");
2087                (repo_id.to_string(), Some(rev), String::new())
2088            } else {
2089                (path, None, String::new())
2090            };
2091
2092            Ok(Self {
2093                repo: HfRepo::new(repo_type, repo_id, revision),
2094                path: path_in_repo,
2095            })
2096        }

2098        /// Split a string after `@` into (revision, path_in_repo).
2099        /// Handles special refs like `refs/convert/parquet` and `refs/pr/10`.
2100        fn parse_revision(rev_and_path: &str) -> (String, String) {
2101            if !rev_and_path.contains('/') {
2102                return (rev_and_path.to_string(), String::new());
2103            }
2104
2105            // Match special refs: refs/(convert|pr)/<segment>
2106            if let Some(rest) = rev_and_path.strip_prefix("refs/convert/") {
2107                return match rest.split_once('/') {
2108                    Some((segment, path)) => (format!("refs/convert/{segment}"), path.to_string()),
2109                    None => (rev_and_path.to_string(), String::new()),
2110                };
2111            }
2112            if let Some(rest) = rev_and_path.strip_prefix("refs/pr/") {
2113                return match rest.split_once('/') {
2114                    Some((segment, path)) => (format!("refs/pr/{segment}"), path.to_string()),
2115                    None => (rev_and_path.to_string(), String::new()),
2116                };
2117            }
2118
2119            // Regular revision: split on first /
2120            let (rev, path) = rev_and_path.split_once('/').unwrap();
2121            (rev.to_string(), path.to_string())
2122        }

2169        /// Build the file tree API URL for this URI.
2170        pub fn file_tree_url(
2171            &self,
2172            endpoint: &str,
2173            recursive: bool,
2174            cursor: Option<&str>,
2175        ) -> String {
2176            // HF answers a trailing slash with a 302 to the slash-less URL, so
2177            // the separator lives in the segment rather than the template.
2178            let path_segment = if self.path.is_empty() {
2179                String::new()
2180            } else {
2181                format!("/{}", percent_encode_path(&self.path))
2182            };
2183
2184            let mut url = if self.repo.is_bucket() {
2185                format!(
2186                    "{}/api/buckets/{}/tree{}?expand=True",
2187                    endpoint, self.repo.repo_id, path_segment,
2188                )
2189            } else {
2190                format!(
2191                    "{}/api/{}/{}/tree/{}{}?expand=True",
2192                    endpoint,
2193                    self.repo.repo_type.as_plural_str(),
2194                    self.repo.repo_id,
2195                    percent_encode_revision(self.revision()),
2196                    path_segment,
2197                )
2198            };
2199
2200            if recursive {
2201                url.push_str("&recursive=True");
2202            } else if self.repo.is_bucket() {
2203                // Bucket tree API defaults to recursive; must opt out explicitly.
2204                url.push_str("&recursive=false");
2205            }
2206
2207            if let Some(cursor_val) = cursor {
2208                url.push_str(&format!("&cursor={}", cursor_val));
2209            }
2210
2211            url
2212        }
