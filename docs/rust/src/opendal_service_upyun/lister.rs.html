18use std::sync::Arc;
19
20use bytes::Buf;
21use opendal_core::raw::*;
22use opendal_core::*;
23
24use super::core::{ErrorContext, ListObjectsResponse, UpyunCore, parse_error};
25
26pub struct UpyunLister {
27    core: Arc<UpyunCore>,
28    ctx: OperationContext,
29
30    path: String,
31    limit: Option<usize>,
32}
33
34impl UpyunLister {
35    pub(super) fn new(
36        core: Arc<UpyunCore>,
37        ctx: OperationContext,
38        path: &str,
39        limit: Option<usize>,
40    ) -> Self {
41        UpyunLister {
42            core,
43            ctx,
44            path: path.to_string(),
45            limit,
46        }
47    }
48}
49
50impl oio::PageList for UpyunLister {
51    async fn next_page(&self, ctx: &mut oio::PageContext) -> Result<()> {
52        let resp = self
53            .core
54            .list_objects(&self.ctx, &self.path, &ctx.token, self.limit)
55            .await?;
56
57        if resp.status() == http::StatusCode::NOT_FOUND {
58            ctx.done = true;
59            return Ok(());
60        }
61
62        match resp.status() {
63            http::StatusCode::OK => {}
64            http::StatusCode::NOT_FOUND => {
65                ctx.done = true;
66                return Ok(());
67            }
68            _ => {
69                return Err(parse_error(
70                    ErrorContext::new(ServiceOperation("ListObjects")),
71                    resp,
72                ));
73            }
74        }
75
76        let bs = resp.into_body();
77
78        let response: ListObjectsResponse =
79            serde_json::from_reader(bs.reader()).map_err(new_json_deserialize_error)?;
80
81        // ref https://help.upyun.com/knowledge-base/rest_api/#e88eb7e58f96e79baee5bd95e69687e4bbb6e58897e8a1a8
82        // when iter is "g2gCZAAEbmV4dGQAA2VvZg", it means the list is done.
83        ctx.done = response.iter == "g2gCZAAEbmV4dGQAA2VvZg";
84
85        ctx.token = response.iter;
86
87        for file in response.files {
88            let path = build_abs_path(&normalize_root(&self.path), &file.name);
89
90            let entry = if file.type_field == "folder" {
91                let path = format!("{path}/");
92                oio::Entry::new(&path, MetadataBuilder::dir().build())
93            } else {
94                let mut m = MetadataBuilder::file(file.length);
95                m.content_type(file.type_field)
96                    .last_modified(Timestamp::from_second(file.last_modified)?);
97                oio::Entry::new(&path, m.build())
98            };
99
100            ctx.entries.push_back(entry);
101        }
102
103        Ok(())
104    }
105}
