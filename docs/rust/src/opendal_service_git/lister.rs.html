18use std::sync::Arc;
19
20use opendal_core::raw::*;
21use opendal_core::*;
22
23use super::core::GitCore;
24use super::core::join_error;
25
26/// Lister returned by the Git service.
27pub struct GitLister {
28    core: Arc<GitCore>,
29    base_path: String,
30    entries: Option<Vec<(String, bool, u64, i64)>>,
31    index: usize,
32}
33
34impl GitLister {
35    pub(super) fn new(core: Arc<GitCore>, base_path: String) -> Self {
36        Self {
37            core,
38            base_path,
39            entries: None,
40            index: 0,
41        }
42    }
43}
44
45impl oio::List for GitLister {
46    async fn next(&mut self) -> Result<Option<oio::Entry>> {
47        if self.entries.is_none() {
48            let core = self.core.clone();
49            let path = self.base_path.clone();
50            let entries = tokio::task::spawn_blocking(move || core.list_dir_blocking(&path))
51                .await
52                .map_err(join_error)??;
53            self.entries = Some(entries);
54        }
55
56        let entries = self
57            .entries
58            .as_ref()
59            .ok_or_else(|| Error::new(ErrorKind::Unexpected, "git lister not initialized"))?;
60        if self.index >= entries.len() {
61            return Ok(None);
62        }
63
64        let (name, is_dir, size, last_modified) = &entries[self.index];
65        self.index += 1;
66
67        let full_path = if self.base_path == "/" {
68            name.clone()
69        } else {
70            format!("{}/{}", self.base_path.trim_end_matches('/'), name)
71        };
72        let path = if *is_dir && !full_path.ends_with('/') {
73            format!("{full_path}/")
74        } else {
75            full_path
76        };
77
78        let mut builder = if *is_dir {
79            MetadataBuilder::dir()
80        } else {
81            MetadataBuilder::file(*size)
82        };
83        if let Ok(timestamp) = Timestamp::from_second(*last_modified) {
84            builder.last_modified(timestamp);
85        }
86
87        Ok(Some(oio::Entry::new(&path, builder.build())))
88    }
89}
