25/// Writer is designed to write data into given path in an asynchronous
26/// manner.
27///
28/// ## Notes
29///
30/// Please make sure either `close` or `abort` has been called before
31/// dropping the writer otherwise the data could be lost.
32///
33/// ## Usage
34///
35/// ### Write Multiple Chunks
36///
37/// Some services support to write multiple chunks of data into given path. Services that doesn't
38/// support write multiple chunks will return [`ErrorKind::Unsupported`] error when calling `write`
39/// at the second time.
40///
41/// ```
42/// use opendal_core::Operator;
43/// use opendal_core::Result;
44///
45/// async fn test(op: Operator) -> Result<()> {
46///     let mut w = op.writer("path/to/file").await?;
47///     w.write(vec![1; 1024]).await?;
48///     w.write(vec![2; 1024]).await?;
49///     w.close().await?;
50///     Ok(())
51/// }
52/// ```
53///
54/// ### Write like `Sink`
55///
56/// ```
57/// use anyhow::Result;
58/// use futures::SinkExt;
59/// use opendal_core::Operator;
60///
61/// async fn test(op: Operator) -> Result<()> {
62///     let mut w = op.writer("path/to/file").await?.into_bytes_sink();
63///     w.send(vec![1; 1024].into()).await?;
64///     w.send(vec![2; 1024].into()).await?;
65///     w.close().await?;
66///     Ok(())
67/// }
68/// ```
69///
70/// ### Write like `AsyncWrite`
71///
72/// ```
73/// use anyhow::Result;
74/// use futures::AsyncWriteExt;
75/// use opendal_core::Operator;
76///
77/// async fn test(op: Operator) -> Result<()> {
78///     let mut w = op.writer("path/to/file").await?.into_futures_async_write();
79///     w.write(&vec![1; 1024]).await?;
80///     w.write(&vec![2; 1024]).await?;
81///     w.close().await?;
82///     Ok(())
83/// }
84/// ```
85///
86/// ### Write with append enabled
87///
88/// Writer also supports to write with append enabled. This is useful when users want to append
89/// some data to the end of the file.
90///
91/// - If file doesn't exist, it will be created and just like calling `write`.
92/// - If file exists, data will be appended to the end of the file.
93///
94/// Possible Errors:
95///
96/// - Some services store normal file and appendable file in different way. Trying to append
97///   on non-appendable file could return [`ErrorKind::Conflict`] error.
98/// - Services that doesn't support append will return [`ErrorKind::Unsupported`] error when
99///   creating writer with `append` enabled.
100pub struct Writer {
101    /// Keep a reference to write context in writer.
102    ctx: Arc<WriteContext>,
103    inner: WriteGenerator<oio::Writer>,
104}
105
106impl Writer {
107    /// Create a new writer from an `oio::Writer`.
108    pub(crate) fn new(ctx: WriteContext) -> Result<Self> {
109        let ctx = Arc::new(ctx);
110        let inner = WriteGenerator::create(ctx.clone())?;
111
112        Ok(Self { ctx, inner })
113    }

115    /// Write [`Buffer`] into writer.
116    ///
117    /// This operation will write all data in given buffer into writer.
118    ///
119    /// ## Examples
120    ///
121    /// ```
122    /// use bytes::Bytes;
123    /// use opendal_core::Operator;
124    /// use opendal_core::Result;
125    ///
126    /// async fn test(op: Operator) -> Result<()> {
127    ///     let mut w = op.writer("hello.txt").await?;
128    ///     // Buffer can be created from continues bytes.
129    ///     w.write("hello, world").await?;
130    ///     // Buffer can also be created from non-continues bytes.
131    ///     w.write(vec![Bytes::from("hello,"), Bytes::from("world!")])
132    ///         .await?;
133    ///
134    ///     // Make sure file has been written completely.
135    ///     w.close().await?;
136    ///     Ok(())
137    /// }
138    /// ```
139    pub async fn write(&mut self, bs: impl Into<Buffer>) -> Result<()> {
140        let mut bs = bs.into();
141        while !bs.is_empty() {
142            let n = self.inner.write(bs.clone()).await?;
143            bs.advance(n);
144        }
145
146        Ok(())
147    }

149    /// Write [`bytes::Buf`] into inner writer.
150    ///
151    /// This operation will write all data in given buffer into writer.
152    ///
153    /// # Notes
154    ///
155    /// This function iterates over each chunk in the `Buf` and collects them
156    /// into a `Buffer`. For `Bytes` chunks, `copy_to_bytes` is zero-copy
157    /// (only a ref-count increment). For other chunk types (e.g., `&[u8]`),
158    /// the data will be copied.
159    ///
160    /// This approach minimizes copies for chained buffers (`Chain<Bytes, Bytes>`)
161    /// where each individual chunk can be extracted without copying.
162    pub async fn write_from(&mut self, bs: impl Buf) -> Result<()> {
163        let mut bs = bs;
164
165        // Fast path: single contiguous chunk.
166        // If chunk length equals remaining bytes, the entire buffer is contiguous.
167        // This avoids Vec allocation and loop overhead for the common case.
168        if bs.chunk().len() == bs.remaining() {
169            let bytes = bs.copy_to_bytes(bs.remaining());
170            return self.write(Buffer::from(bytes)).await;
171        }
172
173        // Slow path: multiple chunks (e.g., Chain<T, U>).
174        // Iterate over each chunk and collect them into a Vec<Bytes>.
175        // For Bytes chunks, copy_to_bytes is zero-copy.
176        // For other types, only the current chunk is copied, not the entire buffer.
177        let mut chunks = Vec::new();
178        while bs.has_remaining() {
179            let chunk_len = bs.chunk().len();
180            chunks.push(bs.copy_to_bytes(chunk_len));
181        }
182        self.write(Buffer::from(chunks)).await
183    }

185    /// Copy a range of an existing object into this writer.
186    ///
187    /// The source path is resolved by the same [`Operator`] that created this
188    /// writer. Calls to `write` and `copy_from` may be interleaved, and their
189    /// call order defines the destination bytes. OpenDAL uses a native
190    /// writer-side copy when the composed service supports it and otherwise
191    /// streams the range through the client. Check
192    /// [`Capability::write_can_copy_from`] to determine whether eligible
193    /// non-append writes use the native path.
194    ///
195    /// Each call accepts at most 5 GiB. Applications must split larger source
196    /// ranges into multiple ordered calls.
197    pub async fn copy_from(&mut self, path: &str, range: impl Into<BytesRange>) -> Result<()> {
198        self.copy_from_options(
199            path,
200            options::ReadOptions {
201                range: range.into(),
202                ..Default::default()
203            },
204        )
205        .await
206    }
207
208    /// Copy an existing object range with read conditions and execution options.
209    pub async fn copy_from_options(
210        &mut self,
211        path: &str,
212        options: options::ReadOptions,
213    ) -> Result<()> {
214        let path = normalize_path(path);
215        if !validate_path(&path, EntryMode::FILE) {
216            return Err(
217                Error::new(ErrorKind::IsADirectory, "copy source path is a directory")
218                    .with_operation("Writer::copy_from")
219                    .with_context("service", self.ctx.service().info().scheme())
220                    .with_context("path", &path),
221            );
222        }
223        let (range, args, options) = options.into();
224        self.inner.copy_from(&path, args, options, range).await
225    }

227    /// Abort the writer and clean up all written data.
228    ///
229    /// ## Notes
230    ///
231    /// Abort should only be called when the writer is not closed or
232    /// aborted, otherwise an unexpected error could be returned.
233    pub async fn abort(&mut self) -> Result<()> {
234        self.inner.abort().await
235    }
236
237    /// Close the writer and make sure all data have been committed.
238    ///
239    /// ## Notes
240    ///
241    /// Close should only be called when the writer is not closed or
242    /// aborted, otherwise an unexpected error could be returned.
243    pub async fn close(&mut self) -> Result<Metadata> {
244        self.inner.close().await
245    }

247    /// Convert this writer into a [`BufferSink`] that implements
248    /// [`Sink<Buffer>`][futures::Sink].
249    ///
250    /// # Notes
251    ///
252    /// BufferSink is a zero-cost abstraction. The underlying writer
253    /// will reuse the Bytes and won't perform any copy operation over data.
254    ///
255    /// # Examples
256    ///
257    /// ## Basic Usage
258    ///
259    /// ```
260    /// use std::io;
261    ///
262    /// use bytes::Bytes;
263    /// use futures::SinkExt;
264    /// use opendal_core::Buffer;
265    /// use opendal_core::Operator;
266    /// use opendal_core::Result;
267    ///
268    /// async fn test(op: Operator) -> io::Result<()> {
269    ///     let mut s = op.writer("hello.txt").await?.into_sink();
270    ///     let bs = "Hello, World!".as_bytes();
271    ///     s.send(Buffer::from(bs)).await?;
272    ///     s.close().await?;
273    ///
274    ///     Ok(())
275    /// }
276    /// ```
277    ///
278    /// ## Concurrent Write
279    ///
280    /// ```
281    /// use std::io;
282    ///
283    /// use bytes::Bytes;
284    /// use futures::SinkExt;
285    /// use opendal_core::Buffer;
286    /// use opendal_core::Operator;
287    /// use opendal_core::Result;
288    ///
289    /// async fn test(op: Operator) -> io::Result<()> {
290    ///     let mut w = op
291    ///         .writer_with("hello.txt")
292    ///         .concurrent(8)
293    ///         .chunk(256)
294    ///         .await?
295    ///         .into_sink();
296    ///     let bs = "Hello, World!".as_bytes();
297    ///     w.send(Buffer::from(bs)).await?;
298    ///     w.close().await?;
299    ///
300    ///     Ok(())
301    /// }
302    /// ```
303    pub fn into_sink(self) -> BufferSink {
304        BufferSink::new(self.inner)
305    }

307    /// Convert writer into [`FuturesAsyncWriter`] which implements [`futures::AsyncWrite`],
308    ///
309    /// # Notes
310    ///
311    /// FuturesAsyncWriter is not a zero-cost abstraction. The underlying writer
312    /// requires an owned [`Buffer`], which involves an extra copy operation.
313    ///
314    /// FuturesAsyncWriter is required to call `close()` to make sure all
315    /// data have been written to the storage.
316    ///
317    /// # Examples
318    ///
319    /// ## Basic Usage
320    ///
321    /// ```
322    /// use std::io;
323    ///
324    /// use futures::io::AsyncWriteExt;
325    /// use opendal_core::Operator;
326    /// use opendal_core::Result;
327    ///
328    /// async fn test(op: Operator) -> io::Result<()> {
329    ///     let mut w = op.writer("hello.txt").await?.into_futures_async_write();
330    ///     let bs = "Hello, World!".as_bytes();
331    ///     w.write_all(bs).await?;
332    ///     w.close().await?;
333    ///
334    ///     Ok(())
335    /// }
336    /// ```
337    ///
338    /// ## Concurrent Write
339    ///
340    /// ```
341    /// use std::io;
342    ///
343    /// use futures::io::AsyncWriteExt;
344    /// use opendal_core::Operator;
345    /// use opendal_core::Result;
346    ///
347    /// async fn test(op: Operator) -> io::Result<()> {
348    ///     let mut w = op
349    ///         .writer_with("hello.txt")

431    fn gen_random_bytes() -> Vec<u8> {
432        let mut rng = rand::rng();
433        // Generate size between 1B..16MB.
434        let size = rng.random_range(1..16 * 1024 * 1024);
435        let mut content = vec![0; size];
436        rng.fill_bytes(&mut content);
437        content
438    }

440    #[tokio::test]
441    async fn test_writer_write() {
442        let op = Operator::new(services::Memory::default()).unwrap();
443        let path = "test_file";
444
445        let content = gen_random_bytes();
446        let mut writer = op.writer(path).await.unwrap();
447        writer
448            .write(content.clone())
449            .await
450            .expect("write must succeed");
451        writer.close().await.expect("close must succeed");
452
453        let buf = op.read(path).await.expect("read to end mut succeed");
454
455        assert_eq!(buf.to_bytes(), content);
456    }

476    #[tokio::test]
477    async fn test_writer_write_from_chain() {
478        let op = Operator::new(services::Memory::default()).unwrap();
479        let path = "test_file";
480
481        let part1 = Bytes::from(gen_random_bytes());
482        let part2 = Bytes::from(gen_random_bytes());
483
484        let mut chain_same = part1.clone().chain(part2.clone());
485        let chain = part1.chain(part2);
486
487        let mut writer = op.writer(path).await.unwrap();
488        writer.write_from(chain).await.expect("write must succeed");
489        writer.close().await.expect("close must succeed");
490
491        let buf = op.read(path).await.expect("read to end mut succeed");
492
493        assert_eq!(
494            buf.to_bytes(),
495            chain_same.copy_to_bytes(chain_same.remaining())
496        );
497    }
