18use std::io;
19
20use bytes::Bytes;
21use futures::StreamExt;
22
23use crate::*;
24
25/// `StdBytesIterator` adapts a [`crate::blocking::Reader`] into an
26/// [`Iterator`].
27///
28/// Users can use this adapter in cases where they need to use [`Iterator`] trait.
29///
30/// StdIterator also implements [`Send`] and [`Sync`].
31pub struct StdBytesIterator {
32    handle: tokio::runtime::Handle,
33    inner: Option<FuturesBytesStream>,
34}
35
36impl StdBytesIterator {
37    /// NOTE: don't allow users to create StdIterator directly.
38    #[inline]
39    pub(crate) fn new(handle: tokio::runtime::Handle, inner: FuturesBytesStream) -> Self {
40        StdBytesIterator {
41            handle,
42            inner: Some(inner),
43        }
44    }
45}
46
47impl Iterator for StdBytesIterator {
48    type Item = io::Result<Bytes>;
49
50    fn next(&mut self) -> Option<Self::Item> {
51        let Some(inner) = self.inner.as_mut() else {
52            return Some(Err(Error::new(
53                ErrorKind::Unexpected,
54                "reader has been dropped",
55            )
56            .into()));
57        };
58
59        self.handle.block_on(inner.next())
60    }
61}
62
63/// Make sure the inner reader is dropped in async context.
64impl Drop for StdBytesIterator {
65    fn drop(&mut self) {
66        if let Some(v) = self.inner.take() {
67            self.handle.block_on(async move { drop(v) });
68        }
69    }
70}
