18use std::mem::MaybeUninit;
19use std::sync::Arc;
20
21const BITMAP_SIZE: usize = size_of::<u16>();
22const OFFSET_SIZE: usize = size_of::<u16>();
23
24/// Immutable storage for a closed set of optional byte fields.
25#[derive(Clone, Debug, Default, Eq, Hash, PartialEq)]
26pub(crate) struct CompactValues(Option<Arc<[u8]>>);
27
28impl CompactValues {
29    #[inline]
30    pub(crate) fn encode<const N: usize>(fields: &[Option<&[u8]>; N]) -> Self {
31        let lengths = fields.map(|value| value.map(<[u8]>::len));
32        Self::encode_with(&lengths, |index, output| {
33            output.write(fields[index].expect("present field has a value"));
34        })
35    }
36
37    pub(crate) fn encode_with<const N: usize>(
38        lengths: &[Option<usize>; N],
39        write_field: impl FnMut(usize, &mut CompactFieldWriter<'_>),
40    ) -> Self {
41        let layout = encoded_layout(lengths);
42        let total_len = layout.2;
43        if total_len == 0 {
44            return Self(None);
45        }
46
47        let mut encoded = Arc::<[u8]>::new_uninit_slice(total_len);
48        let output = Arc::get_mut(&mut encoded).expect("new Arc has one owner");
49        write_encoded_with_layout(lengths, output, layout, write_field);
50
51        // `write_encoded` initializes every byte in the bitmap, offsets, and payload.
52        let encoded = unsafe { encoded.assume_init() };
53        Self(Some(encoded))
54    }
55
56    #[inline]
57    pub(crate) fn fields<const N: usize>(&self) -> [Option<&[u8]>; N] {
58        assert!(N <= u16::BITS as usize);
59
60        let mut fields = [None; N];
61        let Some(encoded) = self.0.as_deref() else {
62            return fields;
63        };
64        let present = read_present(encoded);
65        let mut value_index = 0;
66        let mut start = BITMAP_SIZE + present.count_ones() as usize * OFFSET_SIZE;
67        for (field, value) in fields.iter_mut().enumerate() {
68            if present & (1_u16 << field) == 0 {
69                continue;
70            }
71            let end = read_offset(encoded, value_index);
72            *value = Some(&encoded[start..end]);
73            start = end;
74            value_index += 1;
75        }
76        fields
77    }
78
79    #[inline]
80    pub(crate) fn get_str(&self, field: usize) -> Option<&str> {
81        self.get(field).map(|value| {
82            // String fields only enter compact storage from `str` or `String`. Blocks decoded
83            // by this module were produced by the same internal encoders.
84            unsafe { std::str::from_utf8_unchecked(value) }
85        })
86    }
87
88    pub(crate) fn replace(&self, field: usize, value: &[u8]) -> Self {
89        assert!(field < u16::BITS as usize);
90        let mut fields = self.fields::<{ u16::BITS as usize }>();
91        fields[field] = Some(value);
92        Self::encode(&fields)
93    }
94
95    #[inline]
96    pub(crate) fn get(&self, field: usize) -> Option<&[u8]> {
97        assert!(field < u16::BITS as usize);
98        let encoded = self.0.as_deref()?;
99        if encoded.is_empty() {
100            return None;
101        }
102        let present = read_present(encoded);
103        let bit = 1_u16 << field;
104        if present & bit == 0 {
105            return None;
106        }
107
108        let value_count = present.count_ones() as usize;
109        let header_len = BITMAP_SIZE + value_count * OFFSET_SIZE;
110        let value_index = (present & (bit - 1)).count_ones() as usize;
111        let start = if value_index == 0 {
112            header_len
113        } else {
114            read_offset(encoded, value_index - 1)
115        };
116        let end = read_offset(encoded, value_index);
117        Some(&encoded[start..end])
118    }
119
120    #[cfg(test)]
121    #[inline]
122    pub(crate) fn as_encoded(&self) -> Option<&[u8]> {
123        self.0.as_deref()
124    }
125}
126
127fn encoded_layout(lengths: &[Option<usize>]) -> (u16, usize, usize) {
128    assert!(lengths.len() <= u16::BITS as usize);
129
130    let mut present = 0_u16;
131    let mut payload_len = 0_usize;
132    for (index, value) in lengths.iter().enumerate() {
133        if let Some(value) = value {
134            present |= 1 << index;
135            payload_len = payload_len
136                .checked_add(*value)
137                .expect("compact value block length overflowed");
138        }
139    }
140
141    if present == 0 {
142        return (0, 0, 0);
143    }
144
145    let value_count = present.count_ones() as usize;
146    let header_len = BITMAP_SIZE + value_count * OFFSET_SIZE;
147    let total_len = header_len
148        .checked_add(payload_len)
149        .expect("compact value block length overflowed");
150    assert!(
151        total_len <= u16::MAX as usize,
152        "compact value block exceeds 64 KiB"
153    );
154
155    (present, header_len, total_len)
156}
157
158fn write_encoded_with_layout(
159    lengths: &[Option<usize>],
160    output: &mut [MaybeUninit<u8>],
161    (present, header_len, total_len): (u16, usize, usize),
162    mut write_field: impl FnMut(usize, &mut CompactFieldWriter<'_>),
163) {
164    assert_eq!(output.len(), total_len);
165    if total_len == 0 {
166        return;
167    }
168    write_slice(output, 0, &present.to_le_bytes());
169
170    let mut payload_offset = header_len;
171    let mut value_index = 0;
172    for (field, value_len) in lengths.iter().enumerate() {
173        if let Some(value_len) = value_len {
174            let mut writer =
175                CompactFieldWriter::new(&mut output[payload_offset..payload_offset + value_len]);
176            write_field(field, &mut writer);
177            writer.finish();
178            payload_offset += value_len;
179            let end = u16::try_from(payload_offset).expect("value block length was validated");
180            let offset = BITMAP_SIZE + value_index * OFFSET_SIZE;
181            write_slice(output, offset, &end.to_le_bytes());
182            value_index += 1;
183        }
184    }
185
186    debug_assert_eq!(payload_offset, total_len);
187}
188
189fn write_slice(output: &mut [MaybeUninit<u8>], offset: usize, value: &[u8]) {
190    initialize_bytes(&mut output[offset..offset + value.len()], value);
191}
192
193fn initialize_bytes(output: &mut [MaybeUninit<u8>], value: &[u8]) {
194    assert_eq!(output.len(), value.len());
195    for (output, value) in output.iter_mut().zip(value) {
196        output.write(*value);
197    }
198}

200pub(crate) struct CompactFieldWriter<'a> {
201    output: &'a mut [MaybeUninit<u8>],
202    initialized: usize,
203}
204
205impl<'a> CompactFieldWriter<'a> {
206    fn new(output: &'a mut [MaybeUninit<u8>]) -> Self {
207        Self {
208            output,
209            initialized: 0,
210        }
211    }
212
213    pub(crate) fn write(&mut self, value: &[u8]) {
214        let end = self
215            .initialized
216            .checked_add(value.len())
217            .expect("compact field length overflowed");
218        assert!(
219            end <= self.output.len(),
220            "compact field wrote too many bytes"
221        );
222        initialize_bytes(&mut self.output[self.initialized..end], value);
223        self.initialized = end;
224    }
225
226    fn finish(self) {
227        assert_eq!(
228            self.initialized,
229            self.output.len(),
230            "compact field did not initialize its declared length"
231        );
232    }
233}
234
235#[inline]
236fn read_present(encoded: &[u8]) -> u16 {
237    u16::from_le_bytes(
238        encoded[..BITMAP_SIZE]
239            .try_into()
240            .expect("compact value block contains a bitmap"),
241    )
242}
243
244#[inline]
245fn read_offset(encoded: &[u8], index: usize) -> usize {
246    let offset = BITMAP_SIZE + index * OFFSET_SIZE;
247    u16::from_le_bytes(
248        encoded[offset..offset + OFFSET_SIZE]
249            .try_into()
250            .expect("compact value block contains a complete offset"),
251    ) as usize
252}

254#[cfg(test)]
255mod tests {
256    use super::*;
257
258    #[test]
259    fn compact_values_roundtrip() {
260        let values = CompactValues::encode(&[Some(b"first"), None, Some(b"third"), Some(b"")]);
261
262        assert_eq!(values.get(0), Some(b"first".as_slice()));
263        assert_eq!(values.get(1), None);
264        assert_eq!(values.get(2), Some(b"third".as_slice()));
265        assert_eq!(values.get(3), Some(b"".as_slice()));
266    }
267
268    #[test]
269    fn compact_values_empty_is_default() {
270        let values = CompactValues::encode(&[None, None]);
271        assert_eq!(values, CompactValues::default());
272    }
273
274    #[test]
275    fn compact_values_replace_present_field() {
276        let values = CompactValues::encode(&[Some(b"first"), None, Some(b"third")]);
277        let values = values.replace(2, b"changed");
278
279        assert_eq!(
280            values.fields::<3>(),
281            [Some(b"first".as_slice()), None, Some(b"changed".as_slice())]
282        );
283    }
284
285    #[test]
286    fn compact_values_insert_absent_field() {
287        let values = CompactValues::encode(&[Some(b"first"), None, Some(b"third")]);
288        let values = values.replace(1, b"second");
289
290        assert_eq!(
291            values.fields::<3>(),
292            [
293                Some(b"first".as_slice()),
294                Some(b"second".as_slice()),
295                Some(b"third".as_slice()),
296            ]
297        );
298    }
299
300    #[test]
301    fn compact_values_replace_preserves_high_fields() {
302        let values = CompactValues::encode(&[Some(b"first"), None, None, Some(b"fourth")]);
303        let values = values.replace(0, b"changed");
304
305        assert_eq!(values.get(0), Some(b"changed".as_slice()));
306        assert_eq!(values.get(3), Some(b"fourth".as_slice()));
307    }
308
309    #[test]
310    fn compact_values_accept_maximum_block() {
311        let value = vec![0; u16::MAX as usize - BITMAP_SIZE - OFFSET_SIZE];
312        let values = CompactValues::encode(&[Some(&value)]);
313        assert_eq!(values.as_encoded().unwrap().len(), u16::MAX as usize);
314        assert_eq!(values.get(0), Some(value.as_slice()));
315    }
316
317    #[test]
318    #[should_panic(expected = "compact value block exceeds 64 KiB")]
319    fn compact_values_reject_oversized_block() {
320        let value = vec![0; u16::MAX as usize - BITMAP_SIZE - OFFSET_SIZE + 1];
321        CompactValues::encode(&[Some(&value)]);
322    }
323
324    #[test]
325    #[should_panic(expected = "compact field did not initialize its declared length")]
326    fn compact_values_reject_incomplete_encoder() {
327        CompactValues::encode_with(&[Some(1)], |_, _| {});
328    }
