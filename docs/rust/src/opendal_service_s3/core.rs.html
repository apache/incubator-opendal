163fn format_crc32c_iter(body: Buffer) -> String {
164    let mut digest = CrcDigest::new(CrcAlgorithm::Crc32Iscsi);
165    body.for_each(|b| digest.update(&b));
166
167    let crc = digest.finalize() as u32;
168    BASE64_STANDARD.encode(crc.to_be_bytes())
169}

182    fn sign_error(&self, err: reqsign_core::Error) -> Error {
183        if !self.signers.is_express() {
184            return new_request_sign_error(err.into());
185        }
186
187        let kind = match err.kind() {
188            ReqsignErrorKind::CredentialInvalid => ErrorKind::Unexpected,
189            ReqsignErrorKind::ConfigInvalid => ErrorKind::ConfigInvalid,
190            ReqsignErrorKind::PermissionDenied => ErrorKind::PermissionDenied,
191            ReqsignErrorKind::RateLimited => ErrorKind::RateLimited,
192            ReqsignErrorKind::RequestInvalid | ReqsignErrorKind::Unexpected => {
193                ErrorKind::Unexpected
194            }
195        };
196        let retryable = err.is_retryable();
197        Error::new(kind, "failed to sign S3 Express request")
198            .with_operation("S3Core::sign")
199            .set_source(err)
200            .with_temporary(retryable)
201    }

203    pub async fn sign_query<T>(
204        &self,
205        ctx: &OperationContext,
206        req: Request<T>,
207        duration: Duration,
208    ) -> Result<Request<T>> {
209        if self.skip_signature {
210            return Ok(req);
211        }
212
213        // Sign the request with presigned URL
214        let (mut parts, body) = req.into_parts();
215
216        self.signers
217            .iam()
218            .clone()
219            .with_context(
220                Context::new()
221                    .with_file_read(reqsign_file_read_tokio::TokioFileRead)
222                    .with_http_send(ctx.http_transport().clone())
223                    .with_env(reqsign_core::OsEnv),
224            )
225            .sign(&mut parts, Some(duration))
226            .await
227            .map_err(|err| self.sign_error(err))?;
228
229        // Always remove host header, let users' client to set it based on HTTP
230        // version.
231        //
232        // As discussed in <https://github.com/seanmonstar/reqwest/issues/1809>,
233        // google server could send RST_STREAM of PROTOCOL_ERROR if our request
234        // contains host header.
235        parts.headers.remove(HOST);
236
237        Ok(Request::from_parts(parts, body))
238    }

314    /// Adds SSE-C headers and, for object creation, SSE-S3/SSE-KMS configuration.
315    ///
316    /// Multipart uploads choose SSE-S3/SSE-KMS encryption at initiation.
317    pub fn insert_sse_headers(
318        &self,
319        mut req: http::request::Builder,
320        configure_destination_encryption: bool,
321    ) -> http::request::Builder {
322        if configure_destination_encryption {
323            if let Some(v) = &self.server_side_encryption {
324                let mut v = v.clone();
325                v.set_sensitive(true);
326
327                req = req.header(
328                    HeaderName::from_static(constants::X_AMZ_SERVER_SIDE_ENCRYPTION),
329                    v,
330                )
331            }
332            if let Some(v) = &self.server_side_encryption_aws_kms_key_id {
333                let mut v = v.clone();
334                v.set_sensitive(true);
335
336                req = req.header(
337                    HeaderName::from_static(constants::X_AMZ_SERVER_SIDE_ENCRYPTION_AWS_KMS_KEY_ID),
338                    v,
339                )
340            }
341        }
342
343        if let Some(v) = &self.server_side_encryption_customer_algorithm {
344            let mut v = v.clone();
345            v.set_sensitive(true);
346
347            req = req.header(
348                HeaderName::from_static(constants::X_AMZ_SERVER_SIDE_ENCRYPTION_CUSTOMER_ALGORITHM),
349                v,
350            )
351        }
352        if let Some(v) = &self.server_side_encryption_customer_key {
353            let mut v = v.clone();
354            v.set_sensitive(true);
355
356            req = req.header(
357                HeaderName::from_static(constants::X_AMZ_SERVER_SIDE_ENCRYPTION_CUSTOMER_KEY),
358                v,
359            )
360        }
361        if let Some(v) = &self.server_side_encryption_customer_key_md5 {
362            let mut v = v.clone();
363            v.set_sensitive(true);
364
365            req = req.header(
366                HeaderName::from_static(constants::X_AMZ_SERVER_SIDE_ENCRYPTION_CUSTOMER_KEY_MD5),
367                v,
368            )
369        }
370
371        req
372    }

401    pub fn insert_metadata_headers(
402        &self,
403        mut req: http::request::Builder,
404        size: Option<u64>,
405        args: &OpWrite,
406    ) -> http::request::Builder {
407        if let Some(size) = size {
408            req = req.header(CONTENT_LENGTH, size.to_string())
409        }
410
411        if let Some(mime) = args.content_type() {
412            req = req.header(CONTENT_TYPE, mime)
413        }
414
415        if let Some(pos) = args.content_disposition() {
416            req = req.header(CONTENT_DISPOSITION, pos)
417        }
418
419        if let Some(encoding) = args.content_encoding() {
420            req = req.header(CONTENT_ENCODING, encoding);
421        }
422
423        if let Some(cache_control) = args.cache_control() {
424            req = req.header(CACHE_CONTROL, cache_control)
425        }
426
427        if let Some(if_match) = args.if_match() {
428            req = req.header(IF_MATCH, if_match);
429        }
430
431        if args.if_not_exists() {
432            req = req.header(IF_NONE_MATCH, "*");
433        }
434
435        // Set storage class header
436        if let Some(v) = &self.default_storage_class {
437            req = req.header(HeaderName::from_static(constants::X_AMZ_STORAGE_CLASS), v);
438        }
439
440        // Set user metadata headers.
441        if let Some(user_metadata) = args.user_metadata() {
442            for (key, value) in user_metadata {
443                req = req.header(format!("{X_AMZ_META_PREFIX}{key}"), value)
444            }
445        }
446
447        // Set ACL header.
448        if let Some(acl) = &self.default_acl {
449            req = req.header(constants::X_AMZ_ACL, acl);
450        }
451        req
452    }

469    pub fn s3_head_object_request(&self, path: &str, args: OpStat) -> Result<Request<Buffer>> {
470        let p = build_abs_path(&self.root, path);
471
472        let mut url = format!("{}/{}", self.endpoint, percent_encode_path(&p));
473
474        // Add query arguments to the URL based on response overrides
475        let mut query_args = Vec::new();
476        if let Some(override_content_disposition) = args.override_content_disposition() {
477            query_args.push(format!(
478                "{}={}",
479                constants::RESPONSE_CONTENT_DISPOSITION,
480                percent_encode_path(override_content_disposition)
481            ))
482        }
483        if let Some(override_content_type) = args.override_content_type() {
484            query_args.push(format!(
485                "{}={}",
486                constants::RESPONSE_CONTENT_TYPE,
487                percent_encode_path(override_content_type)
488            ))
489        }
490        if let Some(override_cache_control) = args.override_cache_control() {
491            query_args.push(format!(
492                "{}={}",
493                constants::RESPONSE_CACHE_CONTROL,
494                percent_encode_path(override_cache_control)
495            ))
496        }
497        if let Some(version) = args.version() {
498            query_args.push(format!(
499                "{}={}",
500                constants::S3_QUERY_VERSION_ID,
501                percent_encode_path(version)
502            ))
503        }
504        if !query_args.is_empty() {
505            url.push_str(&format!("?{}", query_args.join("&")));
506        }
507
508        let mut req = Request::head(&url);
509
510        req = self.insert_sse_headers(req, false);
511
512        if let Some(if_none_match) = args.if_none_match() {
513            req = req.header(IF_NONE_MATCH, if_none_match);
514        }
515        if let Some(if_match) = args.if_match() {
516            req = req.header(IF_MATCH, if_match);
517        }
518
519        if let Some(if_modified_since) = args.if_modified_since() {
520            req = req.header(IF_MODIFIED_SINCE, if_modified_since.format_http_date());
521        }
522        if let Some(if_unmodified_since) = args.if_unmodified_since() {
523            req = req.header(IF_UNMODIFIED_SINCE, if_unmodified_since.format_http_date());
524        }
525
526        // Set request payer header if enabled.
527        req = self.insert_request_payer_header(req);
528
529        // Inject operation to the request.
530        req = req
531            .extension(Operation::Stat)
532            .extension(ServiceOperation("HeadObject"));
533
534        let req = req.body(Buffer::new()).map_err(new_request_build_error)?;
535
536        Ok(req)
537    }

542        range: BytesRange,
543        args: &OpRead,

545        let p = build_abs_path(&self.root, path);
546
547        // Construct headers to add to the request
548        let mut url = format!("{}/{}", self.endpoint, percent_encode_path(&p));
549
550        // Add query arguments to the URL based on response overrides
551        let mut query_args = Vec::new();
552        if let Some(override_content_disposition) = args.override_content_disposition() {
553            query_args.push(format!(
554                "{}={}",
555                constants::RESPONSE_CONTENT_DISPOSITION,
556                percent_encode_path(override_content_disposition)
557            ))
558        }
559        if let Some(override_content_type) = args.override_content_type() {
560            query_args.push(format!(
561                "{}={}",
562                constants::RESPONSE_CONTENT_TYPE,
563                percent_encode_path(override_content_type)
564            ))
565        }
566        if let Some(override_cache_control) = args.override_cache_control() {
567            query_args.push(format!(
568                "{}={}",
569                constants::RESPONSE_CACHE_CONTROL,
570                percent_encode_path(override_cache_control)
571            ))
572        }
573        if let Some(version) = args.version() {
574            query_args.push(format!(
575                "{}={}",
576                constants::S3_QUERY_VERSION_ID,

578            ))
579        }

583
584        let mut req = Request::get(&url);
585

588        }
589
590        if let Some(if_none_match) = args.if_none_match() {
591            req = req.header(IF_NONE_MATCH, if_none_match);
592        }

594        if let Some(if_match) = args.if_match() {
595            req = req.header(IF_MATCH, if_match);
596        }
597
598        if let Some(if_modified_since) = args.if_modified_since() {
599            req = req.header(IF_MODIFIED_SINCE, if_modified_since.format_http_date());
600        }
601
602        if let Some(if_unmodified_since) = args.if_unmodified_since() {
603            req = req.header(IF_UNMODIFIED_SINCE, if_unmodified_since.format_http_date());
604        }
605
606        // Set request payer header if enabled.
607        req = self.insert_request_payer_header(req);
608
609        // Set SSE headers.
610        // TODO: how will this work with presign?
611        req = self.insert_sse_headers(req, false);
612
613        // Inject operation to the request.
614        req = req
615            .extension(Operation::Read)
616            .extension(ServiceOperation("GetObject"));
617
618        let req = req.body(Buffer::new()).map_err(new_request_build_error)?;
619
620        Ok(req)
621    }

2210/// Parse error response into Error.
2211pub fn parse_error(ctx: ErrorContext, resp: Response<Buffer>) -> Error {
2212    let (parts, body) = resp.into_parts();
2213    let bs = body.to_bytes();
2214
2215    let (mut kind, mut retryable) = match parts.status.as_u16() {
2216        403 => (ErrorKind::PermissionDenied, false),
2217        404 => (ErrorKind::NotFound, false),
2218        304 | 412 if ctx.caller_condition => (ErrorKind::ConditionNotMatch, false),
2219        // Service like R2 could return 499 error with a message like:
2220        // Client Disconnect, we should retry it.
2221        499 => (ErrorKind::Unexpected, true),
2222        500 | 502 | 503 | 504 => (ErrorKind::Unexpected, true),
2223        429 => (ErrorKind::RateLimited, true),
2224        _ => (ErrorKind::Unexpected, false),
2225    };
2226
2227    let body_content = bs.chunk();
2228    let (message, s3_err) = de::from_reader::<_, S3Error>(body_content.reader())
2229        .map(|s3_err| (format!("{s3_err:?}"), Some(s3_err)))
2230        .unwrap_or_else(|_| (String::from_utf8_lossy(&bs).into_owned(), None));
2231
2232    if let Some(s3_err) = s3_err
2233        && !s3_err.code.is_empty()
2234    {
2235        if let Some(classification) = parse_s3_error_code(ctx, s3_err.code.as_str()) {
2236            (kind, retryable) = classification;
2237        } else if matches!(parts.status.as_u16(), 409 | 412) {
2238            (kind, retryable) = (ErrorKind::Unexpected, false);
2239        }
2240    }
2241
2242    let mut err =
2243        Error::new(kind, message).with_context("service_operation", ctx.service_operation.0);
2244
2245    err = with_error_response_context(err, parts);
2246
2247    if retryable {
2248        err = err.set_temporary();
2249    }
2250
2251    err
2252}

2254/// Returns the `Error kind` of this code and whether the error is retryable.
2255/// All possible error code: <https://docs.aws.amazon.com/AmazonS3/latest/API/ErrorResponses.html#ErrorCodeList>
2256pub fn parse_s3_error_code(ctx: ErrorContext, code: &str) -> Option<(ErrorKind, bool)> {
2257    match code {
2258        // > The specified bucket does not exist.
2259        //
2260        // Although the status code is 404, NoSuchBucket is
2261        // a config invalid error, and it's not retryable from OpenDAL.
2262        "NoSuchBucket" => Some((ErrorKind::ConfigInvalid, false)),
2263        // S3 Express can return 200 OK with an embedded NoSuchKey error for
2264        // CopyObject. Classify the error code explicitly because the HTTP
2265        // status alone cannot preserve the NotFound semantics in this case.
2266        "NoSuchKey" => Some((ErrorKind::NotFound, false)),
2267        // > Your socket connection to the server was not read from
2268        // > or written to within the timeout period."
2269        //
2270        // It's Ok for us to retry it again.
2271        "RequestTimeout" => Some((ErrorKind::Unexpected, true)),
2272        // > An internal error occurred. Try again.
2273        "InternalError" => Some((ErrorKind::Unexpected, true)),
2274        // > A conflicting conditional operation is currently in progress
2275        // > against this resource. Try again.
2276        "OperationAborted" => Some((ErrorKind::Conflict, true)),
2277        // > Please reduce your request rate.
2278        //
2279        // It's Ok to retry since later on the request rate may get reduced.
2280        "SlowDown" => Some((ErrorKind::RateLimited, true)),
2281        // > Service is unable to handle request.
2282        //
2283        // ServiceUnavailable is considered a retryable error because it typically
2284        // indicates a temporary issue with the service or server, such as high load,
2285        // maintenance, or an internal problem.
2286        "ServiceUnavailable" => Some((ErrorKind::Unexpected, true)),
2287        // > Too Many Requests - rate limit exceeded.
2288        //
2289        // It's Ok to retry since later on the request rate may get reduced.
2290        "TooManyRequests" => Some((ErrorKind::RateLimited, true)),
2291        // > Compatibility with Volcengine TOS
2292        //
2293        // TOS returns following error codes along with 429 status code, while both
2294        // of them indicate rate limit exceeded.
2295        // See https://www.volcengine.com/docs/6349/74874 for more details.
2296        "ExceedAccountQPSLimit"
2297        | "ExceedAccountRateLimit"
2298        | "ExceedBucketQPSLimit"
2299        | "ExceedBucketRateLimit" => Some((ErrorKind::RateLimited, true)),
2300        "InvalidRange" => Some((ErrorKind::RangeNotSatisfied, false)),
2301        "PreconditionFailed" | "412 Precondition Failed" if ctx.caller_condition => {
2302            Some((ErrorKind::ConditionNotMatch, false))
2303        }
2304        "PreconditionFailed" | "412 Precondition Failed" if ctx.source_if_match => {
2305            Some((ErrorKind::Conflict, false))
2306        }
2307        "ConditionalRequestConflict" => Some((
2308            ErrorKind::Conflict,
2309            ctx.service_operation == ServiceOperation("PutObject"),
2310        )),
2311        _ => None,
2312    }
2313}
