Full reference
import { Operator } from "opendal";

const operator = new Operator("s3", {
  // --- General ---
  // root of this backend.
  //
  // All operations will happen under this root.
  //
  // default to `/` if not set.
  // root: "/",

  // bucket name of this backend.
  //
  // required.
  bucket: "my-bucket",

  // endpoint of this backend.
  //
  // Endpoint must be full uri, e.g.
  //
  // - AWS S3: `https://s3.amazonaws.com` or `https://s3.{region}.amazonaws.com`
  // - Cloudflare R2: `https://<ACCOUNT_ID>.r2.cloudflarestorage.com`
  // - Aliyun OSS: `https://{region}.aliyuncs.com`
  // - Tencent COS: `https://cos.{region}.myqcloud.com`
  // - Minio: `http://127.0.0.1:9000`
  //
  // If user inputs endpoint without scheme like "s3.amazonaws.com", we
  // will prepend "https://" before it.
  //
  // - If endpoint is set, we will take user's input first.
  // - If not, we will try to load it from environment.
  // - If still not set, default to `https://s3.amazonaws.com`.
  // endpoint: "https://s3.amazonaws.com",

  // Region represent the signing region of this endpoint. This is required
  // if you are using the default AWS S3 endpoint.
  //
  // If using a custom endpoint,
  // - If region is set, we will take user's input first.
  // - If not, we will try to load it from environment.
  // region: "us-east-1",

  // Fallback signing region when neither `region`, `AWS_REGION`, nor
  // `AWS_DEFAULT_REGION` supplies one.
  // default_region: "...",
  // --- Credentials ---
  // AWS profile.
  //
  // By default, reqsign which is the default credential provider, supplies profile in order:
  // - explicit option
  // - `AWS_PROFILE` environment variable, from which reqsign reads profile from:
  // - `~/.aws/credentials` (or the path specified by `AWS_SHARED_CREDENTIALS_FILE`)
  // - `~/.aws/config` (or the path specified by `AWS_CONFIG_FILE`)
  // profile: "development",

  // access_key_id of this backend.
  //
  // - If access_key_id is set, we will take user's input first.
  // - If not, we will try to load it from environment.
  // access_key_id: "...",

  // secret_access_key of this backend.
  //
  // - If secret_access_key is set, we will take user's input first.
  // - If not, we will try to load it from environment.
  // secret_access_key: "...",

  // session_token (aka, security token) of this backend.
  //
  // This token will expire after sometime, it's recommended to set session_token
  // by hand.
  // session_token: "...",

  // Relative URI for AWS container credential metadata.
  //
  // An explicit full endpoint takes precedence over this value. This value
  // takes precedence over the ambient full and relative container credential
  // URIs. OpenDAL joins it to `container_metadata_uri_override` when set,
  // otherwise to `http://169.254.170.2`.
  //
  // Setting any container credential field replaces the ECS slot in the
  // default credential chain. It does not move that slot ahead of earlier
  // default providers.
  // container_credentials_relative_uri: "...",

  // Full URI for AWS container credential metadata.
  //
  // This endpoint takes precedence over the relative container credential
  // URI and both ambient container credential URI forms. It does not change
  // the position or priority of the default ECS credential provider.
  // container_credentials_endpoint: "...",

  // Authorization token sent to the AWS container credential endpoint.
  //
  // This token takes precedence over the configured token file and the
  // ambient direct and file token forms.
  // container_authorization_token: "...",

  // File containing the authorization token sent to the AWS container
  // credential endpoint.
  //
  // This file takes precedence over both ambient token forms. The credential
  // provider trims surrounding whitespace after reading the file.
  // container_authorization_token_file: "...",

  // Base URI used to resolve a relative AWS container credential URI.
  //
  // This override affects relative URIs only. It does not alter a configured
  // or ambient full URI. The override alone does not select an endpoint.
  // container_metadata_uri_override: "...",

  // Disable automatic endpoint loading and the environment/profile
  // credential providers.
  //
  // This skips endpoint environment variables and removes the static
  // environment and shared-profile credential slots. Region environment
  // lookup and other credential providers, including AWS container
  // credentials and EC2 instance metadata, remain enabled.
  // disable_config_load: "true",

  // Disable load credential from ec2 metadata.
  //
  // This option is used to disable the default behavior of opendal
  // to load credential from ec2 metadata, a.k.a., IMDSv2
  // disable_ec2_metadata: "true",

  // Skip signature will skip loading credentials and signing requests.
  // skip_signature: "true",
  // --- Assume role ---
  // role_arn for this backend.
  //
  // If `role_arn` is set, we will use already known config as source
  // credential to assume role with `role_arn`.
  // role_arn: "...",

  // external_id for this backend.
  // external_id: "...",

  // role_session_name for this backend.
  // role_session_name: "...",

  // assume_role_duration_seconds for this backend.
  // assume_role_duration_seconds: "1000",

  // assume_role_session_tags for this backend.
  // assume_role_session_tags: "...",
  // --- Encryption ---
  // server_side_encryption for this backend.
  //
  // Available values: `AES256`, `aws:kms`.
  // server_side_encryption: "...",

  // server_side_encryption_aws_kms_key_id for this backend
  //
  // - If `server_side_encryption` set to `aws:kms`, and `server_side_encryption_aws_kms_key_id`
  // is not set, S3 will use aws managed kms key to encrypt data.
  // - If `server_side_encryption` set to `aws:kms`, and `server_side_encryption_aws_kms_key_id`
  // is a valid kms key id, S3 will use the provided kms key to encrypt data.
  // - If the `server_side_encryption_aws_kms_key_id` is invalid or not found, an error will be
  // returned.
  // - If `server_side_encryption` is not `aws:kms`, setting `server_side_encryption_aws_kms_key_id`
  // is a noop.
  // server_side_encryption_aws_kms_key_id: "...",

  // Control S3 Bucket Keys for SSE-KMS writes and copies.
  //
  // Omit this option to inherit the bucket setting. `false` disables Bucket
  // Keys for destination objects, and `true` enables them.
  // server_side_encryption_bucket_key_enabled: "true",

  // server_side_encryption_customer_algorithm for this backend.
  //
  // Available values: `AES256`.
  // server_side_encryption_customer_algorithm: "...",

  // server_side_encryption_customer_key for this backend.
  //
  // Value: BASE64-encoded key that matches algorithm specified in
  // `server_side_encryption_customer_algorithm`.
  // server_side_encryption_customer_key: "...",

  // Set server_side_encryption_customer_key_md5 for this backend.
  //
  // Value: MD5 digest of key specified in `server_side_encryption_customer_key`.
  // server_side_encryption_customer_key_md5: "...",
  // --- Behavior ---
  // default storage_class for this backend.
  //
  // Available values:
  // - `DEEP_ARCHIVE`
  // - `GLACIER`
  // - `GLACIER_IR`
  // - `INTELLIGENT_TIERING`
  // - `ONEZONE_IA`
  // - `EXPRESS_ONEZONE`
  // - `OUTPOSTS`
  // - `REDUCED_REDUNDANCY`
  // - `STANDARD`
  // - `STANDARD_IA`
  //
  // S3 compatible services don't support all of them
  // default_storage_class: "...",

  // Enable virtual host style so that opendal will send API requests
  // in virtual host style instead of path style.
  //
  // - By default, opendal will send API to `https://s3.us-east-1.amazonaws.com/bucket_name`
  // - Enabled, opendal will send API to `https://bucket_name.s3.us-east-1.amazonaws.com`
  // enable_virtual_host_style: "true",

  // Checksum Algorithm to use when sending checksums in HTTP headers.
  // This is necessary when writing to AWS S3 Buckets with Object Lock enabled for example.
  //
  // Available options:
  // - "crc32c"
  // - "md5"
  // checksum_algorithm: "...",

  // OpenDAL uses List Objects V2 by default to list objects.
  // However, some legacy services do not yet support V2.
  // This option allows users to switch back to the older List Objects V1.
  // disable_list_objects_v2: "true",

  // Indicates whether the client agrees to pay for the requests made to the S3 bucket.
  // enable_request_payer: "true",

  // Default ACL for new objects.
  // Note that some s3 services like minio do not support this option.
  // default_acl: "...",
});

Every option is passed as a string key; OpenDAL parses it into the right type. Some services may require building the binding with the matching services-* feature enabled.
