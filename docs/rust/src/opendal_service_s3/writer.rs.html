31pub type S3Writers = TwoWays<oio::MultipartWriter<S3Writer>, oio::AppendWriter<S3Writer>>;

33pub struct S3Writer {
34    core: Arc<S3Core>,
35    ctx: OperationContext,
36
37    op: OpWrite,
38    path: String,
39}
40
41impl S3Writer {
42    pub fn new(core: Arc<S3Core>, ctx: OperationContext, path: &str, op: OpWrite) -> Self {
43        S3Writer {
44            core,
45            ctx,
46            path: path.to_string(),
47            op,
48        }
49    }
50
51    fn parse_header_into_meta(path: &str, headers: &http::HeaderMap) -> Result<Metadata> {
52        let mut meta = if path.ends_with('/') {
53            MetadataBuilder::dir()
54        } else {
55            MetadataBuilder::unknown()
56        };
57        if let Some(etag) = parse_etag(headers)? {
58            meta.etag(etag);
59        }
60        if let Some(version) = parse_header_to_str(headers, X_AMZ_VERSION_ID)? {
61            meta.version(version);
62        }
63        if !path.ends_with('/')
64            && let Some(value) =
65                parse_header_to_str(headers, X_AMZ_OBJECT_SIZE)?.and_then(|size| size.parse().ok())
66        {
67            meta.set_file(value);
68        }
69        Ok(meta.build())
70    }
71
72    fn error_context(&self, service_operation: ServiceOperation) -> ErrorContext {
73        ErrorContext::new(service_operation).with_caller_condition(self.op.is_conditional())
74    }
75}

77impl oio::MultipartWrite for S3Writer {
78    async fn write_once(&self, size: u64, body: Buffer) -> Result<Metadata> {
79        let req = self
80            .core
81            .s3_put_object_request(&self.path, Some(size), &self.op, body)?;
82
83        let resp = self
84            .core
85            .send(&self.ctx, req, self.core.signers.default())
86            .await?;
87
88        let status = resp.status();
89
90        let meta = S3Writer::parse_header_into_meta(&self.path, resp.headers())?;
91
92        match status {
93            StatusCode::CREATED | StatusCode::OK => Ok(meta),
94            _ => {
95                let err = parse_error(self.error_context(ServiceOperation("PutObject")), resp);
96                if self.op.if_match().is_some() && err.kind() == ErrorKind::NotFound {
97                    Err(Error::new(
98                        ErrorKind::ConditionNotMatch,
99                        "write precondition requires a live target",
100                    ))
101                } else {
102                    Err(err)
103                }
104            }
105        }
106    }
107
108    async fn initiate_part(&self) -> Result<String> {
109        let resp = self
110            .core
111            .s3_initiate_multipart_upload(&self.ctx, &self.path, &self.op)
112            .await?;
113
114        let status = resp.status();
115
116        match status {
117            StatusCode::OK => {
118                let bs = resp.into_body();
119
120                let result: InitiateMultipartUploadResult =
121                    quick_xml::de::from_reader(bs.reader()).map_err(new_xml_deserialize_error)?;
122
123                Ok(result.upload_id)
124            }
125            _ => Err(parse_error(
126                ErrorContext::new(ServiceOperation("CreateMultipartUpload")),
127                resp,
128            )),
129        }
130    }
131
132    async fn write_part(
133        &self,
134        upload_id: &str,
135        part_number: usize,
136        size: u64,
137        body: Buffer,
138    ) -> Result<oio::MultipartPart> {
139        // AWS S3 requires part number must between [1..=10000]
140        let part_number = part_number + 1;
141
142        let checksum = self.core.calculate_checksum(&body);
143
144        let req = self.core.s3_upload_part_request(
145            &self.path,
146            upload_id,
147            part_number,
148            size,
149            body,
150            checksum.clone(),
151        )?;
152
153        let resp = self
154            .core
155            .send(&self.ctx, req, self.core.signers.default())
156            .await?;
157
158        let status = resp.status();
159
160        match status {
161            StatusCode::OK => {
162                let etag = parse_etag(resp.headers())?
163                    .ok_or_else(|| {
164                        Error::new(
165                            ErrorKind::Unexpected,
166                            "ETag not present in returning response",
167                        )
168                    })?
169                    .to_string();
170
171                Ok(oio::MultipartPart {
172                    part_number,
173                    etag,
174                    checksum,
175                    size: None,
176                })
177            }
178            _ => Err(parse_error(
179                ErrorContext::new(ServiceOperation("UploadPart")),
180                resp,
181            )),
182        }
183    }
184
185    async fn copy_part(
186        &self,
187        upload_id: &str,
188        part_number: usize,
189        path: &str,
190        args: OpRead,
191        range: BytesRange,
192    ) -> Result<oio::MultipartPart> {
193        let size = range
194            .size()
195            .expect("multipart writer copy range must be bounded");
196        let part_number = part_number + 1;
197        let error_context = ErrorContext::new(ServiceOperation("UploadPartCopy"))
198            .with_caller_condition(args.is_conditional());
199        let req = self
200            .core
201            .s3_upload_part_copy_request(S3UploadPartCopyRequest {
202                from: path,
203                to: &self.path,
204                source_version: args.version(),
205                if_match: args.if_match(),
206                if_none_match: args.if_none_match(),
207                if_modified_since: args.if_modified_since(),
208                if_unmodified_since: args.if_unmodified_since(),
209                upload_id,
210                part_number,
211                range,
212                operation: Operation::Write,
213            })?;
214
215        let resp = self
216            .core
217            .send(&self.ctx, req, self.core.signers.iam())
218            .await?;
219        match resp.status() {
220            StatusCode::OK => {
221                let (parts, body) = resp.into_parts();
222                let bs = body.to_bytes();
223                let result: CopyObjectResult =
224                    quick_xml::de::from_reader(bs.as_ref()).map_err(new_xml_deserialize_error)?;
225
226                // S3 may return 200 OK with an <Error> body for UploadPartCopy.
227                if result.etag.is_empty() {
228                    return Err(parse_error(
229                        error_context,
230                        Response::from_parts(parts, Buffer::from(bs)),
231                    ));
232                }
233
234                Ok(oio::MultipartPart {
235                    part_number,
236                    etag: result.etag,
237                    checksum: None,
238                    size: Some(size),
239                })
240            }
241            _ => Err(parse_error(error_context, resp)),
242        }
243    }
244
245    async fn complete_part(
246        &self,
247        upload_id: &str,
248        parts: &[oio::MultipartPart],
249    ) -> Result<Metadata> {
250        let parts = parts
251            .iter()
252            .map(|p| match &self.core.checksum_algorithm {
253                None => CompleteMultipartUploadRequestPart {
254                    part_number: p.part_number,
255                    etag: p.etag.clone(),
256                    ..Default::default()
257                },
258                Some(checksum_algorithm) => match checksum_algorithm {
259                    ChecksumAlgorithm::Crc32c => CompleteMultipartUploadRequestPart {
260                        part_number: p.part_number,
261                        etag: p.etag.clone(),
262                        checksum_crc32c: p.checksum.clone(),
263                    },
264                    ChecksumAlgorithm::Md5 => CompleteMultipartUploadRequestPart {
265                        part_number: p.part_number,
266                        etag: p.etag.clone(),
267                        ..Default::default()
268                    },
269                },
270            })
271            .collect();
272
273        let resp = self
274            .core
275            .s3_complete_multipart_upload(&self.ctx, &self.path, upload_id, parts, &self.op)
276            .await?;
277
278        let status = resp.status();
279
280        let meta = S3Writer::parse_header_into_meta(&self.path, resp.headers())?;
281
282        match status {
283            StatusCode::OK => {
284                // still check if there is any error because S3 might return error for status code 200
285                // https://docs.aws.amazon.com/AmazonS3/latest/API/API_CompleteMultipartUpload.html#API_CompleteMultipartUpload_Example_4
286                let (parts, body) = resp.into_parts();
287                let bs = body.to_bytes();
288
289                let ret: CompleteMultipartUploadResult =
290                    quick_xml::de::from_reader(bs.as_ref()).map_err(new_xml_deserialize_error)?;
291                if !ret.code.is_empty() {
292                    let err = parse_error(
293                        self.error_context(ServiceOperation("CompleteMultipartUpload")),
294                        Response::from_parts(parts, Buffer::from(bs)),
295                    );
296                    return if self.op.if_match().is_some() && err.kind() == ErrorKind::NotFound {
297                        Err(Error::new(
298                            ErrorKind::ConditionNotMatch,
299                            "write precondition requires a live target",
300                        ))
301                    } else {
302                        Err(err)
303                    };
304                }
305                let mut meta = meta.into_builder();
306                meta.etag(&ret.etag);
307
308                Ok(meta.build())
309            }
310            _ => {
311                let err = parse_error(
312                    self.error_context(ServiceOperation("CompleteMultipartUpload")),
313                    resp,
314                );
315                if self.op.if_match().is_some() && err.kind() == ErrorKind::NotFound {
316                    Err(Error::new(
317                        ErrorKind::ConditionNotMatch,
318                        "write precondition requires a live target",
319                    ))
320                } else {
321                    Err(err)
322                }
323            }
324        }
325    }
326
327    async fn abort_part(&self, upload_id: &str) -> Result<()> {
328        let resp = self
329            .core
330            .s3_abort_multipart_upload(&self.ctx, &self.path, upload_id)
331            .await?;
332        match resp.status() {
333            // s3 returns code 204 if abort succeeds.
334            StatusCode::NO_CONTENT => Ok(()),
335            _ => Err(parse_error(
336                ErrorContext::new(ServiceOperation("AbortMultipartUpload")),
337                resp,
338            )),
339        }
340    }
341}

343impl oio::AppendWrite for S3Writer {
344    async fn offset(&self) -> Result<u64> {
345        let resp = self
346            .core
347            .s3_head_object(&self.ctx, &self.path, OpStat::default())
348            .await?;
349
350        let status = resp.status();
351
352        match status {
353            StatusCode::OK => Ok(parse_content_length(resp.headers())?.unwrap_or_default()),
354            StatusCode::NOT_FOUND => Ok(0),
355            _ => Err(parse_error(
356                ErrorContext::new(ServiceOperation("HeadObject")),
357                resp,
358            )),
359        }
360    }
361
362    async fn append(&self, offset: u64, size: u64, body: Buffer) -> Result<Metadata> {
363        let error_ctx = if offset == 0 {
364            self.error_context(ServiceOperation("PutObject"))
365        } else {
366            ErrorContext::new(ServiceOperation("PutObject"))
367        };
368        let req = self
369            .core
370            .s3_append_object_request(&self.path, offset, size, &self.op, body)?;
371
372        let resp = self
373            .core
374            .send(&self.ctx, req, self.core.signers.default())
375            .await?;
376
377        let status = resp.status();
378
379        let meta = S3Writer::parse_header_into_meta(&self.path, resp.headers())?;
380
381        match status {
382            StatusCode::CREATED | StatusCode::OK => Ok(meta),
383            _ => Err(parse_error(error_ctx, resp)),
384        }
385    }
