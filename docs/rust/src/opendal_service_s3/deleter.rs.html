39}

41impl oio::BatchDelete for S3Deleter {
42    async fn delete_once(&self, path: String, args: OpDelete) -> Result<()> {
43        // This would delete the bucket, do not perform
44        if self.core.root == "/" && path == "/" {
45            return Ok(());
46        }
47
48        let error_ctx = ErrorContext::new(ServiceOperation("DeleteObject"))
49            .with_caller_condition(args.is_conditional());
50        let resp = self.core.s3_delete_object(&self.ctx, &path, &args).await?;
51
52        let status = resp.status();
53
54        match status {
55            StatusCode::NO_CONTENT => Ok(()),
56            // Allow 404 when deleting a non-existing object
57            // This is not a standard behavior, only some s3 alike service like GCS XML API do this.
58            // ref: <https://cloud.google.com/storage/docs/xml-api/delete-object>
59            StatusCode::NOT_FOUND if args.if_match().is_some() => Err(Error::new(
60                ErrorKind::ConditionNotMatch,
61                "delete precondition requires a live target",
62            )),
63            StatusCode::NOT_FOUND => Ok(()),
64            _ => Err(parse_error(error_ctx, resp)),
65        }
66    }
67
68    async fn delete_batch(&self, batch: Vec<(String, OpDelete)>) -> Result<BatchDeleteResult> {
69        let resp = self.core.s3_delete_objects(&self.ctx, &batch).await?;
70
71        let status = resp.status();
72        if status != StatusCode::OK {
73            return Err(parse_error(
74                ErrorContext::new(ServiceOperation("DeleteObjects")),
75                resp,
76            ));
77        }
78
79        let bs = resp.into_body();
80
81        let result: DeleteObjectsResult =
82            quick_xml::de::from_reader(bs.reader()).map_err(new_xml_deserialize_error)?;
83
84        let mut errors = result.error;
85        let mut batched_result = BatchDeleteResult {
86            // `errors.len()` is server-controlled and may exceed `batch.len()`; use
87            // `saturating_sub` (mirroring services/tos) to avoid an unsigned underflow
88            // that wraps to a huge `with_capacity` (release: abort).
89            succeeded: Vec::with_capacity(batch.len().saturating_sub(errors.len())),
90            failed: Vec::with_capacity(errors.len()),
91        };
92        for (path, op) in batch {
93            let abs_path = build_abs_path(&self.core.root, &path);
94            // Assume errors are rare, so lookup and erase is acceptable.
95            if let Some(idx) = errors
96                .iter()
97                .position(|e| e.key == abs_path && e.version_id.as_deref() == op.version())
98            {
99                let error = errors.swap_remove(idx);
100                let conditional = op.if_match().is_some();
101                batched_result.failed.push((
102                    path,
103                    op,
104                    parse_delete_objects_result_error(error, conditional),
105                ));
106            } else {
107                batched_result.succeeded.push((path, op));
108            }
109        }
110
111        Ok(batched_result)
112    }
113}
114
115fn parse_delete_objects_result_error(err: DeleteObjectsResultError, conditional: bool) -> Error {
116    let error_ctx =
117        ErrorContext::new(ServiceOperation("DeleteObjects")).with_caller_condition(conditional);
118    let (kind, retryable) =
119        parse_s3_error_code(error_ctx, err.code.as_str()).unwrap_or((ErrorKind::Unexpected, false));
120    let kind = if conditional && kind == ErrorKind::NotFound {
121        ErrorKind::ConditionNotMatch
122    } else {
123        kind
124    };
125    let mut err: Error = Error::new(kind, format!("{err:?}"));
126    if retryable {
127        err = err.set_temporary();
128    }
129    err
130}
