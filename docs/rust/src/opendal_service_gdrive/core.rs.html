107    pub async fn trash_path_if_exists(&self, ctx: &OperationContext, path: &str) -> Result<()> {
108        let mut target_id = match self.resolve_path(ctx, path).await? {
109            Some(id) => Some(id),
110            None => self.resolve_path_after_refresh(ctx, path).await?,
111        };
112
113        if let Some(id) = target_id.take() {
114            let mut resp = self.gdrive_trash(ctx, &id).await?;
115            if resp.status() == StatusCode::NOT_FOUND {
116                self.refresh_path(path).await;
117                target_id = self.resolve_path(ctx, path).await?;
118                if let Some(id) = target_id {
119                    resp = self.gdrive_trash(ctx, &id).await?;
120                } else {
121                    return Ok(());
122                }
123            }
124
125            if resp.status() == StatusCode::NOT_FOUND {
126                return Ok(());
127            }
128            if resp.status() != StatusCode::OK {
129                return Err(parse_error(
130                    ErrorContext::new(ServiceOperation("TrashFile")),
131                    resp,
132                ));
133            }
134
135            self.invalidate_file_id(path).await;
136            self.invalidate_dir_id(path).await;
137        }
138
139        Ok(())
140    }

143        let mut recent_entries = self.recent_entries.lock().await;
144        prune_recent_entries(&mut recent_entries);

181    pub async fn recent_entry_for_path(&self, path: &str) -> GdriveRecentPathState {
182        let mut recent_entries = self.recent_entries.lock().await;
183        prune_recent_entries(&mut recent_entries);
184
185        let recent_entry = lookup_recent_entry(&recent_entries.entries, path);
186        let tombstone = lookup_recent_tombstone(&recent_entries.tombstones, path);
187
188        match (recent_entry, tombstone) {
189            (Some(entry), Some(tombstone_expires_at))
190                if tombstone_expires_at > entry.expires_at =>
191            {
192                GdriveRecentPathState::Deleted
193            }
194            (Some(entry), _) => match entry.metadata.clone() {
195                Some(metadata) => GdriveRecentPathState::Present(Box::new(metadata)),
196                None => GdriveRecentPathState::Deleted,
197            },
198            (None, Some(_)) => GdriveRecentPathState::Deleted,
199            (None, None) => GdriveRecentPathState::Missing,
200        }
201    }

203    pub async fn recent_entries_for_list(
204        &self,
205        scope_path: &str,
206        recursive: bool,
207    ) -> Vec<(String, Metadata)> {
208        let mut recent_entries = self.recent_entries.lock().await;
209        prune_recent_entries(&mut recent_entries);
210
211        let mut entries = recent_entries
212            .entries
213            .iter()
214            .filter_map(|(path, entry)| {
215                let metadata = entry.metadata.clone()?;
216                if metadata.mode().is_dir() && path != &normalize_dir_path(path) {
217                    return None;
218                }
219                if let Some(tombstone_expires_at) =
220                    lookup_recent_tombstone(&recent_entries.tombstones, path)
221                    && tombstone_expires_at > entry.expires_at
222                {
223                    return None;
224                }
225                if let Some(latest_entry) = lookup_recent_entry(&recent_entries.entries, path)
226                    && latest_entry.expires_at > entry.expires_at
227                {
228                    return None;
229                }
230                if recent_entry_in_scope(scope_path, path, metadata.mode(), recursive) {
231                    Some((path.clone(), metadata))
232                } else {
233                    None
234                }
235            })
236            .collect::<Vec<_>>();
237
238        entries.sort_by(|left, right| left.0.cmp(&right.0));
239        entries
240    }

261    pub async fn gdrive_get(
262        &self,
263        ctx: &OperationContext,
264        path: &str,
265        range: BytesRange,
266    ) -> Result<Response<HttpBody>> {
267        let path = build_abs_path(&self.root, path);
268        match self.recent_entry_for_path(&path).await {
269            GdriveRecentPathState::Deleted => {
270                return Err(Error::new(
271                    ErrorKind::NotFound,
272                    format!("path not found: {path}"),
273                ));
274            }
275            GdriveRecentPathState::Present(_) | GdriveRecentPathState::Missing => {}
276        }
277        let path_id = match self.resolve_path(ctx, &path).await? {
278            Some(id) => id,
279            None => match self.resolve_path_after_refresh(ctx, &path).await? {
280                Some(id) => id,
281                None => {
282                    return Err(Error::new(
283                        ErrorKind::NotFound,
284                        format!("path not found: {path}"),
285                    ));
286                }
287            },
288        };
289
290        let url: String = format!(
291            "https://www.googleapis.com/drive/v3/files/{path_id}?alt=media&supportsAllDrives=true"
292        );
293
294        let mut req = Request::get(&url)
295            .extension(Operation::Read)
296            .extension(ServiceOperation("DownloadFile"))
297            .header(header::RANGE, range.to_header())
298            .body(Buffer::new())
299            .map_err(new_request_build_error)?;
300        self.sign(ctx, &mut req).await?;
301
302        ctx.http_transport().fetch(req).await
303    }

305    pub async fn gdrive_list(
306        &self,
307        ctx: &OperationContext,
308        file_id: &str,
309        page_size: i32,
310        next_page_token: &str,
311    ) -> Result<Response<Buffer>> {
312        let q = format!("'{file_id}' in parents and trashed = false");
313        let url = "https://www.googleapis.com/drive/v3/files";
314        let mut url = QueryPairsWriter::new(url);
315        url = url.push("pageSize", &page_size.to_string());
316        url = url.push("q", &percent_encode_path(&q));
317        url = url.push(
318            "fields",
319            "nextPageToken,files(id,name,mimeType,size,modifiedTime,parents)",
320        );
321        if !next_page_token.is_empty() {
322            url = url.push("pageToken", next_page_token);
323        };
324        url = url.push("supportsAllDrives", "true");
325        url = url.push("includeItemsFromAllDrives", "true");
326        if let Some(drive_id) = &self.drive_id {
327            url = url.push("corpora", "drive");
328            url = url.push("driveId", &percent_encode_path(drive_id));
329        }
330
331        let mut req = Request::get(url.finish())
332            .extension(Operation::List)
333            .extension(ServiceOperation("ListFiles"))
334            .body(Buffer::new())
335            .map_err(new_request_build_error)?;
336        self.sign(ctx, &mut req).await?;
337
338        ctx.http_transport().send(req).await
339    }

341    /// List multiple directories in a single API call using OR query.
342    /// This is much more efficient than listing directories one by one.
343    /// Google Drive API supports up to ~50 parent IDs in a single query.
344    pub async fn gdrive_list_batch(
345        &self,
346        ctx: &OperationContext,
347        file_ids: &[String],
348        page_size: i32,
349        next_page_token: &str,
350    ) -> Result<Response<Buffer>> {
351        if file_ids.is_empty() {
352            return Err(Error::new(
353                ErrorKind::Unexpected,
354                "file_ids cannot be empty",
355            ));
356        }
357
358        // Build OR query: ('id1' in parents or 'id2' in parents or ...)
359        let parents_query = file_ids
360            .iter()
361            .map(|id| format!("'{}' in parents", id))
362            .collect::<Vec<_>>()
363            .join(" or ");
364        let q = format!("({}) and trashed = false", parents_query);
365
366        let url = "https://www.googleapis.com/drive/v3/files";
367        let mut url = QueryPairsWriter::new(url);
368        url = url.push("pageSize", &page_size.to_string());
369        url = url.push("q", &percent_encode_path(&q));
370        // Include 'parents' field to know which directory each file belongs to
371        url = url.push(
372            "fields",
373            "nextPageToken,files(id,name,mimeType,size,modifiedTime,parents)",
374        );
375        if !next_page_token.is_empty() {
376            url = url.push("pageToken", next_page_token);
377        };
378        url = url.push("supportsAllDrives", "true");
379        url = url.push("includeItemsFromAllDrives", "true");
380        if let Some(drive_id) = &self.drive_id {
381            url = url.push("corpora", "drive");
382            url = url.push("driveId", &percent_encode_path(drive_id));
383        }
384
385        let mut req = Request::get(url.finish())
386            .extension(Operation::List)
387            .extension(ServiceOperation("ListFiles"))
388            .body(Buffer::new())
389            .map_err(new_request_build_error)?;
390        self.sign(ctx, &mut req).await?;
391
392        ctx.http_transport().send(req).await
393    }

396    pub async fn gdrive_patch_metadata_request(
397        &self,
398        ctx: &OperationContext,
399        source: &str,
400        target: &str,
401    ) -> Result<Response<Buffer>> {
402        let source_file_id = match self.resolve_path(ctx, source).await? {
403            Some(id) => id,
404            None => self
405                .resolve_path_after_refresh(ctx, source)
406                .await?
407                .ok_or(Error::new(
408                    ErrorKind::NotFound,
409                    format!("source path not found: {source}"),
410                ))?,
411        };
412        let source_parent = get_parent(source);
413        let source_parent_id = match self.resolve_path(ctx, source_parent).await? {
414            Some(id) => id,
415            None => self
416                .resolve_path_after_refresh(ctx, source_parent)
417                .await?
418                .expect("old parent must exist"),
419        };
420
421        let target_parent_id = self.ensure_dir(ctx, get_parent(target)).await?;
422        let target_file_name = get_basename(target);
423
424        let metadata = &json!({
425            "name": target_file_name,
426            "removeParents": [source_parent_id],
427            "addParents": [target_parent_id],
428        });
429
430        let url = format!(
431            "https://www.googleapis.com/drive/v3/files/{source_file_id}?fields=id,name,mimeType,size,modifiedTime,parents&supportsAllDrives=true"
432        );
433        let mut req = Request::patch(url)
434            .extension(Operation::Rename)
435            .extension(ServiceOperation("MoveFile"))
436            .body(Buffer::from(Bytes::from(metadata.to_string())))
437            .map_err(new_request_build_error)?;
438
439        self.sign(ctx, &mut req).await?;
440
441        ctx.http_transport().send(req).await
442    }

469    pub async fn gdrive_upload_simple_request(
470        &self,
471        ctx: &OperationContext,
472        path: &str,
473        size: u64,
474        body: Buffer,
475    ) -> Result<Response<Buffer>> {
476        let parent = self.ensure_dir(ctx, get_parent(path)).await?;
477
478        let url = "https://www.googleapis.com/upload/drive/v3/files?uploadType=multipart&supportsAllDrives=true";
479
480        let file_name = get_basename(path);
481
482        let metadata = serde_json::to_vec(&json!({
483            "name": file_name,
484            "parents": [parent],
485        }))
486        .map_err(new_json_serialize_error)?;
487
488        let req = Request::post(url)
489            .header("X-Upload-Content-Length", size)
490            .extension(Operation::Write)
491            .extension(ServiceOperation("UploadFile"));
492
493        let multipart = Multipart::new()
494            .part(
495                FormDataPart::new("metadata")
496                    .header(
497                        header::CONTENT_TYPE,
498                        "application/json; charset=UTF-8".parse().unwrap(),
499                    )
500                    .content(metadata),
501            )
502            .part(
503                FormDataPart::new("file")
504                    .header(
505                        header::CONTENT_TYPE,
506                        "application/octet-stream".parse().unwrap(),
507                    )
508                    .content(body),
509            );
510
511        let mut req = multipart.apply(req)?;
512
513        self.sign(ctx, &mut req).await?;
514
515        ctx.http_transport().send(req).await
516    }

553    pub async fn gdrive_copy(
554        &self,
555        ctx: &OperationContext,
556        from: &str,
557        to: &str,
558    ) -> Result<Response<Buffer>> {
559        let from = build_abs_path(&self.root, from);
560
561        let from_file_id = match self.resolve_path(ctx, &from).await? {
562            Some(id) => id,
563            None => match self.resolve_path_after_refresh(ctx, &from).await? {
564                Some(id) => id,
565                None => {
566                    return Err(Error::new(
567                        ErrorKind::NotFound,
568                        "the file to copy does not exist",
569                    ));
570                }
571            },
572        };
573
574        let to_name = get_basename(to);
575        let to_path = build_abs_path(&self.root, to);
576        let to_parent_id = self.ensure_dir(ctx, get_parent(&to_path)).await?;
577
578        self.trash_path_if_exists(ctx, &to_path).await?;
579
580        let url = format!(
581            "https://www.googleapis.com/drive/v3/files/{from_file_id}/copy?fields=id,name,mimeType,size,modifiedTime,parents&supportsAllDrives=true"
582        );
583
584        let request_body = &json!({
585            "name": to_name,
586            "parents": [to_parent_id],
587        });
588        let body = Buffer::from(Bytes::from(request_body.to_string()));
589
590        let mut req = Request::post(&url)
591            .extension(Operation::Copy)
592            .extension(ServiceOperation("CopyFile"))
593            .body(body)
594            .map_err(new_request_build_error)?;
595        self.sign(ctx, &mut req).await?;
596
597        ctx.http_transport().send(req).await
598    }

601#[derive(Clone)]
602pub struct GdriveRecentEntry {
603    metadata: Option<Metadata>,
604    expires_at: Timestamp,
605}
606
607#[derive(Default)]
608pub struct GdriveRecentState {
609    entries: HashMap<String, GdriveRecentEntry>,
610    tombstones: HashMap<String, Timestamp>,
611}
612
613#[derive(Clone, Debug)]
614pub enum GdriveRecentPathState {
615    Missing,
616    Deleted,
617    Present(Box<Metadata>),
618}
619
620pub(crate) fn normalize_dir_path(path: &str) -> String {
621    if path.is_empty() || path.ends_with('/') {
622        path.to_string()
623    } else {
624        format!("{path}/")
625    }
626}
627
628fn canonical_recent_path(path: &str, mode: EntryMode) -> String {
629    if mode.is_dir() {
630        normalize_dir_path(path)
631    } else {
632        path.to_string()
633    }
634}
635
636fn prune_recent_entries(entries: &mut GdriveRecentState) {
637    let now = Timestamp::now();
638    entries.entries.retain(|_, entry| entry.expires_at > now);
639    entries.tombstones.retain(|_, expires_at| *expires_at > now);
640}
641
642fn lookup_recent_entry<'a>(
643    entries: &'a HashMap<String, GdriveRecentEntry>,
644    path: &str,
645) -> Option<&'a GdriveRecentEntry> {
646    entries.get(path)
647}
648
649fn lookup_recent_tombstone(
650    tombstones: &HashMap<String, Timestamp>,
651    path: &str,
652) -> Option<Timestamp> {
653    let candidates = recent_path_candidates(path);
654    tombstones
655        .iter()
656        .filter(|(tombstone, _)| {
657            candidates
658                .iter()
659                .any(|candidate| path_under_tombstone(candidate, tombstone))
660        })
661        .map(|(_, expires_at)| *expires_at)
662        .max()
663}
664
665fn revive_recent_parent_dirs(entries: &mut GdriveRecentState, path: &str, expires_at: Timestamp) {
666    let mut parent = get_parent(path).to_string();
667
668    while !parent.is_empty() && parent != "/" {
669        if lookup_recent_tombstone(&entries.tombstones, &parent).is_some() {
670            insert_recent_entry(
671                &mut entries.entries,
672                &parent,
673                EntryMode::DIR,
674                GdriveRecentEntry {
675                    metadata: Some(MetadataBuilder::dir().build()),
676                    expires_at,
677                },
678            );
679        }
680
681        parent = get_parent(&parent).to_string();
682    }
683}
684
685fn insert_recent_entry(
686    entries: &mut HashMap<String, GdriveRecentEntry>,
687    path: &str,
688    mode: EntryMode,
689    entry: GdriveRecentEntry,
690) {
691    entries.insert(path.to_string(), entry.clone());
692
693    if mode.is_dir() {
694        let alias = path.trim_end_matches('/');
695        if !alias.is_empty() {
696            entries.insert(alias.to_string(), entry);
697        }
698    }
699}

701fn recent_path_candidates(path: &str) -> Vec<String> {
702    let mut candidates = vec![path.to_string()];
703    let dir_path = normalize_dir_path(path);
704    if dir_path != path {
705        candidates.push(dir_path);
706    }
707    candidates
708}
709
710fn path_under_tombstone(path: &str, tombstone: &str) -> bool {
711    tombstone.is_empty() || path == tombstone || path.starts_with(tombstone)
712}
713
714fn recent_entry_in_scope(scope_path: &str, path: &str, mode: EntryMode, recursive: bool) -> bool {
715    if recursive {
716        if scope_path.is_empty() {
717            return true;
718        }
719
720        if scope_path.ends_with('/') {
721            return path.starts_with(scope_path) && path != scope_path;
722        }
723
724        return path == canonical_recent_path(scope_path, mode) || path.starts_with(scope_path);
725    }
726
727    let parent = get_parent(path);
728    if scope_path.is_empty() {
729        parent == "/"
730    } else {
731        parent == scope_path
732    }
733}

735#[derive(Clone)]
736pub struct GdriveSigner {
737    pub client_id: String,
738    pub client_secret: String,
739    pub refresh_token: String,
740
741    pub access_token: String,
742    pub expires_in: Timestamp,
743}
744
745impl GdriveSigner {
746    /// Create a new signer.
747    pub fn new() -> Self {
748        GdriveSigner {
749            client_id: "".to_string(),
750            client_secret: "".to_string(),
751            refresh_token: "".to_string(),
752            access_token: "".to_string(),
753            expires_in: Timestamp::MIN,
754        }
755    }
756
757    /// Sign a request.
758    pub async fn sign<T>(&mut self, ctx: &OperationContext, req: &mut Request<T>) -> Result<()> {
759        if !self.access_token.is_empty() && self.expires_in > Timestamp::now() {
760            let value = format!("Bearer {}", self.access_token)
761                .parse()
762                .expect("access token must be valid header value");
763
764            req.headers_mut().insert(header::AUTHORIZATION, value);
765            return Ok(());
766        }
767
768        let url = format!(
769            "https://oauth2.googleapis.com/token?refresh_token={}&client_id={}&client_secret={}&grant_type=refresh_token",
770            self.refresh_token, self.client_id, self.client_secret
771        );
772
773        {
774            let req = Request::post(url)
775                .header(header::CONTENT_LENGTH, 0)
776                .body(Buffer::new())
777                .map_err(new_request_build_error)?;
778
779            let resp = ctx.http_transport().send(req).await?;
780            let status = resp.status();
781
782            match status {
783                StatusCode::OK => {
784                    let resp_body = resp.into_body();
785                    let token: GdriveTokenResponse = serde_json::from_reader(resp_body.reader())
786                        .map_err(new_json_deserialize_error)?;
787                    self.access_token.clone_from(&token.access_token);
788                    self.expires_in = Timestamp::now() + Duration::from_secs(token.expires_in)
789                        - Duration::from_secs(120);
790                }
791                _ => {
792                    return Err(parse_error(
793                        ErrorContext::new(ServiceOperation("RefreshAccessToken")),
794                        resp,
795                    ));
796                }
797            }
798        }
799
800        let auth_header_content = format!("Bearer {}", self.access_token);
801        req.headers_mut()
802            .insert(header::AUTHORIZATION, auth_header_content.parse().unwrap());
803
804        Ok(())
805    }
806}

819impl crate::path_index::GdrivePathQueryer for GdrivePathQuery {
820    async fn root(&self, _: &OperationContext) -> Result<String> {
821        // The id of a shared drive is also the id of its root folder.
822        match &self.drive_id {
823            Some(drive_id) => Ok(drive_id.clone()),
824            None => Ok("root".to_string()),
825        }
826    }
827
828    async fn query(
829        &self,
830        ctx: &OperationContext,
831        parent_id: &str,
832        name: &str,
833    ) -> Result<Option<String>> {
834        let mut queries = vec![
835            // Make sure name has been replaced with escaped name.
836            //
837            // ref: <https://developers.google.com/drive/api/guides/ref-search-terms>
838            format!(
839                "name = '{}'",
840                name.replace('\'', "\\'").trim_end_matches('/')
841            ),
842            format!("'{}' in parents", parent_id),
843            "trashed = false".to_string(),
844        ];
845        if name.ends_with('/') {
846            queries.push("mimeType = 'application/vnd.google-apps.folder'".to_string());
847        }
848        let query = queries.join(" and ");
849        let order_by = "modifiedTime desc,createdTime desc";
850
851        let mut url = format!(
852            "https://www.googleapis.com/drive/v3/files?q={}&orderBy={}&pageSize=1&supportsAllDrives=true&includeItemsFromAllDrives=true",
853            percent_encode_path(query.as_str()),
854            percent_encode_path(order_by)
855        );
856        if let Some(drive_id) = &self.drive_id {
857            url.push_str("&corpora=drive&driveId=");
858            url.push_str(&percent_encode_path(drive_id));
859        }
860
861        let mut req = Request::get(&url)
862            .extension(Operation::Stat)
863            .extension(ServiceOperation("ListFiles"))
864            .body(Buffer::new())
865            .map_err(new_request_build_error)?;
866
867        self.signer.lock().await.sign(ctx, &mut req).await?;
868
869        let resp = ctx.http_transport().send(req).await?;
870        let status = resp.status();
871
872        match status {
873            StatusCode::OK => {
874                let body = resp.into_body();
875                let meta: GdriveFileList =
876                    serde_json::from_reader(body.reader()).map_err(new_json_deserialize_error)?;
877
878                if let Some(f) = meta.files.first() {
879                    Ok(Some(f.id.clone()))
880                } else {
881                    Ok(None)
882                }
883            }
884            _ => Err(parse_error(
885                ErrorContext::new(ServiceOperation("ListFiles")),
886                resp,
887            )),
888        }
889    }
890
891    async fn create_dir(
892        &self,
893        ctx: &OperationContext,
894        parent_id: &str,
895        name: &str,
896    ) -> Result<String> {
897        let url = "https://www.googleapis.com/drive/v3/files?supportsAllDrives=true";
898
899        let content = serde_json::to_vec(&json!({
900            "name": name.trim_end_matches('/'),
901            "mimeType": "application/vnd.google-apps.folder",
902            // If the parent is not provided, the folder will be created in the root folder.
903            "parents": [parent_id],
904        }))
905        .map_err(new_json_serialize_error)?;
906
907        let mut req = Request::post(url)
908            .extension(Operation::CreateDir)
909            .extension(ServiceOperation("CreateFolder"))
910            .header(header::CONTENT_TYPE, "application/json")
911            .body(Buffer::from(Bytes::from(content)))
912            .map_err(new_request_build_error)?;
913
914        self.signer.lock().await.sign(ctx, &mut req).await?;
915
916        let resp = ctx.http_transport().send(req).await?;
917        if !resp.status().is_success() {
918            return Err(parse_error(
919                ErrorContext::new(ServiceOperation("CreateFolder")),
920                resp,
921            ));
922        }
923
924        let body = resp.into_body();
925        let file: GdriveFile =
926            serde_json::from_reader(body.reader()).map_err(new_json_deserialize_error)?;
927        Ok(file.id)
928    }
929}

1143        Self { service_operation }
1144    }

1147/// Parse an error response using its service request context.
1148pub(crate) fn parse_error(ctx: ErrorContext, resp: Response<Buffer>) -> Error {
1149    let (parts, body) = resp.into_parts();
1150    let bs = body.to_bytes();
1151
1152    let (mut kind, mut retryable) = match parts.status {
1153        StatusCode::NOT_FOUND => (ErrorKind::NotFound, false),
1154        StatusCode::FORBIDDEN => (ErrorKind::PermissionDenied, false),
1155        StatusCode::INTERNAL_SERVER_ERROR
1156        | StatusCode::BAD_GATEWAY
1157        | StatusCode::SERVICE_UNAVAILABLE
1158        | StatusCode::GATEWAY_TIMEOUT
1159        // Gdrive sometimes return METHOD_NOT_ALLOWED for our requests for abuse detection.
1160        | StatusCode::METHOD_NOT_ALLOWED => (ErrorKind::Unexpected, true),
1161        _ => (ErrorKind::Unexpected, false),
1162    };
1163
1164    let (message, gdrive_err) = serde_json::from_slice::<GdriveError>(bs.as_ref())
1165        .map(|gdrive_err| (format!("{gdrive_err:?}"), Some(gdrive_err)))
1166        .unwrap_or_else(|_| (String::from_utf8_lossy(&bs).into_owned(), None));
1167
1168    if let Some(gdrive_err) = gdrive_err {
1169        (kind, retryable) =
1170            parse_gdrive_error_code(gdrive_err.error.message.as_str()).unwrap_or((kind, retryable));
1171    }
1172
1173    let mut err = Error::new(kind, message);
1174
1175    err = err.with_context("service_operation", ctx.service_operation.0);
1176    err = with_error_response_context(err, parts);
1177
1178    if retryable {
1179        err = err.set_temporary();
1180    }
1181
1182    err
1183}

1188        //
1189        // It's Ok to retry since later on the request rate may get reduced.
1190        "User rate limit exceeded." => Some((ErrorKind::RateLimited, true)),
1191        _ => None,
1192    }
