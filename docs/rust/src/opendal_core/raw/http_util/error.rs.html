18use http::Uri;
19use http::response::Parts;

21use crate::Error;
22use crate::ErrorKind;

24/// Create a new error happened during building request.
25pub fn new_request_build_error(err: http::Error) -> Error {
26    Error::new(ErrorKind::Unexpected, "building http request")
27        .with_operation("http::Request::build")
28        .set_source(err)
29}
30
31/// Create a new error happened during signing request.
32pub fn new_request_credential_error(err: anyhow::Error) -> Error {
33    Error::new(
34        ErrorKind::Unexpected,
35        "loading credential to sign http request",
36    )
37    .set_temporary()
38    .with_operation("reqsign::LoadCredential")
39    .set_source(err)
40}
41
42/// Wrap a signing failure, preserving reqsign's retryability when available.
43pub fn new_request_sign_error(err: anyhow::Error) -> Error {
44    let error = Error::new(ErrorKind::Unexpected, "signing http request");
45    #[cfg(feature = "reqsign")]
46    let error = error.with_temporary(
47        err.downcast_ref::<reqsign_core::Error>()
48            .is_some_and(reqsign_core::Error::is_retryable),
49    );
50    error.with_operation("reqsign::Sign").set_source(err)
51}

53/// Add response context to error.
54///
55/// This helper function will:
56///
57/// - remove sensitive or useless headers from parts.
58/// - fetch uri if parts extensions contains `Uri`.
59pub fn with_error_response_context(mut err: Error, mut parts: Parts) -> Error {
60    if let Some(uri) = parts.extensions.get::<Uri>() {
61        err = err.with_context("uri", uri.to_string());
62    }
63
64    // The following headers may contains sensitive information.
65    parts.headers.remove("Set-Cookie");
66    parts.headers.remove("WWW-Authenticate");
67    parts.headers.remove("Proxy-Authenticate");
68
69    if parts.headers.contains_key(http::header::LOCATION) {
70        let location = crate::HttpUri::from_response_location(&mut parts)
71            .map(|uri| uri.redacted_uri())
72            .unwrap_or("<invalid URL>")
73            .parse()
74            .unwrap_or_else(|_| http::HeaderValue::from_static("<invalid URL>"));
75        parts.headers.insert(http::header::LOCATION, location);
76    }
77
78    err = err.with_context("response", format!("{parts:?}"));
79
80    err
81}
