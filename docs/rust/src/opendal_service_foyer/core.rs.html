39/// FoyerCore holds the foyer HybridCache instance.
40///
41/// It supports lazy initialization when constructed from URI.
42#[derive(Clone)]
43pub struct FoyerCore {
44    /// OnceLock for lazy cache initialization.
45    cache: std::sync::OnceLock<Arc<HybridCache<FoyerKey, FoyerValue>>>,
46    deleted_keys: Arc<Mutex<HashSet<FoyerKey>>>,
47    /// Config
48    config: FoyerConfig,
49}
50
51impl Debug for FoyerCore {
52    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
53        f.debug_struct("FoyerCore")
54            .field("name", &self.config.name)
55            .finish_non_exhaustive()
56    }
57}
58
59const FOYER_DEFAULT_MEMORY_BYTES: usize = 1024 * 1024 * 1024;
60const FOYER_DEFAULT_DISK_FILE_SIZE: usize = 1024 * 1024;
61const FOYER_DEFAULT_SHARDS: usize = 1;
62
63fn parse_recover_mode(mode_str: &str) -> Result<RecoverMode> {
64    match mode_str.to_lowercase().as_str() {
65        "none" => Ok(RecoverMode::None),
66        "quiet" => Ok(RecoverMode::Quiet),
67        "strict" => Ok(RecoverMode::Strict),
68        _ => Err(Error::new(
69            ErrorKind::ConfigInvalid,
70            format!(
71                "invalid recover_mode: {}, expected 'none', 'quiet', or 'strict'",
72                mode_str
73            ),
74        )),
75    }
76}
77
78impl FoyerCore {
79    /// Create a new FoyerCore with a pre-built cache.
80    ///
81    /// This is used when the user provides a `HybridCache` via `Foyer::new(cache)`.
82    pub fn new(config: FoyerConfig) -> Self {
83        Self {
84            cache: std::sync::OnceLock::new(),
85            deleted_keys: Arc::new(Mutex::new(HashSet::new())),
86            config,
87        }
88    }
89
90    /// Initialize the cache with the given pre-built instance.
91    pub(crate) fn with_cache(self, cache: Arc<HybridCache<FoyerKey, FoyerValue>>) -> Self {
92        let _ = self.cache.set(cache);
93        self
94    }
95
96    /// Get the cache, initializing lazily if needed.
97    async fn get_cache(&self) -> Result<Arc<HybridCache<FoyerKey, FoyerValue>>> {
98        if let Some(cache) = self.cache.get() {
99            return Ok(cache.clone());
100        }
101
102        let memory = self.config.memory.unwrap_or(FOYER_DEFAULT_MEMORY_BYTES);
103        let shards = self.config.shards.unwrap_or(FOYER_DEFAULT_SHARDS);
104        let recover_mode = if let Some(ref mode_str) = self.config.recover_mode {
105            parse_recover_mode(mode_str)?
106        } else {
107            RecoverMode::None
108        };
109
110        let mut builder = HybridCacheBuilder::new()
111            .memory(memory)
112            .with_shards(shards)
113            .storage()
114            .with_recover_mode(recover_mode);
115
116        // Configure disk storage if disk_path is provided
117        if let Some(ref disk_path) = self.config.disk_path {
118            let path = PathBuf::from(disk_path);
119
120            let mut device = FsDeviceBuilder::new(path);
121
122            if let Some(capacity) = self.config.disk_capacity {
123                device = device.with_capacity(capacity);
124            }
125
126            let block_size = self
127                .config
128                .disk_file_size
129                .unwrap_or(FOYER_DEFAULT_DISK_FILE_SIZE);
130            let device = device.build().map_err(|e| {
131                Error::new(ErrorKind::Unexpected, "failed to build foyer cache device")
132                    .set_source(e)
133            })?;
134
135            builder = builder
136                .with_engine_config(BlockEngineConfig::new(device).with_block_size(block_size));
137        }
138
139        let cache = Arc::new(builder.build().await.map_err(|e| {
140            Error::new(ErrorKind::Unexpected, "failed to build foyer cache").set_source(e)
141        })?);
142
143        if self.cache.set(cache.clone()).is_err() {
144            return self.cache.get().cloned().ok_or_else(|| {
145                Error::new(ErrorKind::Unexpected, "failed to initialize foyer cache")
146            });
147        }
148
149        Ok(cache)
150    }
151
152    pub fn name(&self) -> Option<&str> {
153        self.config.name.as_deref()
154    }
155
156    pub async fn get(&self, key: &str) -> Result<Option<Buffer>> {
157        let cache = self.get_cache().await?;
158        let foyer_key = FoyerKey {
159            path: key.to_string(),
160        };
161        if self
162            .deleted_keys
163            .lock()
164            .expect("deleted keys lock poisoned")
165            .contains(&foyer_key)
166        {
167            return Ok(None);
168        }
169
170        match cache.get(&foyer_key).await {
171            Ok(Some(entry)) => Ok(Some(entry.value().0.clone())),
172            Ok(None) => Ok(None),
173            Err(_) => Ok(None),
174        }
175    }
176
177    pub async fn insert(&self, key: &str, value: Buffer) -> Result<()> {
178        let cache = self.get_cache().await?;
179        let key = FoyerKey {
180            path: key.to_string(),
181        };
182        cache.insert(key.clone(), FoyerValue(value));
183        self.deleted_keys
184            .lock()
185            .expect("deleted keys lock poisoned")
186            .remove(&key);
187        Ok(())
188    }
189
190    pub async fn remove(&self, key: &str) -> Result<()> {
191        let cache = self.get_cache().await?;
192        let key = FoyerKey {
193            path: key.to_string(),
194        };
195        cache.remove(&key);
196        self.deleted_keys
197            .lock()
198            .expect("deleted keys lock poisoned")
199            .insert(key);
200        Ok(())
201    }
202}
