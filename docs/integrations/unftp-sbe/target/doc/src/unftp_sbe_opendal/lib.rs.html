18//! `unftp-sbe-opendal` is an [unftp](https://crates.io/crates/unftp) `StorageBackend` implementation using opendal.
19//!
20//! This crate can help you to access ANY storage services with the same ftp API.
21//!
22//! # Example
23//!
24//! This example demonstrates how to use `unftp-sbe-opendal` with the `S3` service.
25//!
26//! ```no_run
27//! use anyhow::Result;
28//! use opendal::Operator;
29//! use opendal::services;
30//! use unftp_sbe_opendal::OpendalStorage;
31//!
32//! #[tokio::main]
33//! async fn main() -> Result<()> {
34//!     // Create any service desired
35//!     let op = opendal::Operator::from_iter::<services::S3>(
36//!         [
37//!             ("bucket".to_string(), "my_bucket".to_string()),
38//!             ("access_key".to_string(), "my_access_key".to_string()),
39//!             ("secret_key".to_string(), "my_secret_key".to_string()),
40//!             ("endpoint".to_string(), "my_endpoint".to_string()),
41//!             ("region".to_string(), "my_region".to_string()),
42//!         ],
43//!     )?;
44//!
45//!     // Wrap the operator with `OpendalStorage`
46//!     let backend = OpendalStorage::new(op);
47//!
48//!     // Build the actual unftp server
49//!     let server = libunftp::ServerBuilder::new(Box::new(move || backend.clone())).build()?;
50//!
51//!     // Start the server
52//!     server.listen("0.0.0.0:0").await?;
53//!
54//!     Ok(())
55//! }
56//! ```

80/// A wrapper around [`opendal::Metadata`] to implement [`storage::Metadata`].
81#[derive(Debug, Clone, Eq, PartialEq)]
82pub struct OpendalMetadata(opendal::Metadata);
83
84impl storage::Metadata for OpendalMetadata {
85    fn len(&self) -> u64 {
86        self.0.content_length()
87    }
88
89    fn is_dir(&self) -> bool {
90        self.0.is_dir()
91    }
92
93    fn is_file(&self) -> bool {
94        self.0.is_file()
95    }
96
97    fn is_symlink(&self) -> bool {
98        false
99    }
100
101    fn modified(&self) -> storage::Result<std::time::SystemTime> {
102        match self.0.last_modified() {
103            Some(ts) => Ok(ts.into()),
104            None => Err(Error::new(
105                storage::ErrorKind::LocalError,
106                "no last modified time",
107            )),
108        }
109    }
110
111    fn gid(&self) -> u32 {
112        0
113    }
114
115    fn uid(&self) -> u32 {
116        0
117    }
118}

120fn convert_err(err: opendal::Error) -> Error {
121    let kind = match err.kind() {
122        opendal::ErrorKind::NotFound => storage::ErrorKind::PermanentFileNotAvailable,
123        opendal::ErrorKind::AlreadyExists => storage::ErrorKind::PermanentFileNotAvailable,
124        opendal::ErrorKind::PermissionDenied => storage::ErrorKind::PermissionDenied,
125        _ => storage::ErrorKind::LocalError,
126    };
127    Error::new(kind, err)
128}
129
130fn convert_path(path: &Path) -> storage::Result<&str> {
131    path.to_str().ok_or_else(|| {
132        Error::new(
133            storage::ErrorKind::LocalError,
134            "Path is not a valid UTF-8 string",
135        )
136    })
137}

139async fn copy_read_write_loop<R, W>(input: &mut R, output: &mut W) -> std::io::Result<u64>
140where
141    R: tokio::io::AsyncRead + Unpin + ?Sized,
142    W: tokio::io::AsyncWrite + Unpin + ?Sized,
143{
144    let mut copied = 0u64;
145    let mut buf = [0u8; 8 * 1024];
146
147    loop {
148        let n = input.read(&mut buf).await?;
149        if n == 0 {
150            return Ok(copied);
151        }
152
153        output.write_all(&buf[..n]).await?;
154        copied += n as u64;
155    }
156}

158#[async_trait::async_trait]
159impl<User: UserDetail> StorageBackend<User> for OpendalStorage {
160    type Metadata = OpendalMetadata;
161
162    async fn metadata<P: AsRef<Path> + Send + Debug>(
163        &self,
164        _: &User,
165        path: P,
166    ) -> storage::Result<Self::Metadata> {
167        let metadata = self
168            .op
169            .stat(convert_path(path.as_ref())?)
170            .await
171            .map_err(convert_err)?;
172        Ok(OpendalMetadata(metadata))
173    }
174
175    async fn list<P: AsRef<Path> + Send + Debug>(
176        &self,
177        _: &User,
178        path: P,
179    ) -> storage::Result<Vec<storage::Fileinfo<PathBuf, Self::Metadata>>>
180    where
181        Self::Metadata: storage::Metadata,
182    {
183        let ret = self
184            .op
185            .list(convert_path(path.as_ref())?)
186            .await
187            .map_err(convert_err)?
188            .into_iter()
189            .map(|x| {
190                let (path, metadata) = x.into_parts();
191                storage::Fileinfo {
192                    path: path.into(),
193                    metadata: OpendalMetadata(metadata),
194                }
195            })
196            .collect();
197        Ok(ret)
198    }
199
200    async fn get<P: AsRef<Path> + Send + Debug>(
201        &self,
202        _: &User,
203        path: P,
204        start_pos: u64,
205    ) -> storage::Result<Box<dyn tokio::io::AsyncRead + Send + Sync + Unpin>> {
206        let reader = self
207            .op
208            .reader(convert_path(path.as_ref())?)
209            .await
210            .map_err(convert_err)?
211            .into_futures_async_read(start_pos..)
212            .await
213            .map_err(convert_err)?
214            .compat();
215        Ok(Box::new(reader))
216    }
217
218    async fn put<
219        P: AsRef<Path> + Send + Debug,
220        R: tokio::io::AsyncRead + Send + Sync + Unpin + 'static,
221    >(
222        &self,
223        _: &User,
224        mut input: R,
225        path: P,
226        _: u64,
227    ) -> storage::Result<u64> {
228        let mut w = self
229            .op
230            .writer(convert_path(path.as_ref())?)
231            .await
232            .map_err(convert_err)?
233            .into_futures_async_write()
234            .compat_write();
235        // Avoid `tokio::io::copy`'s pending-read flush path and keep buffering policy explicit.
236        let copy_result = copy_read_write_loop(&mut input, &mut w).await;
237        let shutdown_result = w.shutdown().await;
238        match (copy_result, shutdown_result) {
239            (Ok(len), Ok(())) => Ok(len),
240            (Err(copy_err), Ok(())) => Err(Error::new(
241                storage::ErrorKind::LocalError,
242                format!("Failed to copy data: {}", copy_err),
243            )),
244            (Ok(_), Err(shutdown_err)) => Err(Error::new(
245                storage::ErrorKind::LocalError,
246                format!("Failed to shutdown writer: {}", shutdown_err),
247            )),
248            (Err(copy_err), Err(shutdown_err)) => Err(Error::new(
249                storage::ErrorKind::LocalError,
250                format!(
251                    "Failed to copy data: {} AND failed to shutdown writer: {}",
252                    copy_err, shutdown_err
253                ),
254            )),
255        }
256    }
257
258    async fn del<P: AsRef<Path> + Send + Debug>(&self, _: &User, path: P) -> storage::Result<()> {
259        self.op
260            .delete(convert_path(path.as_ref())?)
261            .await
262            .map_err(convert_err)
263    }
264
265    async fn mkd<P: AsRef<Path> + Send + Debug>(&self, _: &User, path: P) -> storage::Result<()> {
266        let mut path_str = convert_path(path.as_ref())?.to_string();
267        if !path_str.ends_with('/') {
268            path_str.push('/');
269        }
270        self.op.create_dir(&path_str).await.map_err(convert_err)
271    }
272
273    async fn rename<P: AsRef<Path> + Send + Debug>(
274        &self,
275        _: &User,
276        from: P,
277        to: P,
278    ) -> storage::Result<()> {
279        let (from, to) = (convert_path(from.as_ref())?, convert_path(to.as_ref())?);
280        self.op.rename(from, to).await.map_err(convert_err)
281    }
282
283    async fn rmd<P: AsRef<Path> + Send + Debug>(&self, _: &User, path: P) -> storage::Result<()> {
284        self.op
285            .delete_with(convert_path(path.as_ref())?)
286            .recursive(true)
287            .await
288            .map_err(convert_err)
289    }
290
291    async fn cwd<P: AsRef<Path> + Send + Debug>(&self, _: &User, path: P) -> storage::Result<()> {
292        use opendal::ErrorKind::*;
293
294        match self.op.stat(convert_path(path.as_ref())?).await {
295            Ok(_) => Ok(()),
296            Err(e) if matches!(e.kind(), NotFound | NotADirectory) => Err(Error::new(
297                storage::ErrorKind::PermanentDirectoryNotAvailable,
298                e,
299            )),
300            Err(e) => Err(convert_err(e)),
301        }
302    }
303}
